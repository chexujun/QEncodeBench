def build_oracle(qc, problem_qubits, ancilla_qubits):
    target = ancilla_qubits[0]
    for bits in ((0, 0, 0), (1, 0, 0), (1, 0, 1), (1, 1, 0)):
        for qubit, bit in zip(problem_qubits, bits):
            if bit == 0:
                qc.x(qubit)
        qc.mcx(problem_qubits, target)
        for qubit, bit in zip(problem_qubits, bits):
            if bit == 0:
                qc.x(qubit)
