def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses=[[[3,1],[4,0],[5,1]],[[1,1],[2,0],[3,1]],[[0,0],[1,0],[2,0]],[[0,1],[1,0],[4,1]],[[1,0],[2,1],[4,1]],[[0,0],[1,1],[2,0]]]; flags=list(ancilla_qubits[:6])
    def compute_clause(clause,target):
        controls=[problem_qubits[i] for i,_ in clause]; flips=[problem_qubits[i] for i,value in clause if value==1]
        for qubit in flips: qc.x(qubit)
        qc.mcx(controls,target)
        for qubit in reversed(flips): qc.x(qubit)
    for clause,flag in zip(clauses,flags): compute_clause(clause,flag)
    for flag in flags: qc.x(flag)
    qc.h(flags[-1]); qc.mcx(flags[:-1],flags[-1]); qc.h(flags[-1])
    for flag in reversed(flags): qc.x(flag)
    for clause,flag in reversed(list(zip(clauses,flags))): compute_clause(clause,flag)
