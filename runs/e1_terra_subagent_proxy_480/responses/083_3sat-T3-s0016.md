def build_oracle(qc: QuantumCircuit, problem_qubits: list[int], ancilla_qubits: list[int]) -> None:
    x = problem_qubits
    flags = ancilla_qubits[:11]
    phase_target = ancilla_qubits[11]

    # Each tuple lists variables that must be 0 while making its clause false.
    # Variables not listed must be 1.
    clauses = [
        ([2, 5, 6], [2, 5]),       # ~x2 | x5 | ~x6
        ([0, 1, 6], [0, 1]),       # ~x0 | ~x1 | x6
        ([0, 2, 4], [0, 2, 4]),    # x0 | x2 | x4
        ([0, 3, 6], [0, 3, 6]),    # x0 | x3 | x6
        ([2, 4, 5], [2, 4]),       # x2 | x4 | ~x5
        ([3, 5, 6], [3, 6]),       # x3 | ~x5 | x6
        ([2, 5, 6], [6]),          # ~x2 | ~x5 | x6
        ([4, 5, 6], [4, 5, 6]),    # x4 | x5 | x6
        ([2, 3, 5], [2, 5]),       # x2 | ~x3 | x5
        ([0, 4, 5], [0, 5]),       # x0 | ~x4 | x5
        ([0, 1, 2], []),           # ~x0 | ~x1 | ~x2
    ]

    def compute_clause(vars_, zero_vars, target):
        for v in zero_vars:
            qc.x(x[v])
        qc.mcx([x[v] for v in vars_], target)
        for v in reversed(zero_vars):
            qc.x(x[v])

    for (vars_, zero_vars), flag in zip(clauses, flags):
        compute_clause(vars_, zero_vars, flag)

    for flag in flags:
        qc.x(flag)
    qc.x(phase_target)
    qc.mcp(3.141592653589793, flags, phase_target)
    qc.x(phase_target)
    for flag in reversed(flags):
        qc.x(flag)

    for (vars_, zero_vars), flag in reversed(list(zip(clauses, flags))):
        compute_clause(vars_, zero_vars, flag)
