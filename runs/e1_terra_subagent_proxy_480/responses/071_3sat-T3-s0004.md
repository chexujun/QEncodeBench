def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses=[[(1,False),(3,False),(5,False)],[(0,False),(1,False),(2,True)],[(1,False),(2,False),(5,False)],[(1,False),(4,True),(5,False)],[(0,False),(1,False),(5,True)],[(2,True),(4,False),(5,True)],[(2,False),(3,False),(4,False)]]
    def compute_clause(clause,target):
        positive=[problem_qubits[i] for i,is_positive in clause if is_positive]
        for qubit in positive: qc.x(qubit)
        qc.mcx([problem_qubits[i] for i,_ in clause],target)
        for qubit in positive: qc.x(qubit)
        qc.x(target)
    def uncompute_clause(clause,target):
        qc.x(target); positive=[problem_qubits[i] for i,is_positive in clause if is_positive]
        for qubit in positive: qc.x(qubit)
        qc.mcx([problem_qubits[i] for i,_ in clause],target)
        for qubit in positive: qc.x(qubit)
    clause_bits=ancilla_qubits[:7]
    for clause,bit in zip(clauses,clause_bits): compute_clause(clause,bit)
    qc.mcp(3.141592653589793,clause_bits[:-1],clause_bits[-1])
    for clause,bit in reversed(list(zip(clauses,clause_bits))): uncompute_clause(clause,bit)
