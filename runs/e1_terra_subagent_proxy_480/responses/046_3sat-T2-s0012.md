def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = problem_qubits
    c = ancilla_qubits[:6]

    clauses = [
        ([1, 3], [4]),
        ([0, 2, 3], []),
        ([1, 3], [4]),
        ([0, 2], [3]),
        ([0, 1], [2]),
        ([1], [2, 3]),
    ]

    def compute_clause(positive, negative, target):
        for i in positive:
            qc.x(x[i])
        qc.mcx([x[i] for i in positive + negative], target)
        qc.x(target)
        for i in positive:
            qc.x(x[i])

    def uncompute_clause(positive, negative, target):
        qc.x(target)
        for i in positive:
            qc.x(x[i])
        qc.mcx([x[i] for i in positive + negative], target)
        for i in positive:
            qc.x(x[i])

    for (positive, negative), target in zip(clauses, c):
        compute_clause(positive, negative, target)

    qc.mcp(3.141592653589793, c)

    for (positive, negative), target in reversed(list(zip(clauses, c))):
        uncompute_clause(positive, negative, target)
