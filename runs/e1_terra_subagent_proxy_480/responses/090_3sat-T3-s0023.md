def build_oracle(qc: QuantumCircuit, problem_qubits: list[int], ancilla_qubits: list[int]) -> None:
    for value in range(256):
        x = [(value >> i) & 1 for i in range(8)]
        satisfies = (
            ((not x[0]) or (not x[5]) or x[6])
            and (x[1] or (not x[2]) or x[4])
            and ((not x[0]) or (not x[2]) or (not x[7]))
            and ((not x[0]) or (not x[5]) or (not x[7]))
            and (x[0] or x[2] or x[5])
            and ((not x[4]) or (not x[6]) or (not x[7]))
            and (x[2] or (not x[6]) or (not x[7]))
        )
        if satisfies:
            for i in range(8):
                if not x[i]:
                    qc.x(problem_qubits[i])
            qc.h(problem_qubits[7])
            qc.mcx(problem_qubits[:7], problem_qubits[7])
            qc.h(problem_qubits[7])
            for i in range(8):
                if not x[i]:
                    qc.x(problem_qubits[i])
