import math
def build_oracle(qc, problem_qubits, ancilla_qubits):
 q=problem_qubits;c=ancilla_qubits[:8];clauses=[([1,6],[3]),([5],[3,6]),([0,4],[3]),([1,5],[3]),([3,7],[0]),([0,2,3],[]),([3,6],[4]),([5],[2,7])]
 def compute_clause(zero_literals,one_literals,target):
  for i in zero_literals: qc.x(q[i])
  qc.mcx([q[i] for i in zero_literals+one_literals],target)
  for i in reversed(zero_literals): qc.x(q[i])
  qc.x(target)
 for target,(zeros,ones) in zip(c,clauses): compute_clause(zeros,ones,target)
 qc.mcp(math.pi,c[:-1],c[-1])
 for target,(zeros,ones) in reversed(list(zip(c,clauses))): compute_clause(zeros,ones,target)
