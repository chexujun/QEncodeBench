def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses=[[[0,0],[1,0],[2,0]],[[0,1],[1,0],[3,0]],[[0,0],[2,1],[3,0]],[[0,0],[1,1],[3,0]],[[1,0],[2,1],[3,0]],[[0,0],[2,0],[3,1]]]
    clause_bits=list(ancilla_qubits[:6]); phase_bit=ancilla_qubits[6]
    def compute_clause(clause,target):
        flipped=[]
        for index,value in clause:
            if value==1: qc.x(problem_qubits[index]); flipped.append(index)
        qc.mcx([problem_qubits[index] for index,_ in clause],target)
        for index in reversed(flipped): qc.x(problem_qubits[index])
    for clause,target in zip(clauses,clause_bits): compute_clause(clause,target)
    for qubit in clause_bits: qc.x(qubit)
    qc.x(phase_bit); qc.mcp(3.141592653589793,clause_bits,phase_bit); qc.x(phase_bit)
    for qubit in clause_bits: qc.x(qubit)
    for clause,target in reversed(list(zip(clauses,clause_bits))): compute_clause(clause,target)
