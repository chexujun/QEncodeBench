def build_oracle(qc, problem_qubits, ancilla_qubits):
    flags = ancilla_qubits[:7]
    clauses = [[(1, True), (4, False), (5, True)],[(0, True), (1, True), (2, True)],[(1, True), (2, False), (5, False)],[(3, True), (4, False), (5, True)],[(0, True), (4, True), (5, False)],[(1, False), (4, False), (5, True)],[(0, False), (1, True), (2, True)]]
    def compute_clause(clause, target):
        qc.x(target)
        for index, positive in clause:
            if positive: qc.x(problem_qubits[index])
        qc.mcx([problem_qubits[index] for index, _ in clause], target)
        for index, positive in reversed(clause):
            if positive: qc.x(problem_qubits[index])
    for clause, flag in zip(clauses, flags): compute_clause(clause, flag)
    qc.h(flags[6]); qc.mcx(flags[:6], flags[6]); qc.h(flags[6])
    for clause, flag in reversed(list(zip(clauses, flags))): compute_clause(clause, flag)
