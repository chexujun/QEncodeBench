def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [
        [(0, 0), (1, 0), (2, 0)],
        [(0, 0), (1, 1), (2, 0)],
        [(0, 1), (1, 1), (4, 0)],
        [(0, 0), (2, 1), (3, 0)],
    ]
    clause_bits = ancilla_qubits[:4]

    def compute_clause(clause, target):
        qc.x(target)
        flipped = []
        for index, value in clause:
            if value == 1:
                qc.x(problem_qubits[index])
                flipped.append(problem_qubits[index])
        qc.mcx([problem_qubits[index] for index, _ in clause], target)
        for qubit in reversed(flipped):
            qc.x(qubit)

    for clause, target in zip(clauses, clause_bits):
        compute_clause(clause, target)

    qc.h(clause_bits[-1])
    qc.mcx(clause_bits[:-1], clause_bits[-1])
    qc.h(clause_bits[-1])

    for clause, target in reversed(list(zip(clauses, clause_bits))):
        compute_clause(clause, target)

    return qc
