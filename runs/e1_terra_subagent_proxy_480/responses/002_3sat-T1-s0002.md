def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0,x1,x2=problem_qubits; a0,a1,a2,a3,phase=ancilla_qubits
    qc.x(x1); qc.mcx([x0,x1,x2],a0); qc.x(x1)
    qc.x(x1); qc.x(x2); qc.mcx([x0,x1,x2],a1); qc.x(x2); qc.x(x1)
    qc.x(x0); qc.x(x1); qc.x(x2); qc.mcx([x0,x1,x2],a2); qc.x(x2); qc.x(x1); qc.x(x0)
    qc.x(x1); qc.x(x2); qc.mcx([x0,x1,x2],a3); qc.x(x2); qc.x(x1)
    qc.x(a0); qc.x(a1); qc.x(a2); qc.x(a3); qc.mcx([a0,a1,a2,a3],phase); qc.z(phase); qc.mcx([a0,a1,a2,a3],phase); qc.x(a3); qc.x(a2); qc.x(a1); qc.x(a0)
    qc.x(x1); qc.x(x2); qc.mcx([x0,x1,x2],a3); qc.x(x2); qc.x(x1)
    qc.x(x0); qc.x(x1); qc.x(x2); qc.mcx([x0,x1,x2],a2); qc.x(x2); qc.x(x1); qc.x(x0)
    qc.x(x1); qc.x(x2); qc.mcx([x0,x1,x2],a1); qc.x(x2); qc.x(x1)
    qc.x(x1); qc.mcx([x0,x1,x2],a0); qc.x(x1)
