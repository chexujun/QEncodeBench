def build_oracle(qc, problem_qubits, ancilla_qubits):
    p = problem_qubits
    a = ancilla_qubits

    qc.mcx([p[0], p[1], p[2]], a[0])

    qc.x([p[0], p[1], p[2]])
    qc.mcx([p[0], p[1], p[2]], a[1])
    qc.x([p[0], p[1], p[2]])

    qc.x([p[0], p[1]])
    qc.mcx([p[0], p[1], p[2]], a[2])
    qc.x([p[0], p[1]])

    qc.x(p[0])
    qc.mcx([p[0], p[1], p[2]], a[3])
    qc.x(p[0])

    qc.x([a[0], a[1], a[2], a[3]])
    qc.x(a[4])
    qc.h(a[4])
    qc.mcx([a[0], a[1], a[2], a[3]], a[4])
    qc.h(a[4])
    qc.x(a[4])
    qc.x([a[0], a[1], a[2], a[3]])

    qc.x(p[0])
    qc.mcx([p[0], p[1], p[2]], a[3])
    qc.x(p[0])

    qc.x([p[0], p[1]])
    qc.mcx([p[0], p[1], p[2]], a[2])
    qc.x([p[0], p[1]])

    qc.x([p[0], p[1], p[2]])
    qc.mcx([p[0], p[1], p[2]], a[1])
    qc.x([p[0], p[1], p[2]])

    qc.mcx([p[0], p[1], p[2]], a[0])
