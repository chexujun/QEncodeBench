def build_oracle(qc: QuantumCircuit, problem_qubits: list[int], ancilla_qubits: list[int]) -> None:
    clauses = [
        [(0, True), (6, True), (7, True)], [(1, False), (3, False), (7, True)],
        [(1, True), (3, False), (4, True)], [(0, True), (6, True), (7, False)],
        [(2, False), (5, False), (7, True)], [(1, True), (5, False), (6, False)],
        [(1, False), (3, True), (6, True)], [(1, False), (4, True), (5, False)],
        [(4, True), (5, False), (7, False)], [(2, False), (4, False), (7, False)],
    ]

    def compute_clause(clause, target):
        positive_vars = [problem_qubits[i] for i, is_positive in clause if is_positive]
        for qubit in positive_vars:
            qc.x(qubit)
        qc.mcx([problem_qubits[i] for i, _ in clause], target)
        qc.x(target)
        for qubit in positive_vars:
            qc.x(qubit)

    flags = ancilla_qubits[:10]
    phase_qubit = ancilla_qubits[10]
    for clause, flag in zip(clauses, flags):
        compute_clause(clause, flag)
    qc.x(phase_qubit)
    qc.h(phase_qubit)
    qc.mcx(flags, phase_qubit)
    qc.h(phase_qubit)
    qc.x(phase_qubit)
    for clause, flag in reversed(list(zip(clauses, flags))):
        compute_clause(clause, flag)
