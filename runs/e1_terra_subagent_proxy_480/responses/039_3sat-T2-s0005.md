def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2, x3, x4 = problem_qubits
    c0, c1, c2, c3, c4, c5, phase = ancilla_qubits

    def compute_bad(zeros, ones, target):
        for q in zeros:
            qc.x(q)
        qc.mcx(zeros + ones, target)
        for q in reversed(zeros):
            qc.x(q)

    def uncompute_bad(zeros, ones, target):
        for q in zeros:
            qc.x(q)
        qc.mcx(zeros + ones, target)
        for q in reversed(zeros):
            qc.x(q)

    clauses = [
        ([x0], [x1, x4], c0),
        ([x4], [x0, x1], c1),
        ([x2], [x0, x3], c2),
        ([x4], [x0, x3], c3),
        ([x3, x4], [x2], c4),
        ([x3, x4], [x1], c5),
    ]

    for zeros, ones, target in clauses:
        compute_bad(zeros, ones, target)

    qc.h(phase)
    for q in (c0, c1, c2, c3, c4, c5):
        qc.x(q)
    qc.mcx([c0, c1, c2, c3, c4, c5], phase)
    for q in reversed((c0, c1, c2, c3, c4, c5)):
        qc.x(q)
    qc.h(phase)

    for zeros, ones, target in reversed(clauses):
        uncompute_bad(zeros, ones, target)
