def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = problem_qubits
    c = ancilla_qubits

    def unsat(target, positive, negative):
        for i in positive:
            qc.x(x[i])
        qc.mcx([x[i] for i in positive + negative], target)
        for i in positive:
            qc.x(x[i])

    # Compute whether each clause is unsatisfied.
    unsat(c[0], [4], [0, 2])
    unsat(c[1], [2, 4], [5])
    unsat(c[2], [1, 2], [4])
    unsat(c[3], [1, 2], [5])
    unsat(c[4], [], [0, 1, 4])
    unsat(c[5], [0, 2, 4], [])
    unsat(c[6], [5], [0, 4])

    # Phase flip exactly when no clause is unsatisfied.
    for i in range(7):
        qc.x(c[i])
    qc.h(c[7])
    qc.mcx(c[:7], c[7])
    qc.h(c[7])
    for i in range(7):
        qc.x(c[i])

    # Uncompute clause flags.
    unsat(c[6], [5], [0, 4])
    unsat(c[5], [0, 2, 4], [])
    unsat(c[4], [], [0, 1, 4])
    unsat(c[3], [1, 2], [5])
    unsat(c[2], [1, 2], [4])
    unsat(c[1], [2, 4], [5])
    unsat(c[0], [4], [0, 2])
