def build_oracle(qc, problem_qubits, ancilla_qubits):
 clauses=[[(1,False),(3,True),(7,True)],[(0,True),(2,True),(7,True)],[(3,False),(4,True),(5,False)],[(3,False),(5,False),(6,False)],[(0,False),(1,False),(5,True)],[(4,False),(6,False),(7,True)],[(0,True),(4,True),(7,False)]];clause_bits=ancilla_qubits[:7]

 for clause,target in zip(clauses,clause_bits):compute_clause(clause,target)
