def build_oracle(qc, problem_qubits, ancilla_qubits):
    flags = ancilla_qubits[:5]
    phase = ancilla_qubits[5]

    clauses = [
        [(0, 0), (1, 1), (4, 0)],
        [(2, 1), (3, 1), (4, 1)],
        [(0, 0), (3, 1), (4, 0)],
        [(1, 0), (2, 0), (4, 1)],
        [(0, 0), (2, 0), (4, 1)],
    ]

    for flag, controls in zip(flags, clauses):
        zero_controls = [problem_qubits[i] for i, value in controls if value == 0]
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx([problem_qubits[i] for i, _ in controls], flag)
        for qubit in zero_controls:
            qc.x(qubit)

    for flag in flags:
        qc.x(flag)
    qc.x(phase)
    qc.h(phase)
    qc.mcx(flags, phase)
    qc.h(phase)
    qc.x(phase)
    for flag in flags:
        qc.x(flag)

    for flag, controls in reversed(list(zip(flags, clauses))):
        zero_controls = [problem_qubits[i] for i, value in controls if value == 0]
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx([problem_qubits[i] for i, _ in controls], flag)
        for qubit in zero_controls:
            qc.x(qubit)
