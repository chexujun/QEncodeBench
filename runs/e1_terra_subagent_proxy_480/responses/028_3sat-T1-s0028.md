def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [[(1, True), (2, False), (3, False)], [(1, False), (2, True), (3, True)], [(0, False), (2, True), (3, True)], [(0, True), (1, True), (3, False)], [(0, False), (2, True), (3, False)]]
    for clause, flag in zip(clauses, ancilla_qubits[:5]):
        positive = [index for index, value in clause if value]
        for index in positive: qc.x(problem_qubits[index])
        qc.mcx([problem_qubits[index] for index, _ in clause], flag)
        for index in reversed(positive): qc.x(problem_qubits[index])
    for flag in ancilla_qubits[:5]: qc.x(flag)
    qc.h(ancilla_qubits[0])
    qc.mcx(ancilla_qubits[1:5], ancilla_qubits[0])
    qc.h(ancilla_qubits[0])
    for flag in reversed(ancilla_qubits[:5]): qc.x(flag)
    for clause, flag in reversed(list(zip(clauses, ancilla_qubits[:5]))):
        positive = [index for index, value in clause if value]
        for index in positive: qc.x(problem_qubits[index])
        qc.mcx([problem_qubits[index] for index, _ in clause], flag)
        for index in reversed(positive): qc.x(problem_qubits[index])
