def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = problem_qubits
    a = ancilla_qubits[:6]

    clauses = [
        ([x[1], x[2], x[5]], []),
        ([x[0], x[3], x[5]], [x[0], x[3], x[5]]),
        ([x[1], x[2], x[4]], [x[1], x[2]]),
        ([x[2], x[3], x[4]], [x[2], x[3]]),
        ([x[1], x[4], x[5]], [x[4]]),
        ([x[3], x[4], x[5]], [x[5]]),
    ]

    def compute_clause(target, controls, flips):
        for qubit in flips:
            qc.x(qubit)
        qc.mcx(controls, target)
        qc.x(target)
        for qubit in reversed(flips):
            qc.x(qubit)

    def uncompute_clause(target, controls, flips):
        for qubit in flips:
            qc.x(qubit)
        qc.x(target)
        qc.mcx(controls, target)
        for qubit in reversed(flips):
            qc.x(qubit)

    for target, (controls, flips) in zip(a, clauses):
        compute_clause(target, controls, flips)

    qc.h(a[5])
    qc.mcx(a[:5], a[5])
    qc.h(a[5])

    for target, (controls, flips) in zip(reversed(a), reversed(clauses)):
        uncompute_clause(target, controls, flips)
