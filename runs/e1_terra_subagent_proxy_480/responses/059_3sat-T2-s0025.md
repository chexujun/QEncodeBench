def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = problem_qubits
    c = ancilla_qubits[:5]
    flag = ancilla_qubits[5]

    clauses = [
        [(0, 1), (1, 1), (2, 1)],
        [(0, 0), (1, 0), (2, 0)],
        [(0, 1), (2, 1), (3, 0)],
        [(0, 1), (2, 0), (4, 1)],
        [(0, 0), (1, 1), (3, 0)],
    ]

    for clause, target in zip(clauses, c):
        zeros = [i for i, value in clause if value == 0]
        for i in zeros:
            qc.x(x[i])
        qc.mcx([x[i] for i, _ in clause], target)
        for i in reversed(zeros):
            qc.x(x[i])

    qc.mcx(c, flag)
    qc.z(flag)
    qc.mcx(c, flag)

    for clause, target in reversed(list(zip(clauses, c))):
        zeros = [i for i, value in clause if value == 0]
        for i in zeros:
            qc.x(x[i])
        qc.mcx([x[i] for i, _ in clause], target)
        for i in reversed(zeros):
            qc.x(x[i])
