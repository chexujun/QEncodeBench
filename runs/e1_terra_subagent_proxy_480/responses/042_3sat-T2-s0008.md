def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [
        ((0, 0), (1, 0), (3, 1)),
        ((2, 0), (3, 1), (4, 0)),
        ((0, 1), (1, 0), (2, 1)),
        ((0, 1), (2, 1), (3, 1)),
        ((0, 0), (2, 0), (5, 1)),
        ((1, 0), (2, 0), (5, 1)),
    ]

    for clause, ancilla in zip(clauses, ancilla_qubits[:6]):
        positive = [problem_qubits[i] for i, value in clause if value == 1]
        controls = [problem_qubits[i] for i, _ in clause]
        for qubit in positive:
            qc.x(qubit)
        qc.mcx(controls, ancilla)
        qc.x(ancilla)
        for qubit in positive:
            qc.x(qubit)

    qc.h(ancilla_qubits[5])
    qc.mcx(ancilla_qubits[:5], ancilla_qubits[5])
    qc.h(ancilla_qubits[5])

    for clause, ancilla in reversed(list(zip(clauses, ancilla_qubits[:6]))):
        positive = [problem_qubits[i] for i, value in clause if value == 1]
        controls = [problem_qubits[i] for i, _ in clause]
        for qubit in positive:
            qc.x(qubit)
        qc.x(ancilla)
        qc.mcx(controls, ancilla)
        for qubit in positive:
            qc.x(qubit)
