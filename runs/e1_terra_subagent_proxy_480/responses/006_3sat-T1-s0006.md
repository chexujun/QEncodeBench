def build_oracle(qc, problem_qubits, ancilla_qubits) -> None:
    x = problem_qubits
    clauses = ancilla_qubits[:5]
    phase = ancilla_qubits[5]

    clause_specs = [
        ([0, 1, 2], [0]), ([0, 1, 2], [0, 1]),
        ([0, 1, 2], [0, 2]), ([0, 1, 2], [0]),
        ([0, 1, 2], [1]),
    ]
    for target, (_, invert) in zip(clauses, clause_specs):
        qc.x(target)
        for i in invert:
            qc.x(x[i])
        qc.mcx(x, target)
        for i in reversed(invert):
            qc.x(x[i])

    qc.x(phase)
    qc.h(phase)
    qc.mcx(clauses, phase)
    qc.h(phase)
    qc.x(phase)

    for target, (_, invert) in reversed(list(zip(clauses, clause_specs))):
        for i in invert:
            qc.x(x[i])
        qc.mcx(x, target)
        for i in reversed(invert):
            qc.x(x[i])
        qc.x(target)
