def build_oracle(qc, problem_qubits, ancilla_qubits):
    p = problem_qubits
    clauses = (
        ((0, 0), (1, 1), (4, 0)),
        ((0, 0), (1, 1), (2, 1)),
        ((0, 1), (3, 0), (4, 0)),
        ((1, 1), (2, 1), (3, 1)),
        ((0, 0), (2, 0), (4, 0)),
        ((0, 0), (2, 0), (4, 1)),
    )
    for state in range(32):
        bits = tuple((state >> i) & 1 for i in range(5))
        if all(any(bits[i] == value for i, value in clause) for clause in clauses):
            for i, bit in enumerate(bits):
                if not bit:
                    qc.x(p[i])
            qc.mcp(3.141592653589793, p[:4], p[4])
            for i, bit in enumerate(bits):
                if not bit:
                    qc.x(p[i])
