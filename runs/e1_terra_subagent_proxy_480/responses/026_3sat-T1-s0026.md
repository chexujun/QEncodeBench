def build_oracle(qc, problem_qubits, ancilla_qubits):
    q0, q1, q2 = problem_qubits
    for bits in ((1, 0, 0), (1, 0, 1), (0, 1, 0), (1, 1, 1)):
        for qubit, bit in zip((q0, q1, q2), bits):
            if bit == 0:
                qc.x(qubit)
        qc.h(q2)
        qc.mcx([q0, q1], q2)
        qc.h(q2)
        for qubit, bit in zip((q0, q1, q2), bits):
            if bit == 0:
                qc.x(qubit)
