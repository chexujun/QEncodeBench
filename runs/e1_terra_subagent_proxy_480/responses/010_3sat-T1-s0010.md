from qiskit import QuantumCircuit
from math import pi

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int], ancilla_qubits: list[int]) -> None:
    for value in range(16):
        x = [(value >> i) & 1 for i in range(4)]
        if ((x[1] or not x[2] or not x[3]) and
            (not x[0] or not x[2] or x[3]) and
            (not x[1] or x[2] or x[3]) and
            (x[0] or x[2] or x[3]) and
            (x[0] or not x[2] or not x[3])):
            zero_qubits = [problem_qubits[i] for i in range(4) if not x[i]]
            for qubit in zero_qubits:
                qc.x(qubit)
            qc.mcp(pi, problem_qubits[:3], problem_qubits[3])
            for qubit in reversed(zero_qubits):
                qc.x(qubit)
