def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses=[[(1,0),(3,0),(4,0)],[(3,0),(4,0),(5,1)],[(0,0),(2,0),(5,0)],[(0,1),(1,1),(3,0)],[(2,0),(4,1),(5,1)]]
    for value in range(64):
        bits=[(value>>i)&1 for i in range(6)]
        if all(any(bits[i]==polarity for i,polarity in clause) for clause in clauses):
            zero_qubits=[problem_qubits[i] for i in range(6) if not bits[i]]
            qc.x(zero_qubits); qc.mcp(3.141592653589793,problem_qubits); qc.x(zero_qubits)
