def build_oracle(qc: QuantumCircuit, problem_qubits: list[int], ancilla_qubits: list[int]) -> None:
    clause_bits = ancilla_qubits[:5]
    phase_bit = ancilla_qubits[5]

    clauses = [
        [(0, True), (2, False), (3, False)],
        [(0, False), (1, True), (2, True)],
        [(0, True), (1, True), (2, True)],
        [(0, False), (1, True), (3, False)],
        [(0, False), (1, False), (3, False)],
    ]

    def compute_clause(clause, target):
        qc.x(target)
        inverted = []
        for index, negated in clause:
            if not negated:
                qubit = problem_qubits[index]
                qc.x(qubit)
                inverted.append(qubit)
        qc.mcx(problem_qubits, target)
        for qubit in reversed(inverted):
            qc.x(qubit)

    for clause, target in zip(clauses, clause_bits):
        compute_clause(clause, target)

    qc.mcx(clause_bits, phase_bit)
    qc.z(phase_bit)
    qc.mcx(clause_bits, phase_bit)

    for clause, target in reversed(list(zip(clauses, clause_bits))):
        compute_clause(clause, target)
