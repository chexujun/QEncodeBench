def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [
        [(0, 1), (4, 1), (5, 0)],
        [(2, 0), (4, 1), (5, 0)],
        [(0, 1), (1, 1), (4, 0)],
        [(1, 0), (4, 1), (5, 1)],
        [(0, 1), (2, 0), (3, 0)],
        [(1, 1), (2, 1), (3, 1)],
    ]
    clause_bits = list(ancilla_qubits[:6])
    phase_bit = ancilla_qubits[6]

    for target, clause in zip(clause_bits, clauses):
        qc.x(target)
        positive_controls = [problem_qubits[i] for i, value in clause if value == 1]
        for qubit in positive_controls:
            qc.x(qubit)
        qc.mcx([problem_qubits[i] for i, _ in clause], target)
        for qubit in positive_controls:
            qc.x(qubit)

    qc.h(phase_bit)
    qc.mcx(clause_bits, phase_bit)
    qc.h(phase_bit)

    for target, clause in reversed(list(zip(clause_bits, clauses))):
        positive_controls = [problem_qubits[i] for i, value in clause if value == 1]
        for qubit in positive_controls:
            qc.x(qubit)
        qc.mcx([problem_qubits[i] for i, _ in clause], target)
        for qubit in positive_controls:
            qc.x(qubit)
        qc.x(target)
