def build_oracle(qc, problem_qubits, ancilla_qubits):
    q = problem_qubits
    a = ancilla_qubits

    clauses = [
        ([q[2], q[3], q[4]], [q[2], q[3], q[4]]),
        ([q[2], q[4], q[5]], [q[2], q[4], q[5]]),
        ([q[0], q[1], q[5]], [q[1], q[5]]),
        ([q[0], q[3], q[5]], [q[0], q[5]]),
        ([q[0], q[3], q[4]], []),
        ([q[0], q[1], q[2]], [q[2]]),
        ([q[0], q[3], q[5]], [q[3], q[5]]),
    ]

    def compute_clause(controls, flipped, target):
        for qubit in flipped:
            qc.x(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(flipped):
            qc.x(qubit)

    for index, (controls, flipped) in enumerate(clauses):
        compute_clause(controls, flipped, a[index])

    for qubit in a[:7]:
        qc.x(qubit)
    qc.x(a[7])
    qc.h(a[7])
    qc.mcx(a[:7], a[7])
    qc.h(a[7])
    qc.x(a[7])
    for qubit in reversed(a[:7]):
        qc.x(qubit)

    for index in reversed(range(len(clauses))):
        controls, flipped = clauses[index]
        compute_clause(controls, flipped, a[index])
