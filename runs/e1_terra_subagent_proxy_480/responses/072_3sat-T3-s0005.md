def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses=(((0,1),(2,1),(4,-1)),((0,-1),(3,1),(5,-1)),((2,1),(3,-1),(5,-1)),((2,-1),(3,1),(6,-1)),((2,1),(4,1),(5,-1)),((0,1),(3,-1),(7,-1)),((1,1),(2,-1),(4,1)),((2,-1),(3,-1),(5,1)),((1,-1),(2,1),(4,-1)),((0,-1),(1,1),(2,-1)))
    clause_bits=ancilla_qubits[:10]
    def compute_clause(clause,target):
        zero_controls=[problem_qubits[i] for i,sign in clause if sign>0]
        for qubit in zero_controls: qc.x(qubit)
        qc.x(target); qc.mcx([problem_qubits[i] for i,_ in clause],target)
        for qubit in reversed(zero_controls): qc.x(qubit)
    for clause,target in zip(clauses,clause_bits): compute_clause(clause,target)
    qc.h(clause_bits[-1]); qc.mcx(clause_bits[:-1],clause_bits[-1]); qc.h(clause_bits[-1])
    for clause,target in reversed(tuple(zip(clauses,clause_bits))): compute_clause(clause,target)
