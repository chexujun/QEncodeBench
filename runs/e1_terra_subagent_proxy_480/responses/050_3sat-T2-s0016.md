def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = problem_qubits
    c = ancilla_qubits[:6]
    phase = ancilla_qubits[6]
    clauses = [
        ([x[0], x[1], x[3]], [x[3]]),
        ([x[1], x[4], x[5]], [x[1], x[4]]),
        ([x[1], x[2], x[3]], [x[1], x[2], x[3]]),
        ([x[0], x[1], x[4]], []),
        ([x[0], x[1], x[2]], []),
        ([x[1], x[3], x[4]], [x[3]]),
    ]
    def compute_clause(target, controls, inverted_controls):
        for qubit in inverted_controls: qc.x(qubit)
        qc.x(target); qc.mcx(controls, target)
        for qubit in reversed(inverted_controls): qc.x(qubit)
    for target, (controls, inverted) in zip(c, clauses): compute_clause(target, controls, inverted)
    qc.x(phase); qc.h(phase); qc.mcx(c, phase); qc.h(phase); qc.x(phase)
    for target, (controls, inverted) in reversed(list(zip(c, clauses))): compute_clause(target, controls, inverted)
