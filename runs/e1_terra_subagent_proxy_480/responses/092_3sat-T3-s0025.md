from math import pi
from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    for state in range(128):
        x = [(state >> i) & 1 for i in range(7)]
        satisfied = (
            ((not x[0]) or (not x[1]) or (not x[6]))
            and ((not x[3]) or (not x[5]) or x[6])
            and (x[2] or (not x[4]) or x[6])
            and ((not x[3]) or (not x[5]) or (not x[6]))
            and (x[1] or (not x[5]) or (not x[6]))
            and (x[3] or x[4] or x[5])
            and ((not x[0]) or (not x[1]) or x[2])
        )
        if satisfied:
            zeros = [problem_qubits[i] for i in range(7) if not x[i]]
            qc.x(zeros)
            qc.mcp(pi, problem_qubits[:-1], problem_qubits[-1])
            qc.x(zeros)
