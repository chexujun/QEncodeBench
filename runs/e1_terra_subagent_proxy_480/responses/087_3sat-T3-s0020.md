def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [
        [(1, False), (3, False), (6, True)],
        [(1, True), (6, True), (7, False)],
        [(2, False), (5, True), (6, False)],
        [(2, False), (3, False), (7, False)],
        [(0, True), (1, True), (6, False)],
        [(0, False), (1, True), (3, True)],
        [(2, True), (3, False), (6, False)],
        [(1, True), (4, False), (6, False)],
        [(2, False), (6, False), (7, True)],
    ]

    def compute_clause_false(clause, target):
        zero_controls = [problem_qubits[i] for i, positive in clause if positive]
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx([problem_qubits[i] for i, _ in clause], target)
        for qubit in reversed(zero_controls):
            qc.x(qubit)

    for clause, ancilla in zip(clauses, ancilla_qubits[:9]):
        compute_clause_false(clause, ancilla)

    for ancilla in ancilla_qubits[:9]:
        qc.x(ancilla)
    qc.mcx(ancilla_qubits[:9], ancilla_qubits[9])
    qc.z(ancilla_qubits[9])
    qc.mcx(ancilla_qubits[:9], ancilla_qubits[9])
    for ancilla in reversed(ancilla_qubits[:9]):
        qc.x(ancilla)

    for clause, ancilla in reversed(list(zip(clauses, ancilla_qubits[:9]))):
        compute_clause_false(clause, ancilla)
