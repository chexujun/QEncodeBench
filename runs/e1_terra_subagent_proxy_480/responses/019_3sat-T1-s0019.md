def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [
        [(1, False), (2, True), (3, True)],
        [(0, False), (1, False), (3, False)],
        [(0, False), (1, True), (2, False)],
        [(1, False), (2, True), (3, False)],
        [(0, False), (2, False), (3, True)],
    ]

    def compute_clause(clause, target):
        for index, positive in clause:
            if positive:
                qc.x(problem_qubits[index])
        qc.mcx([problem_qubits[index] for index, positive in clause], target)
        qc.x(target)
        for index, positive in reversed(clause):
            if positive:
                qc.x(problem_qubits[index])

    for clause, target in zip(clauses, ancilla_qubits[:5]):
        compute_clause(clause, target)

    qc.mcx(ancilla_qubits[:5], ancilla_qubits[5])
    qc.z(ancilla_qubits[5])
    qc.mcx(ancilla_qubits[:5], ancilla_qubits[5])

    for clause, target in zip(reversed(clauses), reversed(ancilla_qubits[:5])):
        compute_clause(clause, target)
