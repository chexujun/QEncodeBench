def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [
        [[0, 0], [1, 1], [3, 1]],
        [[0, 1], [1, 1], [2, 0]],
        [[0, 0], [1, 1], [2, 0]],
        [[1, 0], [2, 1], [3, 1]],
        [[0, 0], [1, 0], [2, 1]],
    ]

    def compute_clause(clause, target):
        positive_controls = [problem_qubits[i] for i, value in clause if value == 1]
        for qubit in positive_controls:
            qc.x(qubit)
        qc.x(target)
        qc.mcx([problem_qubits[i] for i, _ in clause], target)
        for qubit in positive_controls:
            qc.x(qubit)

    for clause, ancilla in zip(clauses, ancilla_qubits):
        compute_clause(clause, ancilla)

    qc.h(ancilla_qubits[-1])
    qc.mcx(ancilla_qubits[:-1], ancilla_qubits[-1])
    qc.h(ancilla_qubits[-1])

    for clause, ancilla in reversed(list(zip(clauses, ancilla_qubits))):
        compute_clause(clause, ancilla)
