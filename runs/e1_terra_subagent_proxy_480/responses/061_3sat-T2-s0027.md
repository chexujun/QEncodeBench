def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [[[3,0],[4,0],[5,0]],[[1,1],[2,0],[4,1]],[[1,0],[3,0],[4,1]],[[1,1],[3,0],[4,0]],[[1,0],[3,1],[4,0]],[[0,0],[3,1],[5,0]]]
    clause_bits = ancilla_qubits[:6]
    phase_bit = ancilla_qubits[6]
    for clause, target in zip(clauses, clause_bits):
        qc.x(target); flips=[]; controls=[]
        for var, sign in clause:
            qubit=problem_qubits[var]; controls.append(qubit)
            if sign == 1: qc.x(qubit); flips.append(qubit)
        qc.mcx(controls,target)
        for qubit in reversed(flips): qc.x(qubit)
    qc.x(phase_bit); qc.h(phase_bit); qc.mcx(clause_bits,phase_bit); qc.h(phase_bit); qc.x(phase_bit)
    for clause,target in reversed(list(zip(clauses,clause_bits))):
        flips=[]; controls=[]
        for var,sign in clause:
            qubit=problem_qubits[var]; controls.append(qubit)
            if sign == 1: qc.x(qubit); flips.append(qubit)
        qc.mcx(controls,target)
        for qubit in reversed(flips): qc.x(qubit)
        qc.x(target)
