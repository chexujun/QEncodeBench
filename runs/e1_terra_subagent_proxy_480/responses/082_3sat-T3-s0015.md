from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    q = problem_qubits
    clause_flags = ancilla_qubits[:10]
    phase_flag = ancilla_qubits[10]

    clauses = [
        ([1, 2, 4], [1, 2, 4]),
        ([1, 2, 3], [1, 2, 3]),
        ([0, 5, 6], []),
        ([3, 5, 6], [3]),
        ([0, 2, 5], [5]),
        ([0, 2, 7], [0, 7]),
        ([0, 3, 7], [0]),
        ([1, 3, 5], [3, 5]),
        ([0, 2, 3], [0, 2, 3]),
        ([0, 1, 3], [0, 1, 3]),
    ]

    def compute_clause(vars_, positive_vars, target):
        qc.x(target)
        for i in positive_vars:
            qc.x(q[i])
        qc.mcx([q[i] for i in vars_], target)
        for i in reversed(positive_vars):
            qc.x(q[i])

    for flag, (vars_, positive_vars) in zip(clause_flags, clauses):
        compute_clause(vars_, positive_vars, flag)

    qc.mcx(
        clause_flags,
        phase_flag,
        ancilla_qubits=q,
        mode="v-chain-dirty",
    )
    qc.z(phase_flag)
    qc.mcx(
        clause_flags,
        phase_flag,
        ancilla_qubits=q,
        mode="v-chain-dirty",
    )

    for flag, (vars_, positive_vars) in reversed(list(zip(clause_flags, clauses))):
        compute_clause(vars_, positive_vars, flag)
