def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [
        ([0, 4, 5], "111"), ([2, 3, 4], "100"),
        ([1, 3, 5], "011"), ([0, 1, 2], "100"),
        ([1, 3, 4], "111"), ([0, 2, 4], "101"),
        ([1, 3, 4], "101"), ([0, 1, 3], "100"),
        ([0, 3, 5], "100"), ([1, 4, 5], "000"),
    ]
    for ancilla, (indices, ctrl_state) in zip(ancilla_qubits[:10], clauses):
        qc.mcx([problem_qubits[i] for i in indices], ancilla, ctrl_state=ctrl_state)
    qc.mcx(ancilla_qubits[:10], ancilla_qubits[10], ctrl_state="0" * 10)
    qc.z(ancilla_qubits[10])
    qc.mcx(ancilla_qubits[:10], ancilla_qubits[10], ctrl_state="0" * 10)
    for ancilla, (indices, ctrl_state) in reversed(list(zip(ancilla_qubits[:10], clauses))):
        qc.mcx([problem_qubits[i] for i in indices], ancilla, ctrl_state=ctrl_state)
