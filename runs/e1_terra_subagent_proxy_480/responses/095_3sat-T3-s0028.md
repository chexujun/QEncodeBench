def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = (
        ((0, True), (1, True), (7, True)),
        ((3, False), (4, True), (6, True)),
        ((0, True), (2, False), (4, False)),
        ((0, True), (1, False), (3, False)),
        ((0, False), (6, True), (7, True)),
        ((5, True), (6, False), (7, False)),
        ((0, True), (4, False), (6, True)),
        ((0, True), (6, False), (7, False)),
    )
    clause_bits = ancilla_qubits[:8]
    flag = ancilla_qubits[8]
    for clause, target in zip(clauses, clause_bits):
        qc.x(target)
        flipped = []
        for index, positive in clause:
            if positive:
                qc.x(problem_qubits[index])
                flipped.append(problem_qubits[index])
        qc.mcx([problem_qubits[index] for index, _ in clause], target)
        for qubit in reversed(flipped):
            qc.x(qubit)
    qc.mcx(clause_bits, flag)
    qc.z(flag)
    qc.mcx(clause_bits, flag)
    for clause, target in reversed(tuple(zip(clauses, clause_bits))):
        flipped = []
        for index, positive in clause:
            if positive:
                qc.x(problem_qubits[index])
                flipped.append(problem_qubits[index])
        qc.mcx([problem_qubits[index] for index, _ in clause], target)
        for qubit in reversed(flipped):
            qc.x(qubit)
        qc.x(target)
