def build_oracle(qc, problem_qubits, ancilla_qubits):
    def clause(target, controls, flipped):
        for i in flipped:
            qc.x(problem_qubits[i])
        qc.mcx([problem_qubits[i] for i in controls], target)
        for i in reversed(flipped):
            qc.x(problem_qubits[i])

    clauses = [
        ((0, 1, 5), (0, 5)),
        ((2, 3, 4), (2, 3)),
        ((1, 4, 5), (1,)),
        ((2, 4, 5), (0, 1, 2)),
        ((0, 3, 5), (0, 3)),
        ((0, 2, 6), (0, 6)),
        ((0, 2, 5), (0, 2)),
        ((1, 3, 4), (0, 1, 2, 3)),
        ((3, 4, 5), (0, 1, 2)),
        ((2, 4, 6), (0, 1, 2, 6)),
        ((2, 3, 6), (0, 2)),
    ]

    for target, (controls, flipped) in zip(ancilla_qubits[:11], clauses):
        clause(target, controls, flipped)

    qc.x(ancilla_qubits[:11])
    qc.x(ancilla_qubits[11])
    qc.h(ancilla_qubits[11])
    qc.mcx(ancilla_qubits[:11], ancilla_qubits[11])
    qc.h(ancilla_qubits[11])
    qc.x(ancilla_qubits[11])
    qc.x(ancilla_qubits[:11])

    for target, (controls, flipped) in reversed(
        list(zip(ancilla_qubits[:11], clauses))
    ):
        clause(target, controls, flipped)
