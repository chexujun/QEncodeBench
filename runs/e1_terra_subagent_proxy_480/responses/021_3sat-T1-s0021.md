def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2, x3 = problem_qubits
    target = ancilla_qubits[0]

    qc.x(x0)
    qc.x(x1)
    qc.x(x3)
    qc.mcx([x0, x1, x3], target)
    qc.x(x3)
    qc.x(x1)
    qc.x(x0)

    qc.x(x1)
    qc.x(x3)
    qc.mcx([x0, x1, x3], target)
    qc.x(x3)
    qc.x(x1)

    qc.x(x3)
    qc.mcx([x0, x1, x3], target)
    qc.x(x3)

    qc.x(x0)
    qc.mcx([x0, x1, x2, x3], target)
    qc.x(x0)

    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2, x3], target)
    qc.x(x2)
    qc.x(x1)
