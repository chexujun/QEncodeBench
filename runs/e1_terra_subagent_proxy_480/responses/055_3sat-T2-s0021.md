def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = problem_qubits
    a = ancilla_qubits
    clauses = [
        [(0, 0), (1, 1), (2, 0)],
        [(1, 0), (2, 1), (4, 1)],
        [(0, 1), (1, 0), (3, 1)],
        [(1, 0), (3, 1), (4, 0)],
        [(1, 1), (2, 0), (3, 0)],
        [(1, 0), (3, 0), (4, 1)],
    ]

    def compute_clause(clause, target):
        flips = [i for i, bit in clause if bit == 1]
        for i in flips:
            qc.x(x[i])
        qc.mcx([x[i] for i, _ in clause], target)
        for i in reversed(flips):
            qc.x(x[i])
        qc.x(target)

    def uncompute_clause(clause, target):
        qc.x(target)
        flips = [i for i, bit in clause if bit == 1]
        for i in flips:
            qc.x(x[i])
        qc.mcx([x[i] for i, _ in clause], target)
        for i in reversed(flips):
            qc.x(x[i])

    for clause, target in zip(clauses, a[:6]):
        compute_clause(clause, target)

    qc.mcx(a[:5], a[5])
    qc.z(a[5])
    qc.mcx(a[:5], a[5])

    for clause, target in reversed(list(zip(clauses, a[:6]))):
        uncompute_clause(clause, target)
