def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [
        [(1, False), (5, True),  (6, True)],
        [(1, True),  (3, True),  (5, True)],
        [(0, True),  (2, True),  (4, True)],
        [(1, True),  (2, False), (4, True)],
        [(0, True),  (2, True),  (5, False)],
        [(1, True),  (2, True),  (5, True)],
        [(2, True),  (5, True),  (6, True)],
        [(1, False), (2, False), (3, False)],
    ]

    def compute_clause(clause, target):
        inverted = []
        controls = []
        for index, positive in clause:
            qubit = problem_qubits[index]
            if positive:
                qc.x(qubit)
                inverted.append(qubit)
            controls.append(qubit)
        qc.mcx(controls, target)
        qc.x(target)
        for qubit in reversed(inverted):
            qc.x(qubit)

    def uncompute_clause(clause, target):
        inverted = []
        controls = []
        for index, positive in clause:
            qubit = problem_qubits[index]
            if positive:
                qc.x(qubit)
                inverted.append(qubit)
            controls.append(qubit)
        qc.x(target)
        qc.mcx(controls, target)
        for qubit in reversed(inverted):
            qc.x(qubit)

    clause_bits = ancilla_qubits[:8]
    phase_bit = ancilla_qubits[8]

    for clause, target in zip(clauses, clause_bits):
        compute_clause(clause, target)

    qc.h(phase_bit)
    qc.mcx(clause_bits, phase_bit)
    qc.h(phase_bit)

    for clause, target in reversed(list(zip(clauses, clause_bits))):
        uncompute_clause(clause, target)
