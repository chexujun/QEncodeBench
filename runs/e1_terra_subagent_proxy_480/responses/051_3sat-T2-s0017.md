def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [[(0,False),(2,False),(3,False)],[(0,True),(2,True),(3,True)],[(0,True),(2,True),(3,False)],[(1,True),(2,True),(3,False)],[(0,True),(1,True),(2,False)]]
    def compute_clause(clause, target):
        flipped=[problem_qubits[i] for i,positive in clause if positive]
        for q in flipped: qc.x(q)
        qc.mcx([problem_qubits[i] for i,_ in clause],target); qc.x(target)
        for q in reversed(flipped): qc.x(q)
    def uncompute_clause(clause,target):
        flipped=[problem_qubits[i] for i,positive in clause if positive]
        for q in flipped: qc.x(q)
        qc.x(target); qc.mcx([problem_qubits[i] for i,_ in clause],target)
        for q in reversed(flipped): qc.x(q)
    for clause,target in zip(clauses,ancilla_qubits): compute_clause(clause,target)
    qc.h(ancilla_qubits[-1]); qc.mcx(list(ancilla_qubits[:-1]),ancilla_qubits[-1]); qc.h(ancilla_qubits[-1])
    for clause,target in reversed(list(zip(clauses,ancilla_qubits))): uncompute_clause(clause,target)
