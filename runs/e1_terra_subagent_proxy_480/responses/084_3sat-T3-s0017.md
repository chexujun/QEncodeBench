def build_oracle(qc, problem_qubits, ancilla_qubits) -> None:
    clauses = [
        [(3, False), (5, False), (6, False)],
        [(2, False), (6, False), (7, False)],
        [(0, True), (3, True), (6, False)],
        [(1, False), (6, True), (7, True)],
        [(2, False), (4, False), (6, True)],
        [(1, True), (2, False), (5, False)],
        [(3, False), (4, True), (5, True)],
    ]

    def compute_clause(clause, target):
        zero_controls = [problem_qubits[i] for i, positive in clause if positive]
        controls = [problem_qubits[i] for i, _ in clause]
        for qubit in zero_controls:
            qc.x(qubit)
        qc.x(target)
        qc.mcx(controls, target)
        for qubit in reversed(zero_controls):
            qc.x(qubit)

    for clause, target in zip(clauses, ancilla_qubits[:7]):
        compute_clause(clause, target)

    qc.mcx(ancilla_qubits[:7], ancilla_qubits[7])
    qc.z(ancilla_qubits[7])
    qc.mcx(ancilla_qubits[:7], ancilla_qubits[7])

    for clause, target in reversed(list(zip(clauses, ancilla_qubits[:7]))):
        compute_clause(clause, target)
