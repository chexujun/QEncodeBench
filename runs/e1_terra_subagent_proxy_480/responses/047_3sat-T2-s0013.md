def build_oracle(qc, problem_qubits, ancilla_qubits):
    p=problem_qubits; a=ancilla_qubits
    def clause(t,c,z):
        for i in z: qc.x(p[i])
        qc.x(t); qc.mcx([p[i] for i in c],t)
        for i in reversed(z): qc.x(p[i])
    specs=[(a[0],[0,2,3],[0,2]),(a[1],[1,2,4],[1]),(a[2],[0,1,4],[4]),(a[3],[1,2,4],[2,4]),(a[4],[2,3,4],[2,3]),(a[5],[0,1,2],[0])]
    for s in specs: clause(*s)
    qc.x(a[6]); qc.h(a[6]); qc.mcx(a[:6],a[6]); qc.h(a[6]); qc.x(a[6])
    for s in reversed(specs): clause(*s)
