def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [
        [[2, 0], [3, 0], [5, 0]],
        [[0, 0], [3, 1], [5, 1]],
        [[1, 1], [4, 1], [5, 1]],
        [[0, 0], [1, 0], [4, 0]],
        [[0, 1], [1, 1], [2, 1]],
        [[0, 1], [1, 0], [5, 1]],
    ]
    flags = list(ancilla_qubits[:6])

    def compute_clause(clause, target):
        qc.x(target)
        inverted = []
        for index, value in clause:
            if value == 1:
                qc.x(problem_qubits[index])
                inverted.append(problem_qubits[index])
        qc.mcx([problem_qubits[index] for index, _ in clause], target)
        for qubit in reversed(inverted):
            qc.x(qubit)
        qc.x(target)

    for clause, flag in zip(clauses, flags):
        compute_clause(clause, flag)

    qc.h(flags[-1])
    qc.mcx(flags[:-1], flags[-1])
    qc.h(flags[-1])

    for clause, flag in reversed(list(zip(clauses, flags))):
        compute_clause(clause, flag)
