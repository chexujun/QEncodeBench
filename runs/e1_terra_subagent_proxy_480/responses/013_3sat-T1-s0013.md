def build_oracle(qc, problem_qubits, ancilla_qubits):
    x1 = problem_qubits[1]
    x2 = problem_qubits[2]
    x3 = problem_qubits[3]
    flag = ancilla_qubits[0]

    patterns = [(0, 0, 0), (0, 0, 1), (1, 0, 1), (1, 1, 0)]

    for a, b, c in patterns:
        if a == 0:
            qc.x(x1)
        if b == 0:
            qc.x(x2)
        if c == 0:
            qc.x(x3)
        qc.mcx([x1, x2, x3], flag)
        if c == 0:
            qc.x(x3)
        if b == 0:
            qc.x(x2)
        if a == 0:
            qc.x(x1)

    qc.z(flag)

    for a, b, c in reversed(patterns):
        if a == 0:
            qc.x(x1)
        if b == 0:
            qc.x(x2)
        if c == 0:
            qc.x(x3)
        qc.mcx([x1, x2, x3], flag)
        if c == 0:
            qc.x(x3)
        if b == 0:
            qc.x(x2)
        if a == 0:
            qc.x(x1)
