from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits, ancilla_qubits):
    x0, x1, x2 = problem_qubits
    flags = ancilla_qubits[:5]
    phase = ancilla_qubits[5]
    violations = [((x0, x2), flags[0]), ((x2,), flags[1]), ((), flags[2]), ((x0,), flags[3]), ((x0, x1), flags[4])]
    for inverted, target in violations:
        for qubit in inverted: qc.x(qubit)
        qc.mcx([x0, x1, x2], target)
        for qubit in reversed(inverted): qc.x(qubit)
    qc.x(phase); qc.h(phase)
    for flag in flags: qc.x(flag)
    qc.mcx(flags, phase)
    for flag in reversed(flags): qc.x(flag)
    qc.h(phase); qc.x(phase)
    for inverted, target in reversed(violations):
        for qubit in inverted: qc.x(qubit)
        qc.mcx([x0, x1, x2], target)
        for qubit in reversed(inverted): qc.x(qubit)
