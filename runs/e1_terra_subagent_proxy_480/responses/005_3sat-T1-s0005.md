def build_oracle(qc, problem_qubits, ancilla_qubits) -> None:
    q0, q1, q2 = problem_qubits

    for flips in ((q0, q1, q2), (q0, q1), (q0, q2), (q1, q2)):
        for qubit in flips:
            qc.x(qubit)
        qc.h(q2)
        qc.mcx([q0, q1], q2)
        qc.h(q2)
        for qubit in reversed(flips):
            qc.x(qubit)
