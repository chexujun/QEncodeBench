from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    q0, q1, q2 = problem_qubits

    def phase_on(bits: tuple[int, int, int]) -> None:
        qubits = (q0, q1, q2)
        for qubit, bit in zip(qubits, bits):
            if bit == 0:
                qc.x(qubit)
        qc.h(q2)
        qc.ccx(q0, q1, q2)
        qc.h(q2)
        for qubit, bit in zip(qubits, bits):
            if bit == 0:
                qc.x(qubit)

    for assignment in ((0, 0, 0), (0, 0, 1), (0, 1, 1), (1, 1, 1)):
        phase_on(assignment)
