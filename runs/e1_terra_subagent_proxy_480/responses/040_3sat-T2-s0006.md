def build_oracle(qc: QuantumCircuit, problem_qubits: list[int], ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3, x4 = problem_qubits
    c0, c1, c2, c3, c4, work = ancilla_qubits

    def clause(out, controls, zero_controls):
        for q in zero_controls:
            qc.x(q)
        qc.mcx(controls, out)
        for q in reversed(zero_controls):
            qc.x(q)
        qc.x(out)

    def unclause(out, controls, zero_controls):
        qc.x(out)
        for q in zero_controls:
            qc.x(q)
        qc.mcx(controls, out)
        for q in reversed(zero_controls):
            qc.x(q)

    clause(c0, [x0, x1, x3], [x0, x3])
    clause(c1, [x0, x1, x3], [x3])
    clause(c2, [x0, x1, x3], [x1, x3])
    clause(c3, [x2, x3, x4], [x3, x4])
    clause(c4, [x0, x2, x4], [x0, x2])
    qc.mcx([c0, c1, c2, c3, c4], work)
    qc.z(work)
    qc.mcx([c0, c1, c2, c3, c4], work)
    unclause(c4, [x0, x2, x4], [x0, x2])
    unclause(c3, [x2, x3, x4], [x3, x4])
    unclause(c2, [x0, x1, x3], [x1, x3])
    unclause(c1, [x0, x1, x3], [x3])
    unclause(c0, [x0, x1, x3], [x0, x3])
