def build_oracle(qc, problem_qubits, ancilla_qubits):
    solutions = []
    for value in range(16):
        x0 = (value >> 0) & 1
        x1 = (value >> 1) & 1
        x2 = (value >> 2) & 1
        x3 = (value >> 3) & 1
        if ((x0 or not x1 or x2) and
                (not x0 or not x1 or not x2) and
                (x0 or not x2 or x3) and
                (not x0 or x1 or x3)):
            solutions.append((x0, x1, x2, x3))

    target = problem_qubits[3]
    controls = problem_qubits[:3]

    for solution in solutions:
        for qubit, bit in zip(problem_qubits, solution):
            if bit == 0:
                qc.x(qubit)

        qc.h(target)
        qc.mcx(controls, target)
        qc.h(target)

        for qubit, bit in zip(problem_qubits, solution):
            if bit == 0:
                qc.x(qubit)

    return qc
