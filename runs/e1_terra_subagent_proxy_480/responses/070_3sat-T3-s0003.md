def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [
        ([1, 3, 4], [3, 4]),
        ([1, 5, 6], [1, 5]),
        ([0, 3, 4], []),
        ([0, 1, 6], []),
        ([1, 4, 6], [1, 6]),
        ([0, 1, 3], [3]),
        ([2, 5, 6], [2]),
    ]

    def compute_clause(clause, flag):
        controls, zero_controls = clause
        for i in zero_controls:
            qc.x(problem_qubits[i])
        qc.mcx([problem_qubits[i] for i in controls], flag)
        for i in zero_controls:
            qc.x(problem_qubits[i])
        qc.x(flag)

    flags = ancilla_qubits[:7]
    result = ancilla_qubits[7]

    for clause, flag in zip(clauses, flags):
        compute_clause(clause, flag)

    qc.mcx(flags, result)
    qc.z(result)
    qc.mcx(flags, result)

    for clause, flag in reversed(list(zip(clauses, flags))):
        compute_clause(clause, flag)
