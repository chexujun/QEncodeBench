from qiskit import QuantumCircuit
from itertools import product

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int], ancilla_qubits: list[int]) -> None:
    for bits in product((0, 1), repeat=4):
        x0, x1, x2, x3 = bits
        satisfied = (
            ((not x0) or x1 or (not x2)) and
            ((not x1) or x2 or (not x3)) and
            ((not x1) or (not x2) or (not x3)) and
            (x1 or x2 or (not x3)) and
            (x0 or (not x2) or (not x3))
        )
        if satisfied:
            for qubit, bit in zip(problem_qubits, bits):
                if not bit:
                    qc.x(qubit)
            qc.h(problem_qubits[3])
            qc.mcx(problem_qubits[:3], problem_qubits[3])
            qc.h(problem_qubits[3])
            for qubit, bit in zip(problem_qubits, bits):
                if not bit:
                    qc.x(qubit)
