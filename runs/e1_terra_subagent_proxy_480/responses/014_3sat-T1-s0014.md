def build_oracle(qc, problem_qubits, ancilla_qubits):
    target = ancilla_qubits[0]

    qc.x(target)
    qc.h(target)

    for assignment in ("000", "001", "010", "100", "101", "110"):
        for i, bit in enumerate(assignment):
            if bit == "0":
                qc.x(problem_qubits[i])

        qc.mcx(problem_qubits, target)

        for i, bit in enumerate(assignment):
            if bit == "0":
                qc.x(problem_qubits[i])

    qc.h(target)
    qc.x(target)
