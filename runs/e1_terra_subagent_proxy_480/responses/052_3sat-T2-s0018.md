def build_oracle(qc, problem_qubits, ancilla_qubits):
    x=problem_qubits; a=ancilla_qubits
    clauses=[([x[1],x[2],x[5]],"001"),([x[1],x[2],x[4]],"101"),([x[0],x[1],x[3]],"111"),([x[2],x[4],x[5]],"011"),([x[1],x[4],x[5]],"110")]
    for target,(controls,state) in zip(a,clauses): qc.mcx(controls,target,ctrl_state=state)
    qc.x(a[4]); qc.mcp(3.141592653589793,a[:4],a[4],ctrl_state="0000"); qc.x(a[4])
    for target,(controls,state) in reversed(list(zip(a,clauses))): qc.mcx(controls,target,ctrl_state=state)
