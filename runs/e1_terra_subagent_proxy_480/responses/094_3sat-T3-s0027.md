from itertools import product
from math import pi
from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int], ancilla_qubits: list[int]) -> None:
    q = problem_qubits
    def satisfies(x):
        x0, x1, x2, x3, x4, x5 = x
        return ((x0 or x1 or x3) and ((not x2) or (not x3) or x4)
          and ((not x1) or x2 or x4) and (x1 or x2 or (not x5)) and (x2 or x3 or x4)
          and ((not x0) or x1 or (not x3)) and ((not x1) or (not x2) or (not x5))
          and (x0 or (not x4) or x5) and ((not x0) or x2 or (not x4)))
    for assignment in product((0, 1), repeat=6):
        if satisfies(assignment):
            zero_qubits = [q[i] for i, bit in enumerate(assignment) if bit == 0]
            for qubit in zero_qubits: qc.x(qubit)
            qc.mcp(pi, q[:5], q[5])
            for qubit in reversed(zero_qubits): qc.x(qubit)
