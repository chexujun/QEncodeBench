def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = list(problem_qubits)
    c = list(ancilla_qubits[:6])

    if len(x) < 4 or len(c) < 6:
        raise ValueError("Requires 4 problem qubits and 6 clean ancilla qubits.")

    clause_data = [
        ([x[0], x[1]], [x[0], x[1], x[2]]),
        ([x[0], x[2]], [x[0], x[2], x[3]]),
        ([x[2], x[3]], [x[1], x[2], x[3]]),
        ([],             [x[0], x[2], x[3]]),
        ([x[1], x[2], x[3]], [x[0], x[1], x[2], x[3]]),
        ([x[1], x[2], x[3]], [x[1], x[2], x[3]]),
    ]

    def compute_clause(target, flip_qubits, controls):
        qc.x(target)
        for qubit in flip_qubits:
            qc.x(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(flip_qubits):
            qc.x(qubit)

    for target, (flips, controls) in zip(c, clause_data):
        compute_clause(target, flips, controls)
    qc.h(c[-1])
    qc.mcx(c[:-1], c[-1])
    qc.h(c[-1])
    for target, (flips, controls) in reversed(list(zip(c, clause_data))):
        compute_clause(target, flips, controls)
