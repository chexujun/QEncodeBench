from math import pi
from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    clauses = [
        [(1, False), (2, False), (3, True)],
        [(2, True), (3, True), (4, False)],
        [(1, True), (2, True), (5, False)],
        [(1, False), (3, True), (5, True)],
        [(0, False), (3, True), (5, False)],
        [(0, False), (1, True), (2, True)],
        [(0, False), (1, True), (3, True)],
        [(2, True), (3, True), (5, False)],
        [(3, True), (4, False), (5, True)],
        [(0, False), (2, True), (5, True)],
        [(0, True), (2, False), (3, True)],
        [(0, False), (1, True), (5, False)],
        [(0, False), (1, False), (4, True)],
    ]

    def compute_clause(clause, target):
        positive = [problem_qubits[i] for i, sign in clause if sign]
        for qubit in positive:
            qc.x(qubit)
        qc.x(target)
        qc.mcx([problem_qubits[i] for i, _ in clause], target)
        for qubit in reversed(positive):
            qc.x(qubit)

    for clause, target in zip(clauses, ancilla_qubits[:13]):
        compute_clause(clause, target)

    phase_target = ancilla_qubits[13]
    qc.x(phase_target)
    qc.h(phase_target)
    qc.mcx(ancilla_qubits[:13], phase_target)
    qc.h(phase_target)
    qc.x(phase_target)

    for clause, target in reversed(list(zip(clauses, ancilla_qubits[:13]))):
        compute_clause(clause, target)
