def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = problem_qubits
    a = ancilla_qubits

    clauses = [
        ([3, 4, 7], [True, False, False]),
        ([2, 4, 7], [True, True, False]),
        ([2, 3, 6], [True, True, True]),
        ([2, 3, 6], [True, False, True]),
        ([1, 4, 5], [False, False, True]),
        ([3, 6, 7], [True, True, True]),
        ([0, 3, 4], [False, False, False]),
    ]

    def clause_flag(indices, positive, target):
        qc.x(target)
        for index, is_positive in zip(indices, positive):
            if is_positive:
                qc.x(x[index])
        qc.mcx([x[index] for index in indices], target)
        for index, is_positive in zip(indices, positive):
            if is_positive:
                qc.x(x[index])

    for i, (indices, positive) in enumerate(clauses):
        clause_flag(indices, positive, a[i])

    qc.mcx(a[:7], a[7], mode="noancilla")
    qc.z(a[7])
    qc.mcx(a[:7], a[7], mode="noancilla")

    for i, (indices, positive) in reversed(list(enumerate(clauses))):
        clause_flag(indices, positive, a[i])
