def build_oracle(qc, problem_qubits, ancilla_qubits):
    clauses = [
        [[2, 0], [3, 1], [5, 1]],
        [[2, 1], [4, 1], [5, 1]],
        [[1, 0], [3, 1], [5, 0]],
        [[1, 0], [4, 0], [5, 1]],
        [[0, 1], [2, 0], [3, 0]],
    ]

    def compute_clause(clause, target):
        zero_controls = []
        for index, value in clause:
            if value == 1:
                qc.x(problem_qubits[index])
                zero_controls.append(problem_qubits[index])
            else:
                zero_controls.append(problem_qubits[index])
        qc.x(target)
        qc.mcx(zero_controls, target)
        for index, value in reversed(clause):
            if value == 1:
                qc.x(problem_qubits[index])

    for clause, target in zip(clauses, ancilla_qubits[:5]):
        compute_clause(clause, target)

    qc.mcp(3.141592653589793, ancilla_qubits[:4], ancilla_qubits[4])

    for clause, target in zip(reversed(clauses), reversed(ancilla_qubits[:5])):
        compute_clause(clause, target)
