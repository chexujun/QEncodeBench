def build_oracle(qc, problem_qubits, ancilla_qubits):
    a = ancilla_qubits
    clauses = [[(0, False), (1, True), (3, False)], [(0, False), (2, True), (3, True)], [(0, True), (1, False), (3, False)], [(0, True), (1, True), (2, True)], [(1, True), (2, False), (3, False)]]
    def compute_clause(clause, target):
        qc.x(target); flipped=[]; controls=[]
        for index, positive in clause:
            qubit=problem_qubits[index]; controls.append(qubit)
            if positive: qc.x(qubit); flipped.append(qubit)
        qc.mcx(controls,target)
        for qubit in reversed(flipped): qc.x(qubit)
    for clause,target in zip(clauses,a[:5]): compute_clause(clause,target)
    qc.x(a[5]); qc.h(a[5]); qc.mcx(a[:5],a[5]); qc.h(a[5]); qc.x(a[5])
    for clause,target in reversed(list(zip(clauses,a[:5]))): compute_clause(clause,target)
