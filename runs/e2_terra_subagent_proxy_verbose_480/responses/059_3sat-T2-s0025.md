from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    flags = ancilla_qubits[:5]

    # Each tuple lists positive literals in a clause.  A flag is computed as
    # 1 iff its corresponding clause is satisfied.
    clauses = (
        (0, 1, 2),  # x0 OR x1 OR x2
        (),         # NOT x0 OR NOT x1 OR NOT x2
        (0, 2),     # x0 OR x2 OR NOT x3
        (0, 4),     # x0 OR NOT x2 OR x4
        (1,),       # NOT x0 OR x1 OR NOT x3
    )
    clause_vars = (
        (0, 1, 2),
        (0, 1, 2),
        (0, 2, 3),
        (0, 2, 4),
        (0, 1, 3),
    )

    def compute_clause(flag, variables, positive_variables):
        # Start at 1, then flip to 0 exactly when all literals are false.
        qc.x(flag)
        for variable in positive_variables:
            qc.x(x[variable])
        qc.mcx([x[variable] for variable in variables], flag)
        for variable in reversed(positive_variables):
            qc.x(x[variable])

    for flag, variables, positives in zip(flags, clause_vars, clauses):
        compute_clause(flag, variables, positives)

    # Phase-flip precisely when every clause flag is 1.
    qc.mcp(3.141592653589793, flags[:-1], flags[-1])

    for flag, variables, positives in reversed(list(zip(flags, clause_vars, clauses))):
        for variable in positives:
            qc.x(x[variable])
        qc.mcx([x[variable] for variable in variables], flag)
        for variable in reversed(positives):
            qc.x(x[variable])
        qc.x(flag)
