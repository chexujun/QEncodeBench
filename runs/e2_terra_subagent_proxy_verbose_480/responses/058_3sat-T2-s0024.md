```python
import math
from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3 = problem_qubits
    a0, a1, a2, a3 = ancilla_qubits[:4]

    # Compute whether each clause is unsatisfied.
    qc.x(x0)
    qc.x(x1)
    qc.x(x3)
    qc.mcx([x0, x1, x3], a0)
    qc.x(x3)
    qc.x(x1)
    qc.x(x0)

    qc.x(x1)
    qc.mcx([x0, x1, x2], a1)
    qc.x(x1)

    qc.x(x2)
    qc.mcx([x0, x2, x3], a2)
    qc.x(x2)

    qc.x(x3)
    qc.mcx([x1, x2, x3], a3)
    qc.x(x3)

    # Mark exactly when no clause is unsatisfied.
    qc.x(a0)
    qc.x(a1)
    qc.x(a2)
    qc.x(a3)
    qc.mcp(math.pi, [a0, a1, a2], a3)
    qc.x(a3)
    qc.x(a2)
    qc.x(a1)
    qc.x(a0)

    # Uncompute clause-unsatisfied flags.
    qc.x(x3)
    qc.mcx([x1, x2, x3], a3)
    qc.x(x3)

    qc.x(x2)
    qc.mcx([x0, x2, x3], a2)
    qc.x(x2)

    qc.x(x1)
    qc.mcx([x0, x1, x2], a1)
    qc.x(x1)

    qc.x(x0)
    qc.x(x1)
    qc.x(x3)
    qc.mcx([x0, x1, x3], a0)
    qc.x(x3)
    qc.x(x1)
    qc.x(x0)
```
