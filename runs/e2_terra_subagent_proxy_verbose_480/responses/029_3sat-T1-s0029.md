def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2 = problem_qubits
    c1, c2, c3, c4 = ancilla_qubits[:4]

    # Compute whether each clause is violated.
    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c1)
    qc.x(x2)
    qc.x(x1)

    qc.x(x0)
    qc.mcx([x0, x1, x2], c2)
    qc.x(x0)

    qc.x(x2)
    qc.mcx([x0, x1, x2], c3)
    qc.x(x2)

    qc.x(x1)
    qc.mcx([x0, x1, x2], c4)
    qc.x(x1)

    # Mark assignments with no violated clauses.
    qc.x(c1)
    qc.x(c2)
    qc.x(c3)
    qc.x(c4)
    qc.mcp(3.141592653589793, [c1, c2, c3], c4)
    qc.x(c4)
    qc.x(c3)
    qc.x(c2)
    qc.x(c1)

    # Uncompute clause-violation ancillas.
    qc.x(x1)
    qc.mcx([x0, x1, x2], c4)
    qc.x(x1)

    qc.x(x2)
    qc.mcx([x0, x1, x2], c3)
    qc.x(x2)

    qc.x(x0)
    qc.mcx([x0, x1, x2], c2)
    qc.x(x0)

    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c1)
    qc.x(x2)
    qc.x(x1)
