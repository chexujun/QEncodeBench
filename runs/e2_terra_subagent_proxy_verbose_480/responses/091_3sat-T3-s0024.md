from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    # One ancilla per clause stores whether that clause is violated.
    clause_bits = ancilla_qubits[:10]
    flag = ancilla_qubits[10]

    # Each tuple lists (variable index, is_positive_literal).  A clause is
    # violated exactly when every positive literal has value 0 and every
    # negated literal has value 1.
    clauses = [
        [(0, True),  (6, True),  (7, True)],
        [(1, False), (3, False), (7, True)],
        [(1, True),  (3, False), (4, True)],
        [(0, True),  (6, True),  (7, False)],
        [(2, False), (5, False), (7, True)],
        [(1, True),  (5, False), (6, False)],
        [(1, False), (3, True),  (6, True)],
        [(1, False), (4, True),  (5, False)],
        [(4, True),  (5, False), (7, False)],
        [(2, False), (4, False), (7, False)],
    ]

    def compute_clause_violations():
        for target, clause in zip(clause_bits, clauses):
            positive_controls = [
                problem_qubits[index]
                for index, is_positive in clause
                if is_positive
            ]
            for qubit in positive_controls:
                qc.x(qubit)

            qc.mcx([problem_qubits[index] for index, _ in clause], target)

            for qubit in reversed(positive_controls):
                qc.x(qubit)

    compute_clause_violations()

    # Convert "clause violated" flags into "clause satisfied" controls.
    for qubit in clause_bits:
        qc.x(qubit)

    # The problem qubits are valid dirty work qubits for this reversible MCX.
    qc.mcx(clause_bits, flag, ancilla_qubits=problem_qubits,
           mode="v-chain-dirty")
    qc.z(flag)
    qc.mcx(clause_bits, flag, ancilla_qubits=problem_qubits,
           mode="v-chain-dirty")

    for qubit in reversed(clause_bits):
        qc.x(qubit)

    # Uncompute the clause-violation workspace.
    for target, clause in reversed(list(zip(clause_bits, clauses))):
        positive_controls = [
            problem_qubits[index]
            for index, is_positive in clause
            if is_positive
        ]
        for qubit in positive_controls:
            qc.x(qubit)

        qc.mcx([problem_qubits[index] for index, _ in clause], target)

        for qubit in reversed(positive_controls):
            qc.x(qubit)
