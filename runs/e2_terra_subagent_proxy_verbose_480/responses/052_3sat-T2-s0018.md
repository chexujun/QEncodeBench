import math
from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    a = ancilla_qubits

    def violation(controls, flipped, target):
        for q in flipped:
            qc.x(x[q])
        qc.mcx([x[q] for q in controls], a[target])
        for q in reversed(flipped):
            qc.x(x[q])

    # Compute whether each clause is violated.
    violation([1, 2, 5], [1, 2, 5], 0)
    violation([1, 2, 4], [2], 1)
    violation([0, 1, 3], [], 2)
    violation([2, 4, 5], [2], 3)
    violation([1, 4, 5], [1, 4], 4)

    # Convert violation bits to clause-satisfaction bits.
    for i in range(5):
        qc.x(a[i])

    # Phase-flip exactly when all clauses are satisfied.
    qc.mcp(math.pi, a[:5])

    # Uncompute.
    for i in reversed(range(5)):
        qc.x(a[i])

    violation([1, 4, 5], [1, 4], 4)
    violation([2, 4, 5], [2], 3)
    violation([0, 1, 3], [], 2)
    violation([1, 2, 4], [2], 1)
    violation([1, 2, 5], [1, 2, 5], 0)
