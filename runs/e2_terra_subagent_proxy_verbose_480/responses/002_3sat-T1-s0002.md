def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2 = problem_qubits
    c1, c2, c3, c4, flag = ancilla_qubits

    def clause_violation(target, negative_controls):
        for qubit in negative_controls:
            qc.x(qubit)
        qc.mcx([x0, x1, x2], target)
        for qubit in reversed(negative_controls):
            qc.x(qubit)

    # Compute whether each clause is violated.
    clause_violation(c1, [x1])          # x0 AND !x1 AND x2
    clause_violation(c2, [x2])          # x0 AND x1 AND !x2
    clause_violation(c3, [x0, x1, x2])  # !x0 AND !x1 AND !x2
    clause_violation(c4, [x1, x2])      # x0 AND !x1 AND !x2

    # The formula is satisfied iff none of the clause-violation bits is set.
    for clause_bit in [c1, c2, c3, c4]:
        qc.x(clause_bit)
    qc.mcx([c1, c2, c3, c4], flag)
    for clause_bit in reversed([c1, c2, c3, c4]):
        qc.x(clause_bit)

    qc.z(flag)

    # Uncompute satisfaction flag and clause-violation bits.
    for clause_bit in [c1, c2, c3, c4]:
        qc.x(clause_bit)
    qc.mcx([c1, c2, c3, c4], flag)
    for clause_bit in reversed([c1, c2, c3, c4]):
        qc.x(clause_bit)

    clause_violation(c4, [x1, x2])
    clause_violation(c3, [x0, x1, x2])
    clause_violation(c2, [x2])
    clause_violation(c1, [x1])
