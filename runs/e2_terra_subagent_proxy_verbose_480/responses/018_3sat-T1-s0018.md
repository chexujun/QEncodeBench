```python
from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    q0, q1, q2 = problem_qubits
    c1, c2, c3, c4, c5, marker = ancilla_qubits

    # Compute whether each clause is violated.
    qc.x(q1)
    qc.mcx([q0, q1, q2], c1)  # (not x0 or x1 or not x2)
    qc.x(q1)

    qc.mcx([q0, q1, q2], c2)  # (not x0 or not x1 or not x2)

    qc.x(q2)
    qc.mcx([q0, q1, q2], c3)  # (not x0 or not x1 or x2)
    qc.x(q2)

    qc.x(q0)
    qc.x(q2)
    qc.mcx([q0, q1, q2], c4)  # (x0 or not x1 or x2)
    qc.x(q2)
    qc.x(q0)

    qc.x(q0)
    qc.x(q1)
    qc.x(q2)
    qc.mcx([q0, q1, q2], c5)  # (x0 or x1 or x2)
    qc.x(q2)
    qc.x(q1)
    qc.x(q0)

    # Mark iff no clause was violated.
    qc.x(c1)
    qc.x(c2)
    qc.x(c3)
    qc.x(c4)
    qc.x(c5)
    qc.mcx([c1, c2, c3, c4, c5], marker)
    qc.z(marker)
    qc.mcx([c1, c2, c3, c4, c5], marker)
    qc.x(c5)
    qc.x(c4)
    qc.x(c3)
    qc.x(c2)
    qc.x(c1)

    # Uncompute clause-violation ancillas.
    qc.x(q0)
    qc.x(q1)
    qc.x(q2)
    qc.mcx([q0, q1, q2], c5)
    qc.x(q2)
    qc.x(q1)
    qc.x(q0)

    qc.x(q0)
    qc.x(q2)
    qc.mcx([q0, q1, q2], c4)
    qc.x(q2)
    qc.x(q0)

    qc.x(q2)
    qc.mcx([q0, q1, q2], c3)
    qc.x(q2)

    qc.mcx([q0, q1, q2], c2)

    qc.x(q1)
    qc.mcx([q0, q1, q2], c1)
    qc.x(q1)
```
