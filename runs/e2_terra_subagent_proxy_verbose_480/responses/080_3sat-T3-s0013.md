def build_oracle(qc, problem_qubits, ancilla_qubits):
    clause_flags = ancilla_qubits[:10]
    phase_ancilla = ancilla_qubits[10]

    # Each tuple describes the variable values that make its clause false.
    unsatisfied_clauses = [
        ((0, 2, 6), (1, 1, 1)),
        ((1, 4, 7), (1, 0, 1)),
        ((0, 2, 4), (1, 1, 0)),
        ((2, 6, 7), (1, 0, 0)),
        ((4, 5, 6), (1, 1, 0)),
        ((1, 5, 7), (1, 0, 1)),
        ((0, 2, 7), (0, 0, 1)),
        ((0, 4, 7), (0, 1, 0)),
        ((0, 1, 7), (0, 0, 1)),
        ((3, 6, 7), (1, 1, 1)),
    ]

    def toggle_clause_flag(variables, required_values, target):
        zero_controls = [
            problem_qubits[i]
            for i, value in zip(variables, required_values)
            if value == 0
        ]
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx([problem_qubits[i] for i in variables], target)
        for qubit in reversed(zero_controls):
            qc.x(qubit)

    # Compute one flag per unsatisfied clause.
    for (variables, values), flag in zip(unsatisfied_clauses, clause_flags):
        toggle_clause_flag(variables, values, flag)

    # The formula is satisfied exactly when every clause flag is zero.
    for flag in clause_flags:
        qc.x(flag)
    qc.mcx(clause_flags, phase_ancilla)
    qc.z(phase_ancilla)
    qc.mcx(clause_flags, phase_ancilla)
    for flag in reversed(clause_flags):
        qc.x(flag)

    # Uncompute all clause flags.
    for (variables, values), flag in reversed(
        list(zip(unsatisfied_clauses, clause_flags))
    ):
        toggle_clause_flag(variables, values, flag)
