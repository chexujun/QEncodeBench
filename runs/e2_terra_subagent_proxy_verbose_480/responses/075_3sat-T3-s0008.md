from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    clauses = [
        ((0, False), (2, False), (5, False)),
        ((1, False), (4, True),  (5, False)),
        ((1, False), (2, True),  (6, False)),
        ((1, False), (3, False), (4, True)),
        ((3, True),  (4, True),  (5, False)),
        ((2, True),  (5, False), (6, True)),
        ((2, True),  (3, False), (5, True)),
        ((0, False), (4, True),  (5, False)),
        ((1, False), (2, False), (4, True)),
        ((1, True),  (4, False), (5, True)),
    ]

    accumulators = ancilla_qubits[:10]
    clause_workspace = ancilla_qubits[10]

    def toggle_clause_value(target: int, clause) -> None:
        # Toggle target by the clause's truth value:
        # target ^= (literal_1 OR literal_2 OR literal_3).
        qc.x(target)

        controls = []
        positive_controls = []
        for variable, is_positive in clause:
            qubit = problem_qubits[variable]
            controls.append(qubit)
            if is_positive:
                # A positive literal is false when its variable is zero.
                qc.x(qubit)
                positive_controls.append(qubit)

        qc.mcx(controls, target)

        for qubit in reversed(positive_controls):
            qc.x(qubit)

    # Compute the first clause into the first accumulator.
    toggle_clause_value(accumulators[0], clauses[0])

    # Build a reversible prefix AND of all clause values.
    for i in range(1, len(clauses)):
        toggle_clause_value(clause_workspace, clauses[i])
        qc.ccx(accumulators[i - 1], clause_workspace, accumulators[i])
        toggle_clause_value(clause_workspace, clauses[i])

    # Apply the phase exactly when every clause is satisfied.
    qc.z(accumulators[-1])

    # Uncompute the prefix AND and every clause workspace value.
    for i in range(len(clauses) - 1, 0, -1):
        toggle_clause_value(clause_workspace, clauses[i])
        qc.ccx(accumulators[i - 1], clause_workspace, accumulators[i])
        toggle_clause_value(clause_workspace, clauses[i])

    toggle_clause_value(accumulators[0], clauses[0])
