def build_oracle(qc, problem_qubits, ancilla_qubits):
    clause_flags = ancilla_qubits[:6]
    result = ancilla_qubits[6]

    # Each entry gives the variable values that make its clause false.
    unsatisfied_patterns = [
        [(1, 1), (2, 1), (3, 0)],
        [(0, 1), (1, 1), (3, 1)],
        [(1, 1), (2, 0), (3, 0)],
        [(0, 1), (1, 0), (2, 1)],
        [(1, 0), (2, 0), (3, 1)],
        [(0, 0), (1, 0), (3, 0)],
    ]

    def compute_clause_flag(pattern, flag):
        flipped = []
        controls = []
        for index, required_value in pattern:
            qubit = problem_qubits[index]
            if required_value == 0:
                qc.x(qubit)
                flipped.append(qubit)
            controls.append(qubit)
        qc.mcx(controls, flag)
        for qubit in reversed(flipped):
            qc.x(qubit)

    # Compute a flag for every unsatisfied clause.
    for pattern, flag in zip(unsatisfied_patterns, clause_flags):
        compute_clause_flag(pattern, flag)

    # The formula is satisfied precisely when every clause flag is zero.
    for flag in clause_flags:
        qc.x(flag)
    qc.mcx(clause_flags, result)
    qc.z(result)
    qc.mcx(clause_flags, result)
    for flag in reversed(clause_flags):
        qc.x(flag)

    # Uncompute the clause flags.
    for pattern, flag in reversed(list(zip(unsatisfied_patterns, clause_flags))):
        compute_clause_flag(pattern, flag)
