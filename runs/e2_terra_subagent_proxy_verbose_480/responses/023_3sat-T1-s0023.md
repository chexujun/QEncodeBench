def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2, x3 = problem_qubits
    c0, c1, c2, c3, c4, flag = ancilla_qubits

    # Compute whether each clause is false.
    qc.x(x3)
    qc.ccx(x0, x2, c0)
    qc.x(x3)

    qc.x(x0)
    qc.x(x2)
    qc.x(x3)
    qc.ccx(x0, x2, c1)
    qc.x(x3)
    qc.x(x2)
    qc.x(x0)

    qc.x(x0)
    qc.x(x3)
    qc.ccx(x0, x2, c2)
    qc.x(x3)
    qc.x(x0)

    qc.x(x0)
    qc.ccx(x0, x1, c3)
    qc.x(x0)

    qc.ccx(x0, x2, c4)

    # flag = 1 exactly when no clause is false.
    for c in (c0, c1, c2, c3, c4):
        qc.x(c)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    for c in (c0, c1, c2, c3, c4):
        qc.x(c)

    qc.z(flag)

    # Uncompute flag.
    for c in (c0, c1, c2, c3, c4):
        qc.x(c)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    for c in (c0, c1, c2, c3, c4):
        qc.x(c)

    # Uncompute clause-false bits.
    qc.ccx(x0, x2, c4)

    qc.x(x0)
    qc.ccx(x0, x1, c3)
    qc.x(x0)

    qc.x(x0)
    qc.x(x3)
    qc.ccx(x0, x2, c2)
    qc.x(x3)
    qc.x(x0)

    qc.x(x0)
    qc.x(x2)
    qc.x(x3)
    qc.ccx(x0, x2, c1)
    qc.x(x3)
    qc.x(x2)
    qc.x(x0)

    qc.x(x3)
    qc.ccx(x0, x2, c0)
    qc.x(x3)
