from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    flags = ancilla_qubits[:6]

    clauses = [
        ([x[0], x[1], x[4]], [x[1]]),          # x0 & ~x1 & x4
        ([x[0], x[1], x[2]], [x[1], x[2]]),    # x0 & ~x1 & ~x2
        ([x[0], x[3], x[4]], [x[0]]),          # ~x0 & x3 & x4
        ([x[1], x[2], x[3]], [x[1], x[2], x[3]]),  # ~x1 & ~x2 & ~x3
        ([x[0], x[2], x[4]], []),              # x0 & x2 & x4
        ([x[0], x[2], x[4]], [x[4]]),          # x0 & x2 & ~x4
    ]

    # Compute one flag per unsatisfied clause.
    for (controls, inverted_controls), flag in zip(clauses, flags):
        for qubit in inverted_controls:
            qc.x(qubit)
        qc.mcx(controls, flag)
        for qubit in reversed(inverted_controls):
            qc.x(qubit)

    # Flip phase iff every unsatisfied-clause flag is zero.
    for flag in flags:
        qc.x(flag)
    qc.h(flags[-1])
    qc.mcx(flags[:-1], flags[-1])
    qc.h(flags[-1])
    for flag in reversed(flags):
        qc.x(flag)

    # Uncompute clause flags.
    for (controls, inverted_controls), flag in reversed(list(zip(clauses, flags))):
        for qubit in inverted_controls:
            qc.x(qubit)
        qc.mcx(controls, flag)
        for qubit in reversed(inverted_controls):
            qc.x(qubit)
