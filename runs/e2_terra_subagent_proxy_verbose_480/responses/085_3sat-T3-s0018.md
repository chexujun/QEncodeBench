import math
from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    # Algebraic normal form of the SAT predicate, excluding its constant
    # term (which only contributes an irrelevant global phase).
    terms = [
        (0,), (1,), (0, 1), (2,), (0, 2), (1, 2), (0, 1, 2),
        (1, 3), (0, 1, 3), (2, 3), (1, 2, 3),
        (4,), (0, 4), (1, 4), (0, 1, 4), (2, 4), (0, 2, 4),
        (1, 2, 4), (0, 1, 2, 4), (3, 4), (0, 3, 4), (1, 3, 4),
        (0, 1, 3, 4), (2, 3, 4), (0, 2, 3, 4), (1, 2, 3, 4),
        (5,), (0, 5), (1, 5), (0, 1, 5), (2, 5), (0, 2, 5),
        (1, 2, 5), (0, 1, 2, 5), (0, 2, 3, 5), (0, 1, 2, 3, 5),
        (4, 5), (0, 4, 5), (1, 4, 5), (0, 1, 4, 5), (2, 4, 5),
        (0, 2, 4, 5), (1, 2, 4, 5), (0, 1, 2, 4, 5), (3, 4, 5),
        (0, 3, 4, 5), (1, 3, 4, 5), (2, 3, 4, 5),
        (0, 2, 3, 4, 5), (1, 2, 3, 4, 5),
    ]

    for term in terms:
        qubits = [problem_qubits[i] for i in term]
        if len(qubits) == 1:
            qc.z(qubits[0])
        else:
            qc.mcp(math.pi, qubits[:-1], qubits[-1])
