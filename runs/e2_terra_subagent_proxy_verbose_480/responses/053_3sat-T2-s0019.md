import math
from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3 = problem_qubits

    # Satisfying assignments are the disjoint union of:
    #   x1 & x0
    #   x1 & ~x0 & x2 & x3
    #   ~x1 & ~x2 & x3

    qc.cz(x0, x1)

    qc.x(x0)
    qc.mcp(math.pi, [x1, x2, x3], x0)
    qc.x(x0)

    qc.x(x1)
    qc.x(x2)
    qc.mcp(math.pi, [x1, x2], x3)
    qc.x(x2)
    qc.x(x1)
