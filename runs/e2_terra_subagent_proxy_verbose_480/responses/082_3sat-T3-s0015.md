from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    clause_ancillas = ancilla_qubits[:10]
    phase_ancilla = ancilla_qubits[10]

    # Each entry is (variable_index, is_positive_literal).
    clauses = [
        [(1, True),  (2, True),  (4, True)],
        [(1, True),  (2, True),  (3, True)],
        [(0, False), (5, False), (6, False)],
        [(3, True),  (5, False), (6, False)],
        [(0, False), (2, False), (5, True)],
        [(0, True),  (2, False), (7, True)],
        [(0, True),  (3, False), (7, False)],
        [(1, False), (3, True),  (5, True)],
        [(0, True),  (2, True),  (3, True)],
        [(0, True),  (1, True),  (3, True)],
    ]

    def compute_unsatisfied(clause, target):
        # A clause is unsatisfied precisely when every positive literal is 0
        # and every negated literal is 1.
        flipped = []
        controls = []
        for variable, is_positive in clause:
            qubit = problem_qubits[variable]
            if is_positive:
                qc.x(qubit)
                flipped.append(qubit)
            controls.append(qubit)
        qc.ccx(controls[0], controls[1], target)
        qc.cx(controls[2], target)
        qc.ccx(controls[0], controls[1], target)
        qc.cx(controls[2], target)
        # The above four gates are equivalent to a three-controlled NOT only
        # when target is initially zero; use the standard MCX directly instead.
        # Undo this temporary identity before applying the actual computation.
        for qubit in reversed(flipped):
            qc.x(qubit)

    # Compute one ancilla flag per clause: 1 iff that clause is unsatisfied.
    for clause, target in zip(clauses, clause_ancillas):
        flipped = []
        controls = []
        for variable, is_positive in clause:
            qubit = problem_qubits[variable]
            if is_positive:
                qc.x(qubit)
                flipped.append(qubit)
            controls.append(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(flipped):
            qc.x(qubit)

    # All clauses are satisfied iff all unsatisfied-clause flags are zero.
    for qubit in clause_ancillas:
        qc.x(qubit)
    qc.mcx(clause_ancillas, phase_ancilla)
    qc.z(phase_ancilla)
    qc.mcx(clause_ancillas, phase_ancilla)
    for qubit in reversed(clause_ancillas):
        qc.x(qubit)

    # Uncompute every clause flag.
    for clause, target in reversed(list(zip(clauses, clause_ancillas))):
        flipped = []
        controls = []
        for variable, is_positive in clause:
            qubit = problem_qubits[variable]
            if is_positive:
                qc.x(qubit)
                flipped.append(qubit)
            controls.append(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(flipped):
            qc.x(qubit)
