from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    clause_ancillas = ancilla_qubits[:11]
    result_ancilla = ancilla_qubits[11]

    # Each tuple gives the variable values that make its clause false.
    # The corresponding ancilla is toggled to 1 exactly when the clause is true.
    false_patterns = [
        ((4, 1), (6, 0), (7, 0)),
        ((2, 1), (3, 1), (5, 1)),
        ((0, 1), (6, 1), (7, 1)),
        ((1, 1), (2, 0), (7, 0)),
        ((4, 0), (5, 1), (6, 1)),
        ((2, 1), (4, 1), (5, 1)),
        ((1, 1), (4, 1), (5, 0)),
        ((3, 1), (4, 0), (6, 1)),
        ((1, 1), (3, 0), (7, 0)),
        ((3, 1), (4, 0), (5, 0)),
        ((0, 0), (4, 1), (7, 0)),
    ]

    def compute_clause(pattern, target):
        zero_controls = []
        controls = []

        for variable, required_value in pattern:
            qubit = problem_qubits[variable]
            controls.append(qubit)
            if required_value == 0:
                qc.x(qubit)
                zero_controls.append(qubit)

        # Toggle for the false pattern, then invert to obtain clause truth.
        qc.mcx(controls, target)

        for qubit in reversed(zero_controls):
            qc.x(qubit)

        qc.x(target)

    # Compute whether every individual clause is satisfied.
    for pattern, target in zip(false_patterns, clause_ancillas):
        compute_clause(pattern, target)

    # Mark exactly when all clauses are true.
    qc.mcx(clause_ancillas, result_ancilla)
    qc.z(result_ancilla)
    qc.mcx(clause_ancillas, result_ancilla)

    # Uncompute all clause ancillas.
    for pattern, target in reversed(list(zip(false_patterns, clause_ancillas))):
        compute_clause(pattern, target)
