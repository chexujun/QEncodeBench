import math
from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    clause_ancillas = ancilla_qubits[:7]
    flag = ancilla_qubits[7]

    # Each entry lists the variables that must be |0> for its clause
    # to be false.  Other clause variables must be |1>.
    clauses = [
        ([6], [0, 5]),       # (~x0 | ~x5 | x6)
        ([1, 4], [2]),       # (x1 | ~x2 | x4)
        ([], [0, 2, 7]),     # (~x0 | ~x2 | ~x7)
        ([], [0, 5, 7]),     # (~x0 | ~x5 | ~x7)
        ([0, 2, 5], []),     # (x0 | x2 | x5)
        ([], [4, 6, 7]),     # (~x4 | ~x6 | ~x7)
        ([2], [6, 7]),       # (x2 | ~x6 | ~x7)
    ]

    def compute_clause_satisfied(zero_vars, one_vars, target):
        # Toggle target iff the clause is satisfied.
        # First toggle on the unique falsifying assignment, then invert.
        for i in zero_vars:
            qc.x(problem_qubits[i])
        qc.mcx(
            [problem_qubits[i] for i in zero_vars + one_vars],
            target,
        )
        for i in reversed(zero_vars):
            qc.x(problem_qubits[i])
        qc.x(target)

    for (zero_vars, one_vars), target in zip(clauses, clause_ancillas):
        compute_clause_satisfied(zero_vars, one_vars, target)

    qc.mcx(clause_ancillas, flag)
    qc.z(flag)
    qc.mcx(clause_ancillas, flag)

    for (zero_vars, one_vars), target in reversed(list(zip(clauses, clause_ancillas))):
        compute_clause_satisfied(zero_vars, one_vars, target)
