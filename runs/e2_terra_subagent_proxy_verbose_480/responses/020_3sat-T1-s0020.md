from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit, problem_qubits: list[int], ancilla_qubits: list[int]
) -> None:
    x0, x1, x2 = problem_qubits
    clauses = ancilla_qubits[:5]

    # Compute each clause into its own ancilla.
    # For a clause l0 OR l1 OR l2, begin at 1 and toggle only when every
    # literal is false.
    clause_specs = [
        ([x1, x2], clauses[0]),      # ~x0 OR x1 OR x2
        ([x0], clauses[1]),          # x0 OR ~x1 OR ~x2
        ([x0, x1, x2], clauses[2]),  # x0 OR x1 OR x2
        ([x0, x2], clauses[3]),      # x0 OR ~x1 OR x2
        ([], clauses[4]),            # ~x0 OR ~x1 OR ~x2
    ]

    for flip_controls, target in clause_specs:
        qc.x(target)
        for qubit in flip_controls:
            qc.x(qubit)
        qc.mcx([x0, x1, x2], target)
        for qubit in reversed(flip_controls):
            qc.x(qubit)

    # Apply -1 iff every clause ancilla is 1.
    qc.h(clauses[4])
    qc.mcx(clauses[:4], clauses[4])
    qc.h(clauses[4])

    # Uncompute clause values.
    for flip_controls, target in reversed(clause_specs):
        for qubit in flip_controls:
            qc.x(qubit)
        qc.mcx([x0, x1, x2], target)
        for qubit in reversed(flip_controls):
            qc.x(qubit)
        qc.x(target)
