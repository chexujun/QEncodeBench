from qiskit import QuantumCircuit
import math

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    q0, q1, q2, q3, q4 = problem_qubits
    c0, c1, c2, c3, c4, flag = ancilla_qubits

    # Compute one ancilla per clause: ancilla is 1 exactly when its clause
    # is unsatisfied.
    qc.x(q0)
    qc.x(q4)
    qc.mcx([q0, q1, q4], c0)
    qc.x(q4)
    qc.x(q0)

    qc.mcx([q2, q3, q4], c1)

    qc.x(q0)
    qc.x(q4)
    qc.mcx([q0, q3, q4], c2)
    qc.x(q4)
    qc.x(q0)

    qc.x(q1)
    qc.x(q2)
    qc.mcx([q1, q2, q4], c3)
    qc.x(q2)
    qc.x(q1)

    qc.x(q0)
    qc.x(q2)
    qc.mcx([q0, q2, q4], c4)
    qc.x(q2)
    qc.x(q0)

    # All clauses are satisfied iff all clause-failure bits are zero.
    qc.x(c0)
    qc.x(c1)
    qc.x(c2)
    qc.x(c3)
    qc.x(c4)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    qc.z(flag)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    qc.x(c4)
    qc.x(c3)
    qc.x(c2)
    qc.x(c1)
    qc.x(c0)

    # Uncompute the clause-failure ancillas.
    qc.x(q0)
    qc.x(q2)
    qc.mcx([q0, q2, q4], c4)
    qc.x(q2)
    qc.x(q0)

    qc.x(q1)
    qc.x(q2)
    qc.mcx([q1, q2, q4], c3)
    qc.x(q2)
    qc.x(q1)

    qc.x(q0)
    qc.x(q4)
    qc.mcx([q0, q3, q4], c2)
    qc.x(q4)
    qc.x(q0)

    qc.mcx([q2, q3, q4], c1)

    qc.x(q0)
    qc.x(q4)
    qc.mcx([q0, q1, q4], c0)
    qc.x(q4)
    qc.x(q0)
