import math
from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    p = problem_qubits
    clause_ancillas = ancilla_qubits[:8]

    # Each tuple describes one clause's unsatisfying assignment:
    # (variables required to be 0, variables required to be 1).
    unsatisfied_patterns = [
        ([5, 6], [1]),       # ~x1 OR x5 OR x6
        ([1, 3, 5], []),     # x1 OR x3 OR x5
        ([0, 2, 4], []),     # x0 OR x2 OR x4
        ([1, 4], [2]),       # x1 OR ~x2 OR x4
        ([0, 2], [5]),       # x0 OR x2 OR ~x5
        ([1, 2, 5], []),     # x1 OR x2 OR x5
        ([2, 5, 6], []),     # x2 OR x5 OR x6
        ([], [1, 2, 3]),     # ~x1 OR ~x2 OR ~x3
    ]

    def compute_unsatisfied(zero_vars, one_vars, target):
        for i in zero_vars:
            qc.x(p[i])
        qc.mcx([p[i] for i in zero_vars + one_vars], target)
        for i in reversed(zero_vars):
            qc.x(p[i])

    # Compute one flag per unsatisfied clause.
    for (zero_vars, one_vars), target in zip(unsatisfied_patterns, clause_ancillas):
        compute_unsatisfied(zero_vars, one_vars, target)

    # All clauses are satisfied exactly when all unsatisfied flags are zero.
    for a in clause_ancillas:
        qc.x(a)
    qc.mcp(math.pi, clause_ancillas[:-1], clause_ancillas[-1])
    for a in reversed(clause_ancillas):
        qc.x(a)

    # Uncompute all clause flags.
    for (zero_vars, one_vars), target in reversed(
        list(zip(unsatisfied_patterns, clause_ancillas))
    ):
        compute_unsatisfied(zero_vars, one_vars, target)
