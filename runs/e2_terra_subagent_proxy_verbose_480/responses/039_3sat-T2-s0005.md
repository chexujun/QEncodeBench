def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2, x3, x4 = problem_qubits
    c0, c1, c2, c3, c4, c5, flag = ancilla_qubits

    clauses = [
        ([x0, x1, x4], [x0]),       # x0 OR ~x1 OR ~x4
        ([x0, x1, x4], [x4]),       # ~x0 OR ~x1 OR x4
        ([x0, x2, x3], [x2]),       # ~x0 OR x2 OR ~x3
        ([x0, x3, x4], [x4]),       # ~x0 OR ~x3 OR x4
        ([x2, x3, x4], [x3, x4]),   # ~x2 OR x3 OR x4
        ([x1, x3, x4], [x3, x4]),   # ~x1 OR x3 OR x4
    ]
    clause_bits = [c0, c1, c2, c3, c4, c5]

    # Compute each clause's violation bit.
    for (controls, positive_literals), target in zip(clauses, clause_bits):
        for qubit in positive_literals:
            qc.x(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(positive_literals):
            qc.x(qubit)

    # Mark exactly when no clause is violated.
    qc.mcx(clause_bits, flag, ctrl_state=0)
    qc.z(flag)
    qc.mcx(clause_bits, flag, ctrl_state=0)

    # Uncompute clause-violation bits.
    for (controls, positive_literals), target in reversed(
        list(zip(clauses, clause_bits))
    ):
        for qubit in positive_literals:
            qc.x(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(positive_literals):
            qc.x(qubit)
