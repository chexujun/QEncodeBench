from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    clause_ancillas = ancilla_qubits[:6]
    result = ancilla_qubits[6]

    clauses = [
        [(1, False), (3, False), (4, True)],
        [(0, True),  (2, True),  (3, True)],
        [(1, True),  (3, True),  (4, False)],
        [(0, False), (2, False), (3, True)],
        [(0, True),  (1, True),  (2, False)],
        [(1, True),  (2, False), (3, False)],
    ]

    def compute_clause(literals, target):
        positive_controls = [problem_qubits[i] for i, positive in literals if positive]
        controls = [problem_qubits[i] for i, _ in literals]

        # Make every control be 1 exactly when its literal is false.
        for qubit in positive_controls:
            qc.x(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(positive_controls):
            qc.x(qubit)

        # Target is now 1 iff the clause is satisfied.
        qc.x(target)

    def uncompute_clause(literals, target):
        positive_controls = [problem_qubits[i] for i, positive in literals if positive]
        controls = [problem_qubits[i] for i, _ in literals]

        qc.x(target)
        for qubit in positive_controls:
            qc.x(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(positive_controls):
            qc.x(qubit)

    for literals, target in zip(clauses, clause_ancillas):
        compute_clause(literals, target)

    qc.mcx(clause_ancillas, result)
    qc.z(result)
    qc.mcx(clause_ancillas, result)

    for literals, target in reversed(list(zip(clauses, clause_ancillas))):
        uncompute_clause(literals, target)
