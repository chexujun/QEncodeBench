from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    clauses = ancilla_qubits[:6]
    flag = ancilla_qubits[6]

    # Each tuple lists variables whose positive literals are false when 0.
    # Negative literals are false when the corresponding variable is 1.
    clause_data = [
        ([2], [0, 3]),  # ~x0 | x2 | ~x3
        ([0, 1], [2]),  # x0 | x1 | ~x2
        ([1, 2], [0]),  # ~x0 | x1 | x2
        ([0, 1, 2], []),  # x0 | x1 | x2
        ([0, 3], [2]),  # x0 | ~x2 | x3
        ([0], [2, 3]),  # x0 | ~x2 | ~x3
    ]

    def compute_clause(positive_vars, negative_vars, target):
        qc.x(target)
        for i in positive_vars:
            qc.x(x[i])
        qc.mcx([x[i] for i in positive_vars + negative_vars], target)
        for i in reversed(positive_vars):
            qc.x(x[i])

    for (positive_vars, negative_vars), target in zip(clause_data, clauses):
        compute_clause(positive_vars, negative_vars, target)

    qc.mcx(clauses, flag)
    qc.z(flag)
    qc.mcx(clauses, flag)

    for (positive_vars, negative_vars), target in reversed(
        list(zip(clause_data, clauses))
    ):
        compute_clause(positive_vars, negative_vars, target)
