from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    clauses = [
        ((0, 2, 4), (False, False, False)),
        ((4, 6, 7), (True,  False, False)),
        ((0, 2, 7), (False, False, False)),
        ((0, 2, 7), (True,  True,  True)),
        ((0, 1, 2), (False, True,  True)),
        ((0, 3, 7), (True,  False, False)),
        ((2, 4, 7), (False, True,  False)),
        ((0, 6, 7), (False, True,  False)),
        ((3, 4, 6), (False, True,  False)),
        ((2, 6, 7), (False, False, False)),
        ((1, 5, 7), (False, False, True)),
    ]

    clause_qubits = ancilla_qubits[:11]
    workspace = ancilla_qubits[11]

    def toggle_clause(clause_index: int) -> None:
        variables, negated = clauses[clause_index]
        controls = [problem_qubits[i] for i in variables]
        target = clause_qubits[clause_index]

        # A positive literal is false on |0>, so invert its variable while
        # detecting the unique assignment on which all three literals fail.
        for control, is_negated in zip(controls, negated):
            if not is_negated:
                qc.x(control)

        qc.x(target)
        qc.mcx(controls, target)

        for control, is_negated in reversed(list(zip(controls, negated))):
            if not is_negated:
                qc.x(control)

    # Compute one satisfaction bit for each clause.
    for i in range(11):
        toggle_clause(i)

    # Apply -1 exactly when all eleven clause-satisfaction bits are one.
    qc.h(clause_qubits[-1])
    qc.mcx(
        clause_qubits[:-1],
        clause_qubits[-1],
        ancilla_qubits=[workspace],
        mode="recursion",
    )
    qc.h(clause_qubits[-1])

    # Uncompute all clause bits and restore every ancilla to |0>.
    for i in reversed(range(11)):
        toggle_clause(i)
