from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    clause_ancillas = ancilla_qubits[:7]
    phase_ancilla = ancilla_qubits[7]

    # Each tuple contains the variables that must be inverted so that all
    # controls are 1 exactly when the corresponding clause is false.
    false_patterns = [
        ([3, 4, 7], [3]),
        ([2, 4, 7], [2, 4]),
        ([2, 3, 6], [2, 3, 6]),
        ([2, 3, 6], [2, 6]),
        ([1, 4, 5], [5]),
        ([3, 6, 7], [3, 6, 7]),
        ([0, 3, 4], []),
    ]

    def compute_clause(controls: list[int], zero_controls: list[int],
                       target: int) -> None:
        # Start at 1 and toggle to 0 precisely when the clause is false.
        qc.x(target)
        for i in zero_controls:
            qc.x(problem_qubits[i])
        qc.mcx([problem_qubits[i] for i in controls], target)
        for i in reversed(zero_controls):
            qc.x(problem_qubits[i])

    for (controls, zero_controls), target in zip(
            false_patterns, clause_ancillas):
        compute_clause(controls, zero_controls, target)

    qc.mcx(clause_ancillas, phase_ancilla)
    qc.z(phase_ancilla)
    qc.mcx(clause_ancillas, phase_ancilla)

    for (controls, zero_controls), target in reversed(
            list(zip(false_patterns, clause_ancillas))):
        compute_clause(controls, zero_controls, target)
