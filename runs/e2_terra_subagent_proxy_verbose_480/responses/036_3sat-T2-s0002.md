from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x = problem_qubits
    clause_flags = ancilla_qubits[:6]
    result = ancilla_qubits[6]

    # Compute whether each clause is false into its corresponding flag.
    # A clause is false exactly when all of its literals are false.
    clauses = [
        ([x[0], x[4]], [x[5]]),          # x0 OR x4 OR NOT x5
        ([x[4]], [x[2], x[5]]),          # NOT x2 OR x4 OR NOT x5
        ([x[0], x[1]], [x[4]]),          # x0 OR x1 OR NOT x4
        ([x[4], x[5]], [x[1]]),          # NOT x1 OR x4 OR x5
        ([x[0]], [x[2], x[3]]),          # x0 OR NOT x2 OR NOT x3
        ([x[1], x[2], x[3]], []),        # x1 OR x2 OR x3
    ]

    for flag, (positive, negative) in zip(clause_flags, clauses):
        for qubit in positive:
            qc.x(qubit)
        qc.mcx(positive + negative, flag)
        for qubit in reversed(positive):
            qc.x(qubit)

    # Convert "clause is false" flags to "clause is satisfied" flags.
    for flag in clause_flags:
        qc.x(flag)

    qc.mcx(clause_flags, result)
    qc.z(result)
    qc.mcx(clause_flags, result)

    for flag in reversed(clause_flags):
        qc.x(flag)

    # Uncompute clause-false flags.
    for flag, (positive, negative) in reversed(list(zip(clause_flags, clauses))):
        for qubit in positive:
            qc.x(qubit)
        qc.mcx(positive + negative, flag)
        for qubit in reversed(positive):
            qc.x(qubit)
