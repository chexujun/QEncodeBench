from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    clauses = ancilla_qubits[:4]
    result = ancilla_qubits[4]

    # Each tuple gives the assignment that makes its clause false.
    false_patterns = [
        (0, 1, 1),  # x0 OR not x1 OR not x2
        (1, 1, 0),  # not x0 OR not x1 OR x2
        (1, 0, 0),  # not x0 OR x1 OR x2
        (1, 1, 1),  # not x0 OR not x1 OR not x2
    ]

    def compute_clause(clause_ancilla, pattern):
        zero_controls = [x[i] for i, bit in enumerate(pattern) if bit == 0]
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx(x, clause_ancilla)
        for qubit in reversed(zero_controls):
            qc.x(qubit)
        qc.x(clause_ancilla)

    def uncompute_clause(clause_ancilla, pattern):
        zero_controls = [x[i] for i, bit in enumerate(pattern) if bit == 0]
        qc.x(clause_ancilla)
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx(x, clause_ancilla)
        for qubit in reversed(zero_controls):
            qc.x(qubit)

    for ancilla, pattern in zip(clauses, false_patterns):
        compute_clause(ancilla, pattern)

    qc.mcx(clauses, result)
    qc.z(result)
    qc.mcx(clauses, result)

    for ancilla, pattern in reversed(list(zip(clauses, false_patterns))):
        uncompute_clause(ancilla, pattern)
