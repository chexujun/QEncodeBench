def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2, x3 = problem_qubits
    c0, c1, c2, c3, c4, flag = ancilla_qubits

    # Compute whether each clause is violated.
    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x2], c0)       # ¬x0 ∧ ¬x1 ∧ x2
    qc.x(x1)
    qc.x(x0)

    qc.x(x1)
    qc.mcx([x0, x1, x2], c1)       # x0 ∧ ¬x1 ∧ x2
    qc.x(x1)

    qc.mcx([x0, x2, x3], c2)       # x0 ∧ x2 ∧ x3

    qc.x(x0)
    qc.mcx([x0, x1, x2], c3)       # ¬x0 ∧ x1 ∧ x2
    qc.x(x0)

    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c4)       # x0 ∧ ¬x1 ∧ ¬x2
    qc.x(x2)
    qc.x(x1)

    # All clauses are satisfied exactly when all violation bits are zero.
    for q in [c0, c1, c2, c3, c4]:
        qc.x(q)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    for q in [c4, c3, c2, c1, c0]:
        qc.x(q)

    qc.z(flag)

    # Uncompute the satisfaction flag.
    for q in [c0, c1, c2, c3, c4]:
        qc.x(q)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    for q in [c4, c3, c2, c1, c0]:
        qc.x(q)

    # Uncompute clause-violation bits.
    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c4)
    qc.x(x2)
    qc.x(x1)

    qc.x(x0)
    qc.mcx([x0, x1, x2], c3)
    qc.x(x0)

    qc.mcx([x0, x2, x3], c2)

    qc.x(x1)
    qc.mcx([x0, x1, x2], c1)
    qc.x(x1)

    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x2], c0)
    qc.x(x1)
    qc.x(x0)
