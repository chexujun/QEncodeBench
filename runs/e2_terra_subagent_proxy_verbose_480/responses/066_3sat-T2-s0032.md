from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    c = ancilla_qubits

    def compute_clause(target, controls_on_one, controls_on_zero):
        qc.x(target)
        for q in controls_on_zero:
            qc.x(q)
        qc.ccx(controls_on_one[0], controls_on_one[1], target)
        # For three-literal clauses, use the third control via an MCX.
        # This branch is retained as all clauses here have three literals.
        for q in controls_on_zero:
            qc.x(q)

    def clause(target, literals):
        # target becomes 1 iff the clause is satisfied.
        # literals: (variable_index, is_positive)
        zero_controls = [x[i] for i, positive in literals if positive]
        one_controls = [x[i] for i, positive in literals if not positive]
        qc.x(target)
        for q in zero_controls:
            qc.x(q)
        qc.mcx(one_controls + zero_controls, target)
        for q in reversed(zero_controls):
            qc.x(q)

    clauses = [
        [(1, False), (3, False), (4, False)],
        [(3, False), (4, False), (5, True)],
        [(0, False), (2, False), (5, False)],
        [(0, True),  (1, True),  (3, False)],
        [(2, False), (4, True),  (5, True)],
    ]

    for target, literals in zip(c[:5], clauses):
        clause(target, literals)

    qc.mcx(c[:5], c[5])
    qc.z(c[5])
    qc.mcx(c[:5], c[5])

    for target, literals in reversed(list(zip(c[:5], clauses))):
        clause(target, literals)
