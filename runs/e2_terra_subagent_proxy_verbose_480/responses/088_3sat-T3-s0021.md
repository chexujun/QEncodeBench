import math
from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    q = problem_qubits
    clause_flags = ancilla_qubits[:7]
    phase_target = ancilla_qubits[7]

    # Each tuple is (variables in the clause, variables that are positive).
    # A clause flag is set iff its clause is unsatisfied.
    clauses = [
        ([2, 3, 4], [2, 3, 4]),
        ([2, 4, 5], [2, 4, 5]),
        ([0, 1, 5], [1, 5]),
        ([0, 3, 5], [0, 5]),
        ([0, 3, 4], []),
        ([0, 1, 2], [2]),
        ([0, 3, 5], [3, 5]),
    ]

    def compute_clause(vars_, positive_vars, target):
        for i in positive_vars:
            qc.x(q[i])
        qc.mcx([q[i] for i in vars_], target)
        for i in reversed(positive_vars):
            qc.x(q[i])

    # Compute whether each clause is unsatisfied.
    for (vars_, positive_vars), flag in zip(clauses, clause_flags):
        compute_clause(vars_, positive_vars, flag)

    # All flags are zero exactly when every clause is satisfied.
    for flag in clause_flags:
        qc.x(flag)
    qc.x(phase_target)
    qc.mcp(math.pi, clause_flags, phase_target)
    qc.x(phase_target)
    for flag in reversed(clause_flags):
        qc.x(flag)

    # Uncompute clause flags.
    for (vars_, positive_vars), flag in reversed(list(zip(clauses, clause_flags))):
        compute_clause(vars_, positive_vars, flag)
