from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:

    a = ancilla_qubits

    # Compute whether each clause is unsatisfied.
    qc.ccx(x[3], x[5], a[0])
    qc.ccx(a[0], x[6], a[0])  # replaced below by proper 3-control construction
