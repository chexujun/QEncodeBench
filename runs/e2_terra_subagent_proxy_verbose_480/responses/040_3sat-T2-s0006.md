from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3, x4 = problem_qubits
    c0, c1, c2, c3, c4, phase = ancilla_qubits

    def mark_violation(target, controls, flips):
        for q in flips:
            qc.x(q)
        qc.ccx(controls[0], controls[1], target) if len(controls) == 2 else qc.mcx(controls, target)
        for q in reversed(flips):
            qc.x(q)

    # Compute one flag per clause: flag = 1 iff that clause is violated.
    mark_violation(c0, [x0, x1, x3], [x0, x3])
    mark_violation(c1, [x0, x1, x3], [x3])
    mark_violation(c2, [x0, x1, x3], [x1, x3])
    mark_violation(c3, [x2, x3, x4], [x3, x4])
    mark_violation(c4, [x0, x2, x4], [x0, x2])

    # A satisfying assignment has every violation flag equal to zero.
    for q in (c0, c1, c2, c3, c4):
        qc.x(q)
    qc.mcx([c0, c1, c2, c3, c4], phase)
    qc.z(phase)
    qc.mcx([c0, c1, c2, c3, c4], phase)
    for q in (c4, c3, c2, c1, c0):
        qc.x(q)

    # Uncompute clause-violation flags.
    mark_violation(c4, [x0, x2, x4], [x0, x2])
    mark_violation(c3, [x2, x3, x4], [x3, x4])
    mark_violation(c2, [x0, x1, x3], [x1, x3])
    mark_violation(c1, [x0, x1, x3], [x3])
    mark_violation(c0, [x0, x1, x3], [x0, x3])
