def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2, x3, x4 = problem_qubits
    c0, c1, c2, c3, c4, c5, out = ancilla_qubits

    # Compute whether each clause is false.
    qc.x(x0); qc.x(x2)
    qc.mcx([x0, x2, x3], c0)
    qc.x(x2); qc.x(x0)

    qc.x(x1); qc.x(x2); qc.x(x4)
    qc.mcx([x1, x2, x4], c1)
    qc.x(x4); qc.x(x2); qc.x(x1)

    qc.x(x1); qc.x(x3)
    qc.mcx([x0, x1, x3], c2)
    qc.x(x3); qc.x(x1)

    qc.x(x1); qc.x(x3)
    qc.mcx([x1, x2, x3], c3)
    qc.x(x3); qc.x(x1)

    qc.x(x3); qc.x(x4)
    qc.mcx([x2, x3, x4], c4)
    qc.x(x4); qc.x(x3)

    qc.x(x1); qc.x(x2); qc.x(x4)
    qc.mcx([x0, x1, x2], c5)
    qc.x(x4); qc.x(x2); qc.x(x1)

    # All clauses are satisfied exactly when all false-clause flags are 0.
    qc.x(c0); qc.x(c1); qc.x(c2); qc.x(c3); qc.x(c4); qc.x(c5)
    qc.mcx([c0, c1, c2, c3, c4, c5], out)
    qc.z(out)
    qc.mcx([c0, c1, c2, c3, c4, c5], out)
    qc.x(c5); qc.x(c4); qc.x(c3); qc.x(c2); qc.x(c1); qc.x(c0)

    # Uncompute clause flags.
    qc.x(x1); qc.x(x2); qc.x(x4)
    qc.mcx([x0, x1, x2], c5)
    qc.x(x4); qc.x(x2); qc.x(x1)

    qc.x(x3); qc.x(x4)
    qc.mcx([x2, x3, x4], c4)
    qc.x(x4); qc.x(x3)

    qc.x(x1); qc.x(x3)
    qc.mcx([x1, x2, x3], c3)
    qc.x(x3); qc.x(x1)

    qc.x(x1); qc.x(x3)
    qc.mcx([x0, x1, x3], c2)
    qc.x(x3); qc.x(x1)

    qc.x(x1); qc.x(x2); qc.x(x4)
    qc.mcx([x1, x2, x4], c1)
    qc.x(x4); qc.x(x2); qc.x(x1)

    qc.x(x0); qc.x(x2)
    qc.mcx([x0, x2, x3], c0)
    qc.x(x2); qc.x(x0)
