from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3 = problem_qubits
    c0, c1, c2, c3, c4, c5, out = ancilla_qubits

    def mark_unsatisfied(zeros, flag):
        # Temporarily map controls requiring 0 to controls requiring 1.
        for qubit in zeros:
            qc.x(qubit)
        qc.mcx([x0, x1, x2], flag)
        for qubit in reversed(zeros):
            qc.x(qubit)

    # Compute one flag per clause: flag = 1 iff its clause is unsatisfied.
    # Clauses use x0, x1, x2 as controls after suitable temporary X gates;
    # x3 replaces an unused control where needed by swapping the control list.
    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x2], c0)       # ~x0 & ~x1 & x2
    qc.x(x1)
    qc.x(x0)

    qc.x(x1)
    qc.x(x3)
    qc.mcx([x1, x2, x3], c1)       # ~x1 & x2 & x3
    qc.x(x3)
    qc.x(x1)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x1, x2, x3], c2)       # x1 & ~x2 & ~x3
    qc.x(x3)
    qc.x(x2)

    qc.x(x0)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c3)       # x0 & x2 & ~x3
    qc.x(x3)
    qc.x(x0)

    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x2, x3], c4)       # ~x0 & ~x2 & x3
    qc.x(x2)
    qc.x(x0)

    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c5)       # ~x0 & x1 & ~x2
    qc.x(x2)
    qc.x(x0)

    # Convert unsatisfied flags to satisfied flags, then AND them.
    for flag in [c0, c1, c2, c3, c4, c5]:
        qc.x(flag)
    qc.mcx([c0, c1, c2, c3, c4, c5], out)
    qc.z(out)
    qc.mcx([c0, c1, c2, c3, c4, c5], out)
    for flag in reversed([c0, c1, c2, c3, c4, c5]):
        qc.x(flag)

    # Uncompute clause flags in reverse order.
    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c5)
    qc.x(x2)
    qc.x(x0)

    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x2, x3], c4)
    qc.x(x2)
    qc.x(x0)

    qc.x(x0)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c3)
    qc.x(x3)
    qc.x(x0)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x1, x2, x3], c2)
    qc.x(x3)
    qc.x(x2)

    qc.x(x1)
    qc.x(x3)
    qc.mcx([x1, x2, x3], c1)
    qc.x(x3)
    qc.x(x1)

    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x2], c0)
    qc.x(x1)
    qc.x(x0)
