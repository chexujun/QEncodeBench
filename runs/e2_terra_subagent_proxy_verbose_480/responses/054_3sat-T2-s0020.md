def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2, x3, x4, x5 = problem_qubits
    c1, c2, c3, c4, c5, phase = ancilla_qubits

    # Compute one flag per clause: flag = 1 exactly when its clause is false.
    qc.x(x5)
    qc.mcx([x1, x4, x5], c1)
    qc.x(x5)

    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x4], c2)
    qc.x(x1)
    qc.x(x0)

    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x5], c3)
    qc.x(x1)
    qc.x(x0)

    qc.x(x1)
    qc.x(x3)
    qc.mcx([x1, x3, x5], c4)
    qc.x(x3)
    qc.x(x1)

    qc.mcx([x0, x3, x5], c5)

    # All flags are zero iff every clause is satisfied.
    qc.h(phase)
    qc.mcx([c1, c2, c3, c4, c5], phase, ctrl_state=0)
    qc.h(phase)

    # Uncompute clause flags.
    qc.mcx([x0, x3, x5], c5)

    qc.x(x1)
    qc.x(x3)
    qc.mcx([x1, x3, x5], c4)
    qc.x(x3)
    qc.x(x1)

    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x5], c3)
    qc.x(x1)
    qc.x(x0)

    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x4], c2)
    qc.x(x1)
    qc.x(x0)

    qc.x(x5)
    qc.mcx([x1, x4, x5], c1)
    qc.x(x5)
