```python
import math
from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3 = problem_qubits
    c1, c2, c3, c4, c5 = ancilla_qubits[:5]

    # Compute whether each clause is unsatisfied.

    # (¬x0 ∨ ¬x2 ∨ x3) is false iff x0=1, x2=1, x3=0.
    qc.x(x3)
    qc.mcx([x0, x2, x3], c1)
    qc.x(x3)

    # (¬x0 ∨ x2 ∨ x3) is false iff x0=1, x2=0, x3=0.
    qc.x(x2)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c2)
    qc.x(x3)
    qc.x(x2)

    # (¬x0 ∨ x1 ∨ x2) is false iff x0=1, x1=0, x2=0.
    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c3)
    qc.x(x2)
    qc.x(x1)

    # (x0 ∨ ¬x1 ∨ x2) is false iff x0=0, x1=1, x2=0.
    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c4)
    qc.x(x2)
    qc.x(x0)

    # (x1 ∨ ¬x2 ∨ x3) is false iff x1=0, x2=1, x3=0.
    qc.x(x1)
    qc.x(x3)
    qc.mcx([x1, x2, x3], c5)
    qc.x(x3)
    qc.x(x1)

    # All clauses are satisfied exactly when all clause-failure flags are 0.
    for clause_flag in (c1, c2, c3, c4, c5):
        qc.x(clause_flag)
    qc.mcp(math.pi, [c1, c2, c3, c4], c5)
    for clause_flag in (c5, c4, c3, c2, c1):
        qc.x(clause_flag)

    # Uncompute clause-failure flags.
    qc.x(x1)
    qc.x(x3)
    qc.mcx([x1, x2, x3], c5)
    qc.x(x3)
    qc.x(x1)

    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c4)
    qc.x(x2)
    qc.x(x0)

    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c3)
    qc.x(x2)
    qc.x(x1)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c2)
    qc.x(x3)
    qc.x(x2)

    qc.x(x3)
    qc.mcx([x0, x2, x3], c1)
    qc.x(x3)
```
