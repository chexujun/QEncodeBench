def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2, x3 = problem_qubits
    c0, c1, c2, c3, c4, out = ancilla_qubits

    # Initialize clause flags to 1, then flip each flag iff its clause is false.
    for c in (c0, c1, c2, c3, c4):
        qc.x(c)

    # (x1 OR NOT x2 OR NOT x3) false iff x1=0, x2=1, x3=1
    qc.x(x2)
    qc.x(x3)
    qc.mcx([x1, x2, x3], c0)
    qc.x(x3)
    qc.x(x2)

    # (NOT x0 OR NOT x2 OR x3) false iff x0=1, x2=1, x3=0
    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x2, x3], c1)
    qc.x(x2)
    qc.x(x0)

    # (NOT x1 OR x2 OR x3) false iff x1=1, x2=0, x3=0
    qc.x(x1)
    qc.mcx([x1, x2, x3], c2)
    qc.x(x1)

    # (x0 OR x2 OR x3) false iff x0=0, x2=0, x3=0
    qc.mcx([x0, x2, x3], c3)

    # (x0 OR NOT x2 OR NOT x3) false iff x0=0, x2=1, x3=1
    qc.x(x2)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c4)
    qc.x(x3)
    qc.x(x2)

    qc.mcx([c0, c1, c2, c3, c4], out)
    qc.z(out)
    qc.mcx([c0, c1, c2, c3, c4], out)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c4)
    qc.x(x3)
    qc.x(x2)

    qc.mcx([x0, x2, x3], c3)

    qc.x(x1)
    qc.mcx([x1, x2, x3], c2)
    qc.x(x1)

    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x2, x3], c1)
    qc.x(x2)
    qc.x(x0)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x1, x2, x3], c0)
    qc.x(x3)
    qc.x(x2)

    for c in (c4, c3, c2, c1, c0):
        qc.x(c)
