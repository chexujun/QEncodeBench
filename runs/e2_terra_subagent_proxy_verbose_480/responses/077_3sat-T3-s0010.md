from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    clauses = [
        ((2, True),  (5, True),  (6, False)),
        ((0, False), (2, True),  (7, False)),
        ((0, False), (2, False), (4, True)),
        ((2, True),  (5, False), (6, True)),
        ((1, True),  (5, True),  (7, True)),
        ((2, True),  (3, True),  (5, False)),
        ((2, False), (4, False), (6, False)),
        ((0, False), (3, True),  (7, True)),
        ((1, False), (2, True),  (3, True)),
        ((0, False), (3, True),  (5, True)),
    ]

    clause_qubits = ancilla_qubits[:10]
    phase_qubit = ancilla_qubits[10]

    def compute_clause(clause, target):
        positive_qubits = [
            problem_qubits[index] for index, is_positive in clause
            if is_positive
        ]
        for qubit in positive_qubits:
            qc.x(qubit)
        qc.x(target)
        qc.mcx([problem_qubits[index] for index, _ in clause], target)
        for qubit in reversed(positive_qubits):
            qc.x(qubit)

    for clause, target in zip(clauses, clause_qubits):
        compute_clause(clause, target)

    qc.x(phase_qubit)
    qc.h(phase_qubit)
    qc.mcx(clause_qubits, phase_qubit)
    qc.h(phase_qubit)
    qc.x(phase_qubit)

    for clause, target in reversed(list(zip(clauses, clause_qubits))):
        compute_clause(clause, target)
