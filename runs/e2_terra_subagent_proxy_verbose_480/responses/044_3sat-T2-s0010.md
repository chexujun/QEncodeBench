```python
from qiskit import QuantumCircuit
import math


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    c = ancilla_qubits[:6]
    phase_ancilla = ancilla_qubits[6]

    # Compute whether each clause is UNSATISFIED into c[0] ... c[5].
    qc.mcx([x[2], x[3], x[5]], c[0])

    qc.x(x[3])
    qc.x(x[5])
    qc.mcx([x[0], x[3], x[5]], c[1])
    qc.x(x[5])
    qc.x(x[3])

    qc.x(x[1])
    qc.x(x[4])
    qc.x(x[5])
    qc.mcx([x[1], x[4], x[5]], c[2])
    qc.x(x[5])
    qc.x(x[4])
    qc.x(x[1])

    qc.mcx([x[0], x[1], x[4]], c[3])

    qc.x(x[0])
    qc.x(x[1])
    qc.x(x[2])
    qc.mcx([x[0], x[1], x[2]], c[4])
    qc.x(x[2])
    qc.x(x[1])
    qc.x(x[0])

    qc.x(x[0])
    qc.x(x[5])
    qc.mcx([x[0], x[1], x[5]], c[5])
    qc.x(x[5])
    qc.x(x[0])

    # Convert clause flags to "satisfied" flags and phase their conjunction.
    for flag in c:
        qc.x(flag)
    qc.x(phase_ancilla)
    qc.mcp(math.pi, c, phase_ancilla)
    qc.x(phase_ancilla)
    for flag in reversed(c):
        qc.x(flag)

    # Uncompute clause flags.
    qc.x(x[0])
    qc.x(x[5])
    qc.mcx([x[0], x[1], x[5]], c[5])
    qc.x(x[5])
    qc.x(x[0])

    qc.x(x[0])
    qc.x(x[1])
    qc.x(x[2])
    qc.mcx([x[0], x[1], x[2]], c[4])
    qc.x(x[2])
    qc.x(x[1])
    qc.x(x[0])

    qc.mcx([x[0], x[1], x[4]], c[3])

    qc.x(x[1])
    qc.x(x[4])
    qc.x(x[5])
    qc.mcx([x[1], x[4], x[5]], c[2])
    qc.x(x[5])
    qc.x(x[4])
    qc.x(x[1])

    qc.x(x[3])
    qc.x(x[5])
    qc.mcx([x[0], x[3], x[5]], c[1])
    qc.x(x[5])
    qc.x(x[3])

    qc.mcx([x[2], x[3], x[5]], c[0])
```
