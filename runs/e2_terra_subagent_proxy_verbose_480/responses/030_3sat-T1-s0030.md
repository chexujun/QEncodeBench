from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x0, x1, x2, x3 = problem_qubits
    c1, c2, c3, c4, c5, flag = ancilla_qubits

    def compute_clause_satisfied(controls, zero_controls, target):
        # target := target XOR (the clause is satisfied)
        qc.x(target)
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(zero_controls):
            qc.x(qubit)

    clauses = [
        ([x0, x1, x3], [x0]),          # x0 OR ~x1 OR ~x3
        ([x0, x1, x3], [x0, x1]),      # x0 OR x1 OR ~x3
        ([x1, x2, x3], [x1, x2, x3]),  # x1 OR x2 OR x3
        ([x0, x1, x3], [x3]),          # ~x0 OR ~x1 OR x3
        ([x0, x1, x2], [x2]),          # ~x0 OR ~x1 OR x2
    ]

    for (controls, zero_controls), target in zip(clauses, [c1, c2, c3, c4, c5]):
        compute_clause_satisfied(controls, zero_controls, target)

    qc.mcx([c1, c2, c3, c4, c5], flag)
    qc.z(flag)
    qc.mcx([c1, c2, c3, c4, c5], flag)

    for (controls, zero_controls), target in reversed(
        list(zip(clauses, [c1, c2, c3, c4, c5]))
    ):
        compute_clause_satisfied(controls, zero_controls, target)
