from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x0, x1, x2 = problem_qubits
    c1, c2, c3, c4, flag = ancilla_qubits[:5]

    def compute_clause(clause_ancilla, false_bits):
        # Clause ancilla becomes 1 except for its unique all-literals-false case.
        qc.x(clause_ancilla)
        for qubit, bit in zip(problem_qubits, false_bits):
            if bit == 0:
                qc.x(qubit)
        qc.mcx(problem_qubits, clause_ancilla)
        for qubit, bit in reversed(list(zip(problem_qubits, false_bits))):
            if bit == 0:
                qc.x(qubit)

    # Compute the four clause values.
    compute_clause(c1, (1, 0, 1))  # ¬x0 OR x1 OR ¬x2
    compute_clause(c2, (1, 1, 1))  # ¬x0 OR ¬x1 OR ¬x2
    compute_clause(c3, (1, 0, 0))  # ¬x0 OR x1 OR x2
    compute_clause(c4, (0, 0, 1))  # x0 OR x1 OR ¬x2

    # Mark exactly when every clause is true.
    qc.mcx([c1, c2, c3, c4], flag)
    qc.z(flag)
    qc.mcx([c1, c2, c3, c4], flag)

    # Uncompute clause values.
    compute_clause(c4, (0, 0, 1))
    compute_clause(c3, (1, 0, 0))
    compute_clause(c2, (1, 1, 1))
    compute_clause(c1, (1, 0, 1))
