def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2 = problem_qubits
    violation_01, violation_12, satisfies = ancilla_qubits[:3]

    # violation_01 = x0 AND (NOT x1)
    qc.x(x1)
    qc.ccx(x0, x1, violation_01)
    qc.x(x1)

    # violation_12 = x1 AND (NOT x2)
    qc.x(x2)
    qc.ccx(x1, x2, violation_12)
    qc.x(x2)

    # satisfies = NOT violation_01 AND NOT violation_12
    qc.x(violation_01)
    qc.x(violation_12)
    qc.ccx(violation_01, violation_12, satisfies)
    qc.x(violation_12)
    qc.x(violation_01)

    qc.z(satisfies)

    # Uncompute satisfies.
    qc.x(violation_01)
    qc.x(violation_12)
    qc.ccx(violation_01, violation_12, satisfies)
    qc.x(violation_12)
    qc.x(violation_01)

    # Uncompute violation_12.
    qc.x(x2)
    qc.ccx(x1, x2, violation_12)
    qc.x(x2)

    # Uncompute violation_01.
    qc.x(x1)
    qc.ccx(x0, x1, violation_01)
    qc.x(x1)
