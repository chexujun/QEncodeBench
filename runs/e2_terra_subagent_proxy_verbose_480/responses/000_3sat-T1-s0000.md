def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2 = problem_qubits
    clause_bits = ancilla_qubits[:5]
    flag = ancilla_qubits[5]

    def compute_violation(target, zero_controls):
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx([x0, x1, x2], target)
        for qubit in reversed(zero_controls):
            qc.x(qubit)

    # Each ancilla records whether its corresponding clause is violated.
    compute_violation(clause_bits[0], [x0, x2])  # ~x0 & x1 & ~x2
    compute_violation(clause_bits[1], [x2])      # x0 & x1 & ~x2
    compute_violation(clause_bits[2], [])        # x0 & x1 & x2
    compute_violation(clause_bits[3], [x0])      # ~x0 & x1 & x2
    compute_violation(clause_bits[4], [x0, x1])  # ~x0 & ~x1 & x2

    # Mark exactly the states with no violated clauses.
    for bit in clause_bits:
        qc.x(bit)
    qc.mcx(clause_bits, flag)
    qc.z(flag)
    qc.mcx(clause_bits, flag)
    for bit in reversed(clause_bits):
        qc.x(bit)

    # Uncompute all clause-violation ancillas.
    compute_violation(clause_bits[4], [x0, x1])
    compute_violation(clause_bits[3], [x0])
    compute_violation(clause_bits[2], [])
    compute_violation(clause_bits[1], [x2])
    compute_violation(clause_bits[0], [x0, x2])
