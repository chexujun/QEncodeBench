from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3 = problem_qubits
    clauses = ancilla_qubits[:6]
    out = ancilla_qubits[6]

    # Each tuple specifies a clause's variables and the bit pattern for
    # which that clause is false.
    false_patterns = [
        ([x0, x1, x2], [1, 1, 1]),
        ([x0, x1, x3], [0, 1, 1]),
        ([x0, x2, x3], [1, 0, 1]),
        ([x0, x1, x3], [1, 0, 1]),
        ([x1, x2, x3], [1, 0, 1]),
        ([x0, x2, x3], [1, 1, 0]),
    ]

    def compute_false(controls, pattern, target):
        for qubit, value in zip(controls, pattern):
            if value == 0:
                qc.x(qubit)
        qc.mcx(controls, target)
        for qubit, value in reversed(list(zip(controls, pattern))):
            if value == 0:
                qc.x(qubit)

    for (controls, pattern), target in zip(false_patterns, clauses):
        compute_false(controls, pattern, target)

    for clause in clauses:
        qc.x(clause)
    qc.mcx(clauses, out)
    qc.z(out)
    qc.mcx(clauses, out)
    for clause in reversed(clauses):
        qc.x(clause)

    for (controls, pattern), target in reversed(list(zip(false_patterns, clauses))):
        compute_false(controls, pattern, target)
