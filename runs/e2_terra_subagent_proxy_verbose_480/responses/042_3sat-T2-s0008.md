from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x = problem_qubits
    c = ancilla_qubits[:6]

    # Each entry lists the variables that must be temporarily inverted in
    # order to detect that its corresponding clause is unsatisfied.
    positive_literals = [
        [3],        # (~x0 | ~x1 | x3)
        [3],        # (~x2 | x3 | ~x4)
        [0, 2],     # (x0 | ~x1 | x2)
        [0, 2, 3],  # (x0 | x2 | x3)
        [5],        # (~x0 | ~x2 | x5)
        [5],        # (~x1 | ~x2 | x5)
    ]
    clause_vars = [
        [0, 1, 3],
        [2, 3, 4],
        [0, 1, 2],
        [0, 2, 3],
        [0, 2, 5],
        [1, 2, 5],
    ]

    # Compute each clause's truth value into c[i].
    for target, vars_, flips in zip(c, clause_vars, positive_literals):
        for bit in flips:
            qc.x(x[bit])
        qc.mcx([x[bit] for bit in vars_], target)
        for bit in reversed(flips):
            qc.x(x[bit])
        qc.x(target)

    # Put the final ancilla in |->, so a controlled NOT applies the phase.
    phase_ancilla = ancilla_qubits[6]
    qc.x(phase_ancilla)
    qc.h(phase_ancilla)
    qc.mcx(c, phase_ancilla)
    qc.h(phase_ancilla)
    qc.x(phase_ancilla)

    # Uncompute clause truth values.
    for target, vars_, flips in reversed(
        list(zip(c, clause_vars, positive_literals))
    ):
        qc.x(target)
        for bit in flips:
            qc.x(x[bit])
        qc.mcx([x[bit] for bit in vars_], target)
        for bit in reversed(flips):
            qc.x(x[bit])
