def build_oracle(qc, problem_qubits, ancilla_qubits):
    flags = ancilla_qubits[:5]
    output = ancilla_qubits[5]

    # Each tuple is (variable indices, values required for the clause to be false).
    false_patterns = [
        ([1, 2, 3], [1, 0, 0]),  # (!x1 | x2 | x3)
        ([0, 1, 3], [1, 1, 1]),  # (!x0 | !x1 | !x3)
        ([0, 1, 2], [1, 0, 1]),  # (!x0 | x1 | !x2)
        ([1, 2, 3], [1, 0, 1]),  # (!x1 | x2 | !x3)
        ([0, 2, 3], [1, 1, 0]),  # (!x0 | !x2 | x3)
    ]

    def compute_false_clause(indices, values, target):
        negative_controls = [
            problem_qubits[i] for i, value in zip(indices, values) if value == 0
        ]
        for qubit in negative_controls:
            qc.x(qubit)
        qc.mcx([problem_qubits[i] for i in indices], target)
        for qubit in reversed(negative_controls):
            qc.x(qubit)

    # Compute whether each clause is false.
    for (indices, values), flag in zip(false_patterns, flags):
        compute_false_clause(indices, values, flag)

    # Mark exactly when no clause is false.
    for flag in flags:
        qc.x(flag)
    qc.mcx(flags, output)
    qc.z(output)
    qc.mcx(flags, output)
    for flag in reversed(flags):
        qc.x(flag)

    # Uncompute clause-failure flags.
    for (indices, values), flag in reversed(list(zip(false_patterns, flags))):
        compute_false_clause(indices, values, flag)
