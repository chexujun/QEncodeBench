from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x = problem_qubits
    clause_flags = ancilla_qubits[:7]
    phase_flag = ancilla_qubits[7]

    # Each entry is the variables that must be temporarily inverted for
    # the corresponding clause to be false.
    false_clause_inversions = [
        [1, 5],       # (x1 OR NOT x4 OR x5)
        [0, 1, 2],    # (x0 OR x1 OR x2)
        [1],          # (x1 OR NOT x2 OR NOT x5)
        [3, 5],       # (x3 OR NOT x4 OR x5)
        [0, 4],       # (x0 OR x4 OR NOT x5)
        [4, 5],       # (NOT x1 OR NOT x4 OR x5)
        [1, 2],       # (NOT x0 OR x1 OR x2)
    ]
    clause_controls = [
        [1, 4, 5],
        [0, 1, 2],
        [1, 2, 5],
        [3, 4, 5],
        [0, 4, 5],
        [1, 4, 5],
        [0, 1, 2],
    ]

    # Compute one flag per unsatisfied clause.
    for inversions, controls, flag in zip(
        false_clause_inversions, clause_controls, clause_flags
    ):
        for i in inversions:
            qc.x(x[i])
        qc.mcx([x[i] for i in controls], flag)
        for i in reversed(inversions):
            qc.x(x[i])

    # Turn "clause is unsatisfied" flags into satisfaction flags, mark
    # their conjunction, then restore all ancillas.
    for flag in clause_flags:
        qc.x(flag)
    qc.mcx(clause_flags, phase_flag)
    qc.z(phase_flag)
    qc.mcx(clause_flags, phase_flag)
    for flag in reversed(clause_flags):
        qc.x(flag)

    # Uncompute clause flags.
    for inversions, controls, flag in reversed(list(zip(
        false_clause_inversions, clause_controls, clause_flags
    ))):
        for i in inversions:
            qc.x(x[i])
        qc.mcx([x[i] for i in controls], flag)
        for i in reversed(inversions):
            qc.x(x[i])
