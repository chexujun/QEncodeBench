import math

def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = problem_qubits
    a = ancilla_qubits

    def clause_unsatisfied(target, positive_literals):
        for bit in positive_literals:
            qc.x(x[bit])
        qc.mcx([x[1], x[3], x[7]], target)

    # Compute each clause's "unsatisfied" flag.
    qc.x(x[3])
    qc.x(x[7])
    qc.mcx([x[1], x[3], x[7]], a[0])
    qc.x(x[7])
    qc.x(x[3])

    qc.x(x[0])
    qc.x(x[2])
    qc.x(x[7])
    qc.mcx([x[0], x[2], x[7]], a[1])
    qc.x(x[7])
    qc.x(x[2])
    qc.x(x[0])

    qc.x(x[4])
    qc.mcx([x[3], x[4], x[5]], a[2])
    qc.x(x[4])

    qc.mcx([x[3], x[5], x[6]], a[3])

    qc.x(x[5])
    qc.mcx([x[0], x[1], x[5]], a[4])
    qc.x(x[5])

    qc.x(x[7])
    qc.mcx([x[4], x[6], x[7]], a[5])
    qc.x(x[7])

    qc.x(x[0])
    qc.x(x[4])
    qc.mcx([x[0], x[4], x[7]], a[6])
    qc.x(x[4])
    qc.x(x[0])

    # Phase flip iff no clause is unsatisfied.
    for q in a[:7]:
        qc.x(q)
    qc.mcp(math.pi, a[:6], a[6])
    for q in reversed(a[:7]):
        qc.x(q)

    # Uncompute clause flags.
    qc.x(x[0])
    qc.x(x[4])
    qc.mcx([x[0], x[4], x[7]], a[6])
    qc.x(x[4])
    qc.x(x[0])

    qc.x(x[7])
    qc.mcx([x[4], x[6], x[7]], a[5])
    qc.x(x[7])

    qc.x(x[5])
    qc.mcx([x[0], x[1], x[5]], a[4])
    qc.x(x[5])

    qc.mcx([x[3], x[5], x[6]], a[3])

    qc.x(x[4])
    qc.mcx([x[3], x[4], x[5]], a[2])
    qc.x(x[4])

    qc.x(x[0])
    qc.x(x[2])
    qc.x(x[7])
    qc.mcx([x[0], x[2], x[7]], a[1])
    qc.x(x[7])
    qc.x(x[2])
    qc.x(x[0])

    qc.x(x[3])
    qc.x(x[7])
    qc.mcx([x[1], x[3], x[7]], a[0])
    qc.x(x[7])
    qc.x(x[3])
