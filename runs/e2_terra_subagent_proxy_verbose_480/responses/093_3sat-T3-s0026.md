from qiskit import QuantumCircuit
import math


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    clause_flags = ancilla_qubits[:10]
    phase_target = ancilla_qubits[10]

    # Each tuple lists the three variable indices and their required values
    # when that clause is violated.
    violations = [
        ((2, 4, 6), (1, 1, 1)),  # (~x2 | ~x4 | ~x6)
        ((4, 5, 6), (0, 0, 1)),  # ( x4 |  x5 | ~x6)
        ((0, 1, 3), (1, 0, 1)),  # (~x0 |  x1 | ~x3)
        ((0, 3, 4), (1, 0, 1)),  # (~x0 |  x3 | ~x4)
        ((3, 5, 6), (1, 0, 0)),  # (~x3 |  x5 |  x6)
        ((1, 2, 6), (1, 1, 0)),  # (~x1 | ~x2 |  x6)
        ((0, 3, 5), (1, 0, 0)),  # (~x0 |  x3 |  x5)
        ((0, 5, 6), (0, 1, 1)),  # ( x0 | ~x5 | ~x6)
        ((0, 3, 5), (0, 1, 1)),  # ( x0 | ~x3 | ~x5)
        ((0, 1, 2), (0, 0, 1)),  # ( x0 |  x1 | ~x2)
    ]

    def compute_clause(indices, values, target):
        for index, value in zip(indices, values):
            if value == 0:
                qc.x(x[index])
        qc.mcx([x[index] for index in indices], target)
        for index, value in reversed(list(zip(indices, values))):
            if value == 0:
                qc.x(x[index])

    # Compute one flag per violated clause.
    for (indices, values), flag in zip(violations, clause_flags):
        compute_clause(indices, values, flag)

    # All clauses are satisfied exactly when every violation flag is zero.
    qc.x(phase_target)
    qc.mcp(math.pi, clause_flags, phase_target, ctrl_state=0)
    qc.x(phase_target)

    # Uncompute all clause flags.
    for (indices, values), flag in reversed(list(zip(violations, clause_flags))):
        compute_clause(indices, values, flag)
