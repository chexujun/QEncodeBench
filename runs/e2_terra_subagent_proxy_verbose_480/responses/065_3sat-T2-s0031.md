import math
from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    clause_bits = ancilla_qubits[:6]
    flag = ancilla_qubits[6]

    # Toggles target iff all controls are 1, after temporarily inverting
    # variables whose corresponding clause literals are positive.
    def compute_unsatisfied(target, controls, invert_controls):
        for i in invert_controls:
            qc.x(x[i])
        qc.mcx([x[i] for i in controls], target)
        for i in reversed(invert_controls):
            qc.x(x[i])

    clauses = [
        # controls, variables to invert (positive literals)
        ([3, 4, 5], [3, 5]),  # x3 OR !x4 OR x5
        ([1, 2, 3], [1, 3]),  # x1 OR !x2 OR x3
        ([0, 1, 2], []),      # !x0 OR !x1 OR !x2
        ([0, 1, 4], [0, 4]),  # x0 OR !x1 OR x4
        ([1, 2, 4], [2, 4]),  # !x1 OR x2 OR x4
        ([0, 1, 2], [1]),     # !x0 OR x1 OR !x2
    ]

    # Compute one bit per clause: 1 exactly when that clause is unsatisfied.
    for target, (controls, inversions) in zip(clause_bits, clauses):
        compute_unsatisfied(target, controls, inversions)

    # Convert unsatisfied bits to satisfied bits, phase-mark assignments where
    # all six clauses are satisfied, then restore the clause bits.
    for bit in clause_bits:
        qc.x(bit)
    qc.mcx(clause_bits, flag)
    qc.z(flag)
    qc.mcx(clause_bits, flag)
    for bit in reversed(clause_bits):
        qc.x(bit)

    # Uncompute all clause bits.
    for target, (controls, inversions) in reversed(list(zip(clause_bits, clauses))):
        compute_unsatisfied(target, controls, inversions)
