    x0, x1, x2 = problem_qubits
    c0, c1, c2, c3, c4, result = ancilla_qubits

    # Compute whether each clause is violated.
    # Clause 1: (x0 OR x1 OR x2), violated by 000.
    qc.x(x0)
    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c0)
    qc.x(x2)
    qc.x(x1)
    qc.x(x0)

    qc.mcx([x0, x1, x2], c1)
    qc.x(x1)

    qc.x(x1)
    qc.mcx([x0, x1, x2], c3)
    qc.x(x1)
    qc.x(x0)

    qc.x(x2)
    qc.mcx([x0, x1, x2], c4)
    qc.x(x2)
    qc.x(x0)

    qc.x(x0)
