def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2 = problem_qubits
    c0, c1, c2, c3, flag = ancilla_qubits

    def compute_unsatisfied(target, zero_controls):
        for q in zero_controls:
            qc.x(q)
        qc.mcx([x0, x1, x2], target)
        for q in reversed(zero_controls):
            qc.x(q)

    # Compute whether each clause is unsatisfied.
    # Unsatisfied assignments: 001, 000, 110, 011 respectively.
    compute_unsatisfied(c0, [x0, x1])
    compute_unsatisfied(c1, [x0, x1, x2])
    compute_unsatisfied(c2, [x2])
    compute_unsatisfied(c3, [x0])

    # flag = 1 exactly when none of the clauses is unsatisfied.
    for q in [c0, c1, c2, c3]:
        qc.x(q)
    qc.mcx([c0, c1, c2, c3], flag)
    for q in reversed([c0, c1, c2, c3]):
        qc.x(q)

    qc.z(flag)

    # Uncompute the satisfying flag.
    for q in [c0, c1, c2, c3]:
        qc.x(q)
    qc.mcx([c0, c1, c2, c3], flag)
    for q in reversed([c0, c1, c2, c3]):
        qc.x(q)

    # Uncompute clause flags in reverse order.
    compute_unsatisfied(c3, [x0])
    compute_unsatisfied(c2, [x2])
    compute_unsatisfied(c1, [x0, x1, x2])
    compute_unsatisfied(c0, [x0, x1])
