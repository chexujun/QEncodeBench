import math
from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3 = problem_qubits
    clauses = ancilla_qubits[:5]

    # Each entry is (variables controlling a clause violation,
    #                  variables that must be inverted to test for 0).
    clause_data = [
        ([x0, x2, x3], [x2, x3]),  # (¬x0 ∨ x2 ∨ x3)
        ([x0, x1, x2], [x0]),      # (x0 ∨ ¬x1 ∨ ¬x2)
        ([x0, x1, x2], [x0]),      # (¬x0 ∨ ¬x1 ∨ ¬x2)
        ([x0, x1, x3], [x0]),      # (x0 ∨ ¬x1 ∨ x3)
        ([x0, x1, x3], [x0, x1, x3]),  # (x0 ∨ x1 ∨ x3)
    ]

    # Compute each clause's truth value into its ancilla.
    for target, (controls, invert_controls) in zip(clauses, clause_data):
        for q in invert_controls:
            qc.x(q)
        qc.ccx(controls[0], controls[1], target) if len(controls) == 2 else qc.mcx(controls, target)
        for q in invert_controls:
            qc.x(q)
        qc.x(target)

    # Apply -1 iff all five clauses are satisfied.
    qc.mcp(math.pi, clauses[:4], clauses[4])

    # Uncompute the clause ancillas.
    for target, (controls, invert_controls) in reversed(list(zip(clauses, clause_data))):
        qc.x(target)
        for q in invert_controls:
            qc.x(q)
        qc.ccx(controls[0], controls[1], target) if len(controls) == 2 else qc.mcx(controls, target)
        for q in invert_controls:
            qc.x(q)
