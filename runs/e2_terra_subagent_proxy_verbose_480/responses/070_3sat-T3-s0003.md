from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x = problem_qubits
    clauses = ancilla_qubits[:7]
    phase = ancilla_qubits[7]

    violations = [
        ([x[1], x[3], x[4]], [x[3], x[4]]),
        ([x[1], x[5], x[6]], [x[1], x[5]]),
        ([x[0], x[3], x[4]], []),
        ([x[0], x[1], x[6]], []),
        ([x[1], x[4], x[6]], [x[1], x[6]]),
        ([x[0], x[1], x[3]], [x[3]]),
        ([x[2], x[5], x[6]], [x[2]]),
    ]

    for target, (controls, zero_controls) in zip(clauses, violations):
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(zero_controls):
            qc.x(qubit)

    for qubit in clauses:
        qc.x(qubit)

    qc.h(phase)
    qc.mcx(clauses, phase)
    qc.h(phase)

    for qubit in reversed(clauses):
        qc.x(qubit)

    for target, (controls, zero_controls) in reversed(
            list(zip(clauses, violations))):
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(zero_controls):
            qc.x(qubit)
