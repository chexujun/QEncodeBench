from math import pi
from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3 = problem_qubits
    c0, c1, c2, c3, out = ancilla_qubits

    # Compute whether each clause is unsatisfied.
    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c0)       # x0=0, x1=1, x2=0
    qc.x(x2)
    qc.x(x0)

    qc.mcx([x0, x1, x2], c1)       # x0=1, x1=1, x2=1

    qc.x(x0)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c2)       # x0=0, x2=1, x3=0
    qc.x(x3)
    qc.x(x0)

    qc.x(x1)
    qc.x(x3)
    qc.mcx([x0, x1, x3], c3)       # x0=1, x1=0, x3=0
    qc.x(x3)
    qc.x(x1)

    # All clauses are satisfied exactly when all unsatisfied flags are zero.
    qc.x(c0)
    qc.x(c1)
    qc.x(c2)
    qc.x(c3)
    qc.mcx([c0, c1, c2, c3], out)
    qc.z(out)
    qc.mcx([c0, c1, c2, c3], out)
    qc.x(c3)
    qc.x(c2)
    qc.x(c1)
    qc.x(c0)

    # Uncompute clause flags.
    qc.x(x1)
    qc.x(x3)
    qc.mcx([x0, x1, x3], c3)
    qc.x(x3)
    qc.x(x1)

    qc.x(x0)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c2)
    qc.x(x3)
    qc.x(x0)

    qc.mcx([x0, x1, x2], c1)

    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c0)
    qc.x(x2)
    qc.x(x0)
