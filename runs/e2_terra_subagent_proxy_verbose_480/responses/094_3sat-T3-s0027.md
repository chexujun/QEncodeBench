import math
from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    flags = ancilla_qubits[:9]

    # Each tuple is (variable indices, required values) for a false clause.
    false_clauses = [
        ([0, 1, 3], [0, 0, 0]),
        ([2, 3, 4], [1, 1, 0]),
        ([1, 2, 4], [1, 0, 0]),
        ([1, 2, 5], [0, 0, 1]),
        ([2, 3, 4], [0, 0, 0]),
        ([0, 1, 3], [1, 0, 1]),
        ([1, 2, 5], [1, 1, 1]),
        ([0, 4, 5], [0, 1, 0]),
        ([0, 2, 4], [1, 0, 1]),
    ]

    def compute_false_clause(indices, values, target):
        inverted = [problem_qubits[i] for i, value in zip(indices, values)
                    if value == 0]
        for qubit in inverted:
            qc.x(qubit)
        qc.mcx([problem_qubits[i] for i in indices], target)
        for qubit in reversed(inverted):
            qc.x(qubit)

    # Compute whether each clause is false.
    for (indices, values), flag in zip(false_clauses, flags):
        compute_false_clause(indices, values, flag)

    # Apply -1 iff every clause-false flag is zero.
    # Convert the final flag to a positive target condition, and similarly
    # convert the first eight zero-controls to ordinary one-controls.
    for flag in flags:
        qc.x(flag)
    qc.mcp(math.pi, flags[:8], flags[8])
    for flag in reversed(flags):
        qc.x(flag)

    # Uncompute all clause flags.
    for (indices, values), flag in reversed(list(zip(false_clauses, flags))):
        compute_false_clause(indices, values, flag)
