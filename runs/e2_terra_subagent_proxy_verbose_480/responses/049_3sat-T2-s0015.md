from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x0, x1, x2, x3 = problem_qubits
    clause_bits = ancilla_qubits[:6]
    flag = ancilla_qubits[6]

    # For each clause, set its ancilla to 1 iff the clause is satisfied.
    # Start at 1 and toggle precisely on the unique unsatisfying assignment.
    clauses = [
        ([x0, x1, x2], [x0, x1]),
        ([x0, x2, x3], [x0, x2]),
        ([x1, x2, x3], [x2, x3]),
        ([x0, x2, x3], []),
        ([x0, x2, x3], [x2, x3]),
        ([x1, x2, x3], [x1, x2, x3]),
    ]

    def compute_clauses():
        for anc, (controls, zero_controls) in zip(clause_bits, clauses):
            qc.x(anc)
            for q in zero_controls:
                qc.x(q)
            qc.mcx(controls, anc)
            for q in reversed(zero_controls):
                qc.x(q)

    def uncompute_clauses():
        for anc, (controls, zero_controls) in reversed(
            list(zip(clause_bits, clauses))
        ):
            for q in zero_controls:
                qc.x(q)
            qc.mcx(controls, anc)
            for q in reversed(zero_controls):
                qc.x(q)
            qc.x(anc)

    compute_clauses()
    qc.mcx(clause_bits, flag)
    qc.z(flag)
    qc.mcx(clause_bits, flag)
    uncompute_clauses()
