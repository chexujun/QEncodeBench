from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3 = problem_qubits
    a0, a1, a2, a3, a4 = ancilla_qubits[:5]

    # Compute one flag per violated clause.
    qc.mcx([x1, x2, x3], a0)

    qc.x(x0)
    qc.mcx([x0, x2, x3], a1)
    qc.x(x0)

    qc.x(x1)
    qc.x(x3)
    qc.mcx([x1, x2, x3], a2)
    qc.x(x3)
    qc.x(x1)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x1, x2, x3], a3)
    qc.x(x3)
    qc.x(x2)

    qc.x(x1)
    qc.mcx([x1, x2, x3], a4)
    qc.x(x1)

    # Mark exactly the case in which no clause was violated.
    for a in (a0, a1, a2, a3, a4):
        qc.x(a)
    qc.h(a4)
    qc.mcx([a0, a1, a2, a3], a4)
    qc.h(a4)
    for a in (a4, a3, a2, a1, a0):
        qc.x(a)

    # Uncompute the violation flags.
    qc.x(x1)
    qc.mcx([x1, x2, x3], a4)
    qc.x(x1)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x1, x2, x3], a3)
    qc.x(x3)
    qc.x(x2)

    qc.x(x1)
    qc.x(x3)
    qc.mcx([x1, x2, x3], a2)
    qc.x(x3)
    qc.x(x1)

    qc.x(x0)
    qc.mcx([x0, x2, x3], a1)
    qc.x(x0)

    qc.mcx([x1, x2, x3], a0)
