from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    clause_flags = ancilla_qubits[:6]
    result = ancilla_qubits[6]

    # Each tuple is (variable_index, is_negated).
    # A clause is false exactly when all its literals are false.
    clauses = [
        [(1, True), (2, True), (5, True)],
        [(0, False), (3, False), (5, False)],
        [(1, False), (2, False), (4, True)],
        [(2, False), (3, False), (4, True)],
        [(1, True), (4, False), (5, True)],
        [(3, True), (4, True), (5, False)],
    ]

    def compute_clause_flag(literals, target):
        # Start at 1; flip to 0 iff all literals are false.
        qc.x(target)
        inverted = []
        for variable, is_negated in literals:
            # For a positive literal, false means x=0, so invert its control.
            # For a negated literal, false means x=1, requiring no inversion.
            if not is_negated:
                qc.x(x[variable])
                inverted.append(variable)
        qc.mcx([x[variable] for variable, _ in literals], target)
        for variable in reversed(inverted):
            qc.x(x[variable])

    def uncompute_clause_flag(literals, target):
        inverted = []
        for variable, is_negated in literals:
            if not is_negated:
                qc.x(x[variable])
                inverted.append(variable)
        qc.mcx([x[variable] for variable, _ in literals], target)
        for variable in reversed(inverted):
            qc.x(x[variable])
        qc.x(target)

    for literals, flag in zip(clauses, clause_flags):
        compute_clause_flag(literals, flag)

    qc.mcx(clause_flags, result)
    qc.z(result)
    qc.mcx(clause_flags, result)

    for literals, flag in reversed(list(zip(clauses, clause_flags))):
        uncompute_clause_flag(literals, flag)
