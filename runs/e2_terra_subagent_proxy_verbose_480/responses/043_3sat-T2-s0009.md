from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3 = problem_qubits
    c0, c1, c2, c3, c4, flag = ancilla_qubits

    def clause_failure(controls, target, inverted):
        for q in inverted:
            qc.x(q)
        qc.mcx(controls, target)
        for q in reversed(inverted):
            qc.x(q)

    # Compute whether each clause is false.
    clause_failure([x0, x1, x3], c0, [x1, x3])
    clause_failure([x0, x1, x2], c1, [x0, x1])
    clause_failure([x0, x1, x2], c2, [x1])
    clause_failure([x1, x2, x3], c3, [x2, x3])
    clause_failure([x0, x1, x2], c4, [x2])

    # Mark assignments for which no clause failed.
    for q in [c0, c1, c2, c3, c4]:
        qc.x(q)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    qc.z(flag)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    for q in reversed([c0, c1, c2, c3, c4]):
        qc.x(q)

    # Uncompute clause-failure flags.
    clause_failure([x0, x1, x2], c4, [x2])
    clause_failure([x1, x2, x3], c3, [x2, x3])
    clause_failure([x0, x1, x2], c2, [x1])
    clause_failure([x0, x1, x2], c1, [x0, x1])
    clause_failure([x0, x1, x3], c0, [x1, x3])
