from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x = problem_qubits
    clause_bits = ancilla_qubits[:10]
    phase_bit = ancilla_qubits[10]

    # For each clause, compute its violation bit (1 iff all literals are false).
    # Each tuple lists variables that must be inverted before the MCX.
    inverted_controls = [
        (),          # (~x0 | ~x4 | ~x5): violated by x0 x4 x5
        (3, 4),      # (~x2 | x3 | x4)
        (1,),        # (x1 | ~x3 | ~x5)
        (0, 1, 2),   # (~x0 | x1 | x2)
        (1, 3, 4),   # (~x1 | ~x3 | ~x4)
        (0, 2, 4),   # (~x0 | x2 | ~x4)
        (1, 3, 4),   # (~x1 | x3 | ~x4)
        (0, 1, 3),   # (~x0 | x1 | x3)
        (0, 3, 5),   # (~x0 | x3 | x5)
        (1, 4, 5),   # (x1 | x4 | x5)
    ]

    clause_controls = [
        (0, 4, 5),
        (2, 3, 4),
        (1, 3, 5),
        (0, 1, 2),
        (1, 3, 4),
        (0, 2, 4),
        (1, 3, 4),
        (0, 1, 3),
        (0, 3, 5),
        (1, 4, 5),
    ]

    for target, controls, flips in zip(clause_bits, clause_controls, inverted_controls):
        for i in flips:
            qc.x(x[i])
        qc.mcx([x[i] for i in controls], target)
        for i in reversed(flips):
            qc.x(x[i])

    # All clause-violation bits are zero exactly when the formula is satisfied.
    for bit in clause_bits:
        qc.x(bit)

    # Prepare |->, apply a controlled-X phase kickback, then restore |0>.
    qc.x(phase_bit)
    qc.h(phase_bit)
    qc.mcx(clause_bits, phase_bit)
    qc.h(phase_bit)
    qc.x(phase_bit)

    for bit in reversed(clause_bits):
        qc.x(bit)

    # Uncompute all clause-violation bits.
    for target, controls, flips in reversed(
        list(zip(clause_bits, clause_controls, inverted_controls))
    ):
        for i in flips:
            qc.x(x[i])
        qc.mcx([x[i] for i in controls], target)
        for i in reversed(flips):
            qc.x(x[i])
