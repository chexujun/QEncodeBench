def build_oracle(qc, problem_qubits, ancilla_qubits):
    q = problem_qubits
    clauses = ancilla_qubits[:6]
    phase_ancilla = ancilla_qubits[6]

    # For each clause: initialize its flag to 1, then flip it to 0 iff
    # the clause is unsatisfied.
    def compute_clause(flag, zero_controls):
        qc.x(flag)
        for qubit in zero_controls:
            qc.x(q[qubit])
        qc.mcx([q[i] for i in (0, 1, 2, 3) if i not in ()], flag)

    # Explicit clause computations, using only the three relevant controls.
    qc.x(clauses[0])
    qc.x(q[3])
    qc.mcx([q[0], q[2], q[3]], clauses[0])
    qc.x(q[3])

    qc.x(clauses[1])
    qc.x(q[0]); qc.x(q[1]); qc.x(q[2])
    qc.mcx([q[0], q[1], q[2]], clauses[1])
    qc.x(q[2]); qc.x(q[1]); qc.x(q[0])

    qc.x(clauses[2])
    qc.x(q[0])
    qc.mcx([q[0], q[1], q[2]], clauses[2])
    qc.x(q[0])

    qc.x(clauses[3])
    qc.x(q[1]); qc.x(q[2])
    qc.mcx([q[0], q[1], q[2]], clauses[3])
    qc.x(q[2]); qc.x(q[1])

    qc.x(clauses[4])
    qc.x(q[1])
    qc.mcx([q[0], q[1], q[2]], clauses[4])
    qc.x(q[1])

    qc.x(clauses[5])
    qc.x(q[1]); qc.x(q[2]); qc.x(q[3])
    qc.mcx([q[1], q[2], q[3]], clauses[5])
    qc.x(q[3]); qc.x(q[2]); qc.x(q[1])

    # Phase kickback: phase -1 precisely when every clause flag is 1.
    qc.h(phase_ancilla)
    qc.z(phase_ancilla)
    qc.mcx(clauses, phase_ancilla)
    qc.z(phase_ancilla)
    qc.h(phase_ancilla)

    # Uncompute clause flags.
    qc.x(q[1]); qc.x(q[2]); qc.x(q[3])
    qc.mcx([q[1], q[2], q[3]], clauses[5])
    qc.x(q[3]); qc.x(q[2]); qc.x(q[1])
    qc.x(clauses[5])

    qc.x(q[1])
    qc.mcx([q[0], q[1], q[2]], clauses[4])
    qc.x(q[1])
    qc.x(clauses[4])

    qc.x(q[1]); qc.x(q[2])
    qc.mcx([q[0], q[1], q[2]], clauses[3])
    qc.x(q[2]); qc.x(q[1])
    qc.x(clauses[3])

    qc.x(q[0])
    qc.mcx([q[0], q[1], q[2]], clauses[2])
    qc.x(q[0])
    qc.x(clauses[2])

    qc.x(q[0]); qc.x(q[1]); qc.x(q[2])
    qc.mcx([q[0], q[1], q[2]], clauses[1])
    qc.x(q[2]); qc.x(q[1]); qc.x(q[0])
    qc.x(clauses[1])

    qc.x(q[3])
    qc.mcx([q[0], q[2], q[3]], clauses[0])
    qc.x(q[3])
    qc.x(clauses[0])
