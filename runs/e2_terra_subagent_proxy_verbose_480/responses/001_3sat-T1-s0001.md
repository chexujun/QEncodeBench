def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2, x3 = problem_qubits
    c0, c1, c2, c3, c4, flag = ancilla_qubits[:6]

    # Compute clause-violation flags.
    # C0 false iff x0=1, x1=0, x3=1.
    qc.x(x1)
    qc.ccx(x0, x1, c0)
    qc.ccx(c0, x3, c0)
    qc.x(x1)

    # C1 false iff x0=1, x2=0, x3=0.
    qc.x(x2)
    qc.x(x3)
    qc.ccx(x0, x2, c1)
    qc.ccx(c1, x3, c1)
    qc.x(x3)
    qc.x(x2)

    # C2 false iff x0=0, x1=1, x3=1.
    qc.x(x0)
    qc.ccx(x0, x1, c2)
    qc.ccx(c2, x3, c2)
    qc.x(x0)

    # C3 false iff x0=0, x1=0, x2=0.
    qc.x(x0)
    qc.x(x1)
    qc.x(x2)
    qc.ccx(x0, x1, c3)
    qc.ccx(c3, x2, c3)
    qc.x(x2)
    qc.x(x1)
    qc.x(x0)

    # C4 false iff x1=0, x2=1, x3=1.
    qc.x(x1)
    qc.ccx(x1, x2, c4)
    qc.ccx(c4, x3, c4)
    qc.x(x1)

    # Mark exactly assignments having no violated clause.
    for q in (c0, c1, c2, c3, c4):
        qc.x(q)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    qc.z(flag)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    for q in (c4, c3, c2, c1, c0):
        qc.x(q)

    # Uncompute clause-violation flags in reverse order.
    qc.x(x1)
    qc.ccx(c4, x3, c4)
    qc.ccx(x1, x2, c4)
    qc.x(x1)

    qc.x(x0)
    qc.x(x1)
    qc.x(x2)
    qc.ccx(c3, x2, c3)
    qc.ccx(x0, x1, c3)
    qc.x(x2)
    qc.x(x1)
    qc.x(x0)

    qc.x(x0)
    qc.ccx(c2, x3, c2)
    qc.ccx(x0, x1, c2)
    qc.x(x0)

    qc.x(x2)
    qc.x(x3)
    qc.ccx(c1, x3, c1)
    qc.ccx(x0, x2, c1)
    qc.x(x3)
    qc.x(x2)

    qc.x(x1)
    qc.ccx(c0, x3, c0)
    qc.ccx(x0, x1, c0)
    qc.x(x1)
