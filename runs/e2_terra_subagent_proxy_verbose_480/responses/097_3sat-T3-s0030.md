from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    flags = ancilla_qubits[:8]
    result = ancilla_qubits[8]

    # Each entry is (variables to invert, three clause variables).
    # After inversion, a clause-failure flag is set iff all controls are 1.
    clauses = [
        ([1, 5], [1, 2, 5]),  # x1 OR ~x2 OR x5
        ([],     [2, 3, 4]),  # ~x2 OR ~x3 OR ~x4
        ([3],    [3, 5, 6]),  # x3 OR ~x5 OR ~x6
        ([0, 6], [0, 4, 6]),  # x0 OR ~x4 OR x6
        ([7],    [2, 5, 7]),  # ~x2 OR ~x5 OR x7
        ([],     [0, 3, 7]),  # ~x0 OR ~x3 OR ~x7
        ([6],    [4, 5, 6]),  # ~x4 OR ~x5 OR x6
        ([4, 7], [1, 4, 7]),  # ~x1 OR x4 OR x7
    ]

    def compute_clause(inverted, variables, target):
        for i in inverted:
            qc.x(problem_qubits[i])
        qc.mcx([problem_qubits[i] for i in variables], target)
        for i in reversed(inverted):
            qc.x(problem_qubits[i])

    for flag, (inverted, variables) in zip(flags, clauses):
        compute_clause(inverted, variables, flag)

    # All clauses are satisfied precisely when every failure flag is zero.
    for flag in flags:
        qc.x(flag)
    qc.mcx(flags, result)
    qc.z(result)
    qc.mcx(flags, result)
    for flag in reversed(flags):
        qc.x(flag)

    for flag, (inverted, variables) in reversed(list(zip(flags, clauses))):
        compute_clause(inverted, variables, flag)
