def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = problem_qubits
    clause_ancillas = ancilla_qubits[:8]
    result_ancilla = ancilla_qubits[8]

    clauses = [
        ([x[1], x[3], x[6]], [x[1], x[6]]),
        ([x[3], x[5], x[6]], [x[5]]),
        ([x[0], x[3], x[4]], [x[0], x[4]]),
        ([x[1], x[3], x[5]], [x[1], x[5]]),
        ([x[0], x[3], x[7]], [x[3], x[7]]),
        ([x[0], x[2], x[3]], [x[0], x[2], x[3]]),
        ([x[3], x[4], x[6]], [x[3], x[6]]),
        ([x[2], x[5], x[7]], [x[5]]),
    ]

    def toggle_clause(ancilla, controls, zero_controls):
        qc.x(ancilla)
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx(controls, ancilla)
        for qubit in reversed(zero_controls):
            qc.x(qubit)

    for ancilla, (controls, zero_controls) in zip(clause_ancillas, clauses):
        toggle_clause(ancilla, controls, zero_controls)

    qc.mcx(clause_ancillas, result_ancilla)
    qc.z(result_ancilla)
    qc.mcx(clause_ancillas, result_ancilla)

    for ancilla, (controls, zero_controls) in reversed(
        list(zip(clause_ancillas, clauses))
    ):
        toggle_clause(ancilla, controls, zero_controls)
