from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3 = problem_qubits
    c1, c2, c3, c4, c5, phase = ancilla_qubits

    # Compute whether each clause is unsatisfied.
    qc.x(x1)
    qc.x(x3)
    qc.mcx([x0, x1, x3], c1)
    qc.x(x3)
    qc.x(x1)

    qc.x(x1)
    qc.mcx([x0, x1, x2], c2)
    qc.x(x1)

    qc.x(x0)
    qc.x(x3)
    qc.mcx([x0, x1, x3], c3)
    qc.x(x3)
    qc.x(x0)

    qc.x(x0)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c4)
    qc.x(x3)
    qc.x(x0)

    qc.x(x1)
    qc.x(x3)
    qc.mcx([x0, x1, x3], c5)
    qc.x(x3)
    qc.x(x1)

    # Mark exactly when no clause is unsatisfied.
    for clause_bit in [c1, c2, c3, c4, c5]:
        qc.x(clause_bit)
    qc.mcx([c1, c2, c3, c4, c5], phase)
    qc.z(phase)
    qc.mcx([c1, c2, c3, c4, c5], phase)
    for clause_bit in [c5, c4, c3, c2, c1]:
        qc.x(clause_bit)

    # Uncompute clause-unsatisfied flags.
    qc.x(x1)
    qc.x(x3)
    qc.mcx([x0, x1, x3], c5)
    qc.x(x3)
    qc.x(x1)

    qc.x(x0)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c4)
    qc.x(x3)
    qc.x(x0)

    qc.x(x0)
    qc.x(x3)
    qc.mcx([x0, x1, x3], c3)
    qc.x(x3)
    qc.x(x0)

    qc.x(x1)
    qc.mcx([x0, x1, x2], c2)
    qc.x(x1)

    qc.x(x1)
    qc.x(x3)
    qc.mcx([x0, x1, x3], c1)
    qc.x(x3)
    qc.x(x1)
