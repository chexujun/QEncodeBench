from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    clause_flags = ancilla_qubits[:7]
    output = ancilla_qubits[7]

    # Each tuple is (controls, controls that must be 0 for the clause to fail).
    # A clause flag is computed as 1 iff its corresponding clause is satisfied.
    failed_clause_conditions = [
        ([x[0], x[1], x[6]], []),            # ~x0 | ~x1 | ~x6
        ([x[3], x[5], x[6]], [x[6]]),        # ~x3 | ~x5 | x6
        ([x[2], x[4], x[6]], [x[2], x[6]]),  # x2 | ~x4 | x6
        ([x[3], x[5], x[6]], []),            # ~x3 | ~x5 | ~x6
        ([x[1], x[5], x[6]], [x[1]]),        # x1 | ~x5 | ~x6
        ([x[3], x[4], x[5]], [x[3], x[4], x[5]]),  # x3 | x4 | x5
        ([x[0], x[1], x[2]], [x[2]]),        # ~x0 | ~x1 | x2
    ]

    def compute_clause_flag(flag, controls, zero_controls):
        qc.x(flag)
        for q in zero_controls:
            qc.x(q)
        qc.mcx(controls, flag)
        for q in reversed(zero_controls):
            qc.x(q)

    def uncompute_clause_flag(flag, controls, zero_controls):
        for q in zero_controls:
            qc.x(q)
        qc.mcx(controls, flag)
        for q in reversed(zero_controls):
            qc.x(q)
        qc.x(flag)

    for flag, (controls, zero_controls) in zip(clause_flags, failed_clause_conditions):
        compute_clause_flag(flag, controls, zero_controls)

    qc.mcx(clause_flags, output)
    qc.z(output)
    qc.mcx(clause_flags, output)

    for flag, (controls, zero_controls) in reversed(
        list(zip(clause_flags, failed_clause_conditions))
    ):
        uncompute_clause_flag(flag, controls, zero_controls)
