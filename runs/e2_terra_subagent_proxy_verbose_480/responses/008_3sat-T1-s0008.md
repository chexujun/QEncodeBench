from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x0, x1, x2, x3 = problem_qubits
    c0, c1, c2, c3, c4, result = ancilla_qubits

    # Compute clause-violation flags.
    # C1 violation: x0=1, x1=0, x2=1
    qc.x(x1)
    qc.mcx([x0, x1, x2], c0)
    qc.x(x1)

    # C2 violation: x1=1, x2=0, x3=1
    qc.x(x2)
    qc.mcx([x1, x2, x3], c1)
    qc.x(x2)

    # C3 violation: x1=1, x2=1, x3=1
    qc.mcx([x1, x2, x3], c2)

    # C4 violation: x1=0, x2=0, x3=1
    qc.x(x1)
    qc.x(x2)
    qc.mcx([x1, x2, x3], c3)
    qc.x(x2)
    qc.x(x1)

    # C5 violation: x0=0, x2=1, x3=1
    qc.x(x0)
    qc.mcx([x0, x2, x3], c4)
    qc.x(x0)

    # Convert violation flags into satisfaction flags, then AND them.
    for clause_flag in (c0, c1, c2, c3, c4):
        qc.x(clause_flag)

    qc.mcx([c0, c1, c2, c3, c4], result)
    qc.z(result)
    qc.mcx([c0, c1, c2, c3, c4], result)

    for clause_flag in (c4, c3, c2, c1, c0):
        qc.x(clause_flag)

    # Uncompute clause-violation flags.
    qc.x(x0)
    qc.mcx([x0, x2, x3], c4)
    qc.x(x0)

    qc.x(x1)
    qc.x(x2)
    qc.mcx([x1, x2, x3], c3)
    qc.x(x2)
    qc.x(x1)

    qc.mcx([x1, x2, x3], c2)

    qc.x(x2)
    qc.mcx([x1, x2, x3], c1)
    qc.x(x2)

    qc.x(x1)
    qc.mcx([x0, x1, x2], c0)
    qc.x(x1)
