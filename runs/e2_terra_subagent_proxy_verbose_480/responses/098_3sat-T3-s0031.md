```python
import math
from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    # Each of the first 12 ancillas stores whether its corresponding clause
    # is unsatisfied.  The last ancilla is not needed: a multi-controlled
    # phase is applied directly to the all-zero clause-ancilla state.
    x = problem_qubits
    clause_bits = ancilla_qubits[:12]

    # A positive literal is false when its variable is 0, so it is temporarily
    # inverted before the three-controlled X.  Negative literals need no flip.
    # Entries are the indices of positive literals in each clause.
    positive_to_flip = [
        (2, 4),       # (x2 OR x4 OR NOT x6)
        (),           # (NOT x0 OR NOT x3 OR NOT x6)
        (3, 6),       # (x3 OR NOT x4 OR x6)
        (5, 6),       # (NOT x4 OR x5 OR x6)
        (0, 3),       # (x0 OR x3 OR NOT x6)
        (2, 5, 6),    # (x2 OR x5 OR x6)
        (0, 6),       # (x0 OR NOT x5 OR x6)
        (5,),         # (NOT x3 OR x5 OR NOT x6)
        (2, 6),       # (NOT x0 OR x2 OR x6)
        (2, 3),       # (x2 OR x3 OR NOT x6)
        (4, 6),       # (x4 OR NOT x5 OR x6)
        (0,),         # (x0 OR NOT x2 OR NOT x5)
    ]
    clause_vars = [
        (2, 4, 6),
        (0, 3, 6),
        (3, 4, 6),
        (4, 5, 6),
        (0, 3, 6),
        (2, 5, 6),
        (0, 5, 6),
        (3, 5, 6),
        (0, 2, 6),
        (2, 3, 6),
        (4, 5, 6),
        (0, 2, 5),
    ]

    # Compute which clauses are unsatisfied.
    for target, variables, flips in zip(clause_bits, clause_vars, positive_to_flip):
        for bit in flips:
            qc.x(x[bit])
        qc.mcx([x[bit] for bit in variables], target)
        for bit in reversed(flips):
            qc.x(x[bit])

    # Mark exactly the state in which every clause-unsatisfied bit is zero.
    # Temporarily invert one target so the all-zero condition becomes an
    # all-zero-controlled phase on the other eleven bits and target=1.
    qc.x(clause_bits[-1])
    qc.mcp(math.pi, clause_bits[:-1], clause_bits[-1], ctrl_state=0)
    qc.x(clause_bits[-1])

    # Uncompute all clause bits, restoring every ancilla to |0>.
    for target, variables, flips in reversed(
        list(zip(clause_bits, clause_vars, positive_to_flip))
    ):
        for bit in flips:
            qc.x(x[bit])
        qc.mcx([x[bit] for bit in variables], target)
        for bit in reversed(flips):
            qc.x(x[bit])
```
