import math
from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    # The first nine ancillas record whether each clause is violated.
    clause_bits = ancilla_qubits[:9]
    flag = ancilla_qubits[9]

    # Each entry lists (variable_index, required_value) for violation.
    # A clause is violated precisely when all its listed values hold.
    violations = [
        [(1, 1), (3, 1), (6, 0)],
        [(1, 0), (6, 0), (7, 1)],
        [(2, 1), (5, 0), (6, 1)],
        [(2, 1), (3, 1), (7, 1)],
        [(0, 0), (1, 0), (6, 1)],
        [(0, 1), (1, 0), (3, 0)],
        [(2, 0), (3, 1), (6, 1)],
        [(1, 0), (4, 1), (6, 1)],
        [(2, 1), (6, 1), (7, 0)],
    ]

    def compute_violation(condition, target):
        inverted = []
        controls = []
        for index, value in condition:
            qubit = problem_qubits[index]
            controls.append(qubit)
            if value == 0:
                qc.x(qubit)
                inverted.append(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(inverted):
            qc.x(qubit)

    # Compute all violated-clause indicators.
    for condition, target in zip(violations, clause_bits):
        compute_violation(condition, target)

    # Convert "no clause is violated" into a flag.
    for target in clause_bits:
        qc.x(target)
    qc.mcx(clause_bits, flag)
    for target in reversed(clause_bits):
        qc.x(target)

    qc.z(flag)

    # Uncompute the satisfying-assignment flag.
    for target in clause_bits:
        qc.x(target)
    qc.mcx(clause_bits, flag)
    for target in reversed(clause_bits):
        qc.x(target)

    # Uncompute all violated-clause indicators.
    for condition, target in reversed(list(zip(violations, clause_bits))):
        compute_violation(condition, target)
