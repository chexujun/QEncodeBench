from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    clause = ancilla_qubits[:6]
    out = ancilla_qubits[6]

    def compute_clauses():
        # Clause 1 false iff x3=x4=x5=1.
        qc.mcx([x[3], x[4], x[5]], clause[0])

        # Clause 2 false iff x1=0, x2=1, x4=0.
        qc.x(x[1])
        qc.x(x[4])
        qc.mcx([x[1], x[2], x[4]], clause[1])
        qc.x(x[4])
        qc.x(x[1])

        # Clause 3 false iff x1=x3=1, x4=0.
        qc.x(x[4])
        qc.mcx([x[1], x[3], x[4]], clause[2])
        qc.x(x[4])

        # Clause 4 false iff x1=0, x3=x4=1.
        qc.x(x[1])
        qc.mcx([x[1], x[3], x[4]], clause[3])
        qc.x(x[1])

        # Clause 5 false iff x1=1, x3=0, x4=1.
        qc.x(x[3])
        qc.mcx([x[1], x[3], x[4]], clause[4])
        qc.x(x[3])

        # Clause 6 false iff x0=1, x3=0, x5=1.
        qc.x(x[3])
        qc.mcx([x[0], x[3], x[5]], clause[5])
        qc.x(x[3])

    compute_clauses()

    # Convert "all clauses satisfied" (all clause flags are zero)
    # into the output flag.
    for a in clause:
        qc.x(a)
    qc.mcx(clause, out)
    for a in reversed(clause):
        qc.x(a)

    qc.z(out)

    for a in clause:
        qc.x(a)
    qc.mcx(clause, out)
    for a in reversed(clause):
        qc.x(a)

    # Reverse the clause computation.
    qc.x(x[3])
    qc.mcx([x[0], x[3], x[5]], clause[5])
    qc.x(x[3])

    qc.x(x[3])
    qc.mcx([x[1], x[3], x[4]], clause[4])
    qc.x(x[3])

    qc.x(x[1])
    qc.mcx([x[1], x[3], x[4]], clause[3])
    qc.x(x[1])

    qc.x(x[4])
    qc.mcx([x[1], x[3], x[4]], clause[2])
    qc.x(x[4])

    qc.x(x[1])
    qc.x(x[4])
    qc.mcx([x[1], x[2], x[4]], clause[1])
    qc.x(x[4])
    qc.x(x[1])

    qc.mcx([x[3], x[4], x[5]], clause[0])
