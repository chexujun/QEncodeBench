def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2, x3 = problem_qubits
    c0, c1, c2, c3, c4, flag = ancilla_qubits

    # Compute whether each clause is unsatisfied.
    qc.x(x1)
    qc.mcx([x1, x2, x3], c0)
    qc.x(x1)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x1, x2, x3], c1)
    qc.x(x3)
    qc.x(x2)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c2)
    qc.x(x3)
    qc.x(x2)

    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x3], c3)
    qc.x(x1)
    qc.x(x0)

    qc.x(x2)
    qc.mcx([x0, x2, x3], c4)
    qc.x(x2)

    # Apply phase iff no clause is unsatisfied.
    for c in [c0, c1, c2, c3, c4]:
        qc.x(c)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    qc.z(flag)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    for c in [c4, c3, c2, c1, c0]:
        qc.x(c)

    # Uncompute clause-unsatisfied flags.
    qc.x(x2)
    qc.mcx([x0, x2, x3], c4)
    qc.x(x2)

    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x3], c3)
    qc.x(x1)
    qc.x(x0)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x0, x2, x3], c2)
    qc.x(x3)
    qc.x(x2)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x1, x2, x3], c1)
    qc.x(x3)
    qc.x(x2)

    qc.x(x1)
    qc.mcx([x1, x2, x3], c0)
    qc.x(x1)
