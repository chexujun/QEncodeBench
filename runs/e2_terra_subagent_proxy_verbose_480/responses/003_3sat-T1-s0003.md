def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2 = problem_qubits
    flag = ancilla_qubits[0]

    # f = x0 XOR x1 XOR (x0 AND x1) XOR (x1 AND x2)
    qc.cx(x0, flag)
    qc.cx(x1, flag)
    qc.ccx(x0, x1, flag)
    qc.ccx(x1, x2, flag)

    qc.z(flag)

    qc.ccx(x1, x2, flag)
    qc.ccx(x0, x1, flag)
    qc.cx(x1, flag)
    qc.cx(x0, flag)
