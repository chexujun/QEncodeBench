def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2 = problem_qubits
    clause_bits = ancilla_qubits[:5]

    # Each target is initialized to 1 and toggled only when its clause is false.
    def compute_clause(target, zero_controls):
        qc.x(target)
        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx([x0, x1, x2], target)
        for qubit in reversed(zero_controls):
            qc.x(qubit)

    # (¬x0 ∨ x1 ∨ x2): false for 100
    compute_clause(clause_bits[0], [x1, x2])
    # (x0 ∨ ¬x1 ∨ ¬x2): false for 011
    compute_clause(clause_bits[1], [x0])
    # (¬x0 ∨ ¬x1 ∨ ¬x2): false for 111
    compute_clause(clause_bits[2], [])
    # (x0 ∨ x1 ∨ ¬x2): false for 001
    compute_clause(clause_bits[3], [x0, x1])
    # (¬x0 ∨ ¬x1 ∨ x2): false for 110
    compute_clause(clause_bits[4], [x2])

    # Apply a -1 phase iff every clause bit is 1.
    qc.h(clause_bits[4])
    qc.mcx(clause_bits[:4], clause_bits[4])
    qc.h(clause_bits[4])

    # Uncompute clauses in reverse order.
    compute_clause(clause_bits[4], [x2])
    compute_clause(clause_bits[3], [x0, x1])
    compute_clause(clause_bits[2], [])
    compute_clause(clause_bits[1], [x0])
    compute_clause(clause_bits[0], [x1, x2])
