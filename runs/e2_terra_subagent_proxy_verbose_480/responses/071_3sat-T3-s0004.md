from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x = problem_qubits
    clause = ancilla_qubits[:7]

    # Each tuple gives the variables in a clause and the values that make
    # that clause false.
    false_patterns = [
        ((1, 3, 5), (1, 1, 1)),
        ((0, 1, 2), (1, 1, 0)),
        ((1, 2, 5), (1, 1, 1)),
        ((1, 4, 5), (1, 0, 1)),
        ((0, 1, 5), (1, 1, 0)),
        ((2, 4, 5), (0, 1, 0)),
        ((2, 3, 4), (1, 1, 1)),
    ]

    def toggle_on_false(target: int, variables: tuple[int, ...],
                        values: tuple[int, ...]) -> None:
        # Convert the clause's false pattern to an all-ones control pattern.
        for variable, value in zip(variables, values):
            if value == 0:
                qc.x(x[variable])

        qc.mcx([x[variable] for variable in variables], target)

        for variable, value in reversed(tuple(zip(variables, values))):
            if value == 0:
                qc.x(x[variable])

    # Compute the seven clause-satisfaction bits.
    for target, (variables, values) in zip(clause, false_patterns):
        qc.x(target)
        toggle_on_false(target, variables, values)

    # Apply -1 exactly when all seven clauses are satisfied.
    qc.h(clause[6])
    qc.mcx(clause[:6], clause[6])
    qc.h(clause[6])

    # Uncompute every clause bit.
    for target, (variables, values) in reversed(
            tuple(zip(clause, false_patterns))):
        toggle_on_false(target, variables, values)
        qc.x(target)
