from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    clauses = [
        [(0, False), (2, False), (4, True)],
        [(2, True), (4, True), (5, False)],
        [(1, True), (2, True), (4, False)],
        [(1, True), (2, True), (5, False)],
        [(0, False), (1, False), (4, False)],
        [(0, True), (2, True), (4, True)],
        [(0, False), (4, False), (5, True)],
    ]

    def toggle_clause_value(clause, target):
        positive_qubits = [
            problem_qubits[index] for index, is_positive in clause
            if is_positive
        ]
        controls = [problem_qubits[index] for index, _ in clause]

        for qubit in positive_qubits:
            qc.x(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(positive_qubits):
            qc.x(qubit)
        qc.x(target)

    clause_qubits = ancilla_qubits[:7]
    phase_qubit = ancilla_qubits[7]

    for clause, target in zip(clauses, clause_qubits):
        toggle_clause_value(clause, target)

    qc.mcx(clause_qubits, phase_qubit)
    qc.z(phase_qubit)
    qc.mcx(clause_qubits, phase_qubit)

    for clause, target in reversed(list(zip(clauses, clause_qubits))):
        toggle_clause_value(clause, target)
