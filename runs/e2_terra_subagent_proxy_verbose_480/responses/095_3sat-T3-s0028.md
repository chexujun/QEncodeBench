def build_oracle(qc, problem_qubits, ancilla_qubits):
    q = problem_qubits
    clause_bits = ancilla_qubits[:8]
    flag = ancilla_qubits[8]

    # For each clause, list the positive literals.  A clause is false when
    # positive literals are 0 and negated literals are 1.
    clauses = [
        ([0, 1, 7], [0, 1, 7]),
        ([3, 4, 6], [4, 6]),
        ([0, 2, 4], [0]),
        ([0, 1, 3], [0]),
        ([0, 6, 7], [6, 7]),
        ([5, 6, 7], [5]),
        ([0, 4, 6], [0, 6]),
        ([0, 6, 7], [0]),
    ]

    # Compute one bit per clause: 1 iff that clause is violated.
    for target, (variables, positive_variables) in zip(clause_bits, clauses):
        for i in positive_variables:
            qc.x(q[i])
        qc.mcx([q[i] for i in variables], target)
        for i in reversed(positive_variables):
            qc.x(q[i])

    # Set flag iff every clause-violation bit is zero.
    for bit in clause_bits:
        qc.x(bit)
    qc.mcx(clause_bits, flag)
    for bit in reversed(clause_bits):
        qc.x(bit)

    qc.z(flag)

    # Uncompute the conjunction flag.
    for bit in clause_bits:
        qc.x(bit)
    qc.mcx(clause_bits, flag)
    for bit in reversed(clause_bits):
        qc.x(bit)

    # Uncompute all clause-violation bits.
    for target, (variables, positive_variables) in reversed(
        list(zip(clause_bits, clauses))
    ):
        for i in positive_variables:
            qc.x(q[i])
        qc.mcx([q[i] for i in variables], target)
        for i in reversed(positive_variables):
            qc.x(q[i])
