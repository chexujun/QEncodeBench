```python
import math
from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x0, x1, x2, x3, x4 = problem_qubits
    a0, a1, a2, a3, _ = ancilla_qubits

    # Compute clause-violation flags.
    # a0 = x1 AND x2 AND (NOT x3)
    qc.x(x3)
    qc.mcx([x1, x2, x3], a0)
    qc.x(x3)

    # a1 = x0 AND x3 AND x4
    qc.mcx([x0, x3, x4], a1)

    # a2 = (NOT x0) AND (NOT x1) AND (NOT x2)
    qc.x(x0)
    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], a2)
    qc.x(x2)
    qc.x(x1)
    qc.x(x0)

    # a3 = x1 AND (NOT x2) AND (NOT x4)
    qc.x(x2)
    qc.x(x4)
    qc.mcx([x1, x2, x4], a3)
    qc.x(x4)
    qc.x(x2)

    # All clauses are satisfied exactly when every violation flag is zero.
    qc.x(a0)
    qc.x(a1)
    qc.x(a2)
    qc.x(a3)
    qc.mcp(math.pi, [a0, a1, a2, a3])
    qc.x(a3)
    qc.x(a2)
    qc.x(a1)
    qc.x(a0)

    # Uncompute clause-violation flags.
    qc.x(x2)
    qc.x(x4)
    qc.mcx([x1, x2, x4], a3)
    qc.x(x4)
    qc.x(x2)

    qc.x(x0)
    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], a2)
    qc.x(x2)
    qc.x(x1)
    qc.x(x0)

    qc.mcx([x0, x3, x4], a1)

    qc.x(x3)
    qc.mcx([x1, x2, x3], a0)
    qc.x(x3)
```
