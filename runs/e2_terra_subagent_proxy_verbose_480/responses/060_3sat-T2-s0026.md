from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x0, x1, x2, x3, x4 = problem_qubits
    c0, c1, c2, c3, out = ancilla_qubits

    # Compute whether each clause is false.
    # Clause 1 false: x2=0, x3=1, x4=0
    qc.x(x2)
    qc.x(x4)
    qc.mcx([x2, x3, x4], c0)
    qc.x(x4)
    qc.x(x2)

    # Clause 2 false: x2=1, x3=1, x4=0
    qc.x(x4)
    qc.mcx([x2, x3, x4], c1)
    qc.x(x4)

    # Clause 3 false: x2=0, x3=0, x4=1
    qc.x(x2)
    qc.x(x3)
    qc.mcx([x2, x3, x4], c2)
    qc.x(x3)
    qc.x(x2)

    # Clause 4 false: x1=0, x2=1, x4=1
    qc.x(x1)
    qc.mcx([x1, x2, x4], c3)
    qc.x(x1)

    # Mark iff no clause is false.
    qc.x(c0)
    qc.x(c1)
    qc.x(c2)
    qc.x(c3)
    qc.mcx([c0, c1, c2, c3], out)
    qc.z(out)
    qc.mcx([c0, c1, c2, c3], out)
    qc.x(c3)
    qc.x(c2)
    qc.x(c1)
    qc.x(c0)

    # Uncompute clause-false flags.
    qc.x(x1)
    qc.mcx([x1, x2, x4], c3)
    qc.x(x1)

    qc.x(x2)
    qc.x(x3)
    qc.mcx([x2, x3, x4], c2)
    qc.x(x3)
    qc.x(x2)

    qc.x(x4)
    qc.mcx([x2, x3, x4], c1)
    qc.x(x4)

    qc.x(x2)
    qc.x(x4)
    qc.mcx([x2, x3, x4], c0)
    qc.x(x4)
    qc.x(x2)
