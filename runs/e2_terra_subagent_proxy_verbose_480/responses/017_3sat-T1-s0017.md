```python
from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x0, x1, x2 = problem_qubits
    c0, c1, c2, c3, c4, flag = ancilla_qubits

    # Compute whether each clause is unsatisfied.
    # Clause 0: (~x0 OR ~x1 OR x2), false iff x0 x1 ~x2.
    qc.x(x2)
    qc.mcx([x0, x1, x2], c0)
    qc.x(x2)

    # Clause 1: (~x0 OR x1 OR x2), false iff x0 ~x1 ~x2.
    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c1)
    qc.x(x2)
    qc.x(x1)

    # Clause 2: (x0 OR x1 OR x2), false iff ~x0 ~x1 ~x2.
    qc.x(x0)
    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c2)
    qc.x(x2)
    qc.x(x1)
    qc.x(x0)

    # Clause 3: (~x0 OR x1 OR ~x2), false iff x0 ~x1 x2.
    qc.x(x1)
    qc.mcx([x0, x1, x2], c3)
    qc.x(x1)

    # Clause 4: (x0 OR ~x1 OR x2), false iff ~x0 x1 ~x2.
    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c4)
    qc.x(x2)
    qc.x(x0)

    # flag = 1 exactly when every clause is satisfied.
    for clause_bit in (c0, c1, c2, c3, c4):
        qc.x(clause_bit)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    qc.z(flag)
    qc.mcx([c0, c1, c2, c3, c4], flag)
    for clause_bit in (c4, c3, c2, c1, c0):
        qc.x(clause_bit)

    # Uncompute clause-unsatisfied bits.
    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c4)
    qc.x(x2)
    qc.x(x0)

    qc.x(x1)
    qc.mcx([x0, x1, x2], c3)
    qc.x(x1)

    qc.x(x0)
    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c2)
    qc.x(x2)
    qc.x(x1)
    qc.x(x0)

    qc.x(x1)
    qc.x(x2)
    qc.mcx([x0, x1, x2], c1)
    qc.x(x2)
    qc.x(x1)

    qc.x(x2)
    qc.mcx([x0, x1, x2], c0)
    qc.x(x2)
```
