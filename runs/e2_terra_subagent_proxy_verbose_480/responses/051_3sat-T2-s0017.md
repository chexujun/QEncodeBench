```python
import math
from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3 = problem_qubits
    c0, c1, c2, c3, c4, phase = ancilla_qubits

    def compute_unsatisfied_clauses():
        # (¬x0 ∨ ¬x2 ∨ ¬x3) is false iff x0=x2=x3=1
        qc.mcx([x0, x2, x3], c0)

        # (x0 ∨ x2 ∨ x3) is false iff x0=x2=x3=0
        qc.x([x0, x2, x3])
        qc.mcx([x0, x2, x3], c1)
        qc.x([x0, x2, x3])

        # (x0 ∨ x2 ∨ ¬x3) is false iff x0=x2=0, x3=1
        qc.x([x0, x2])
        qc.mcx([x0, x2, x3], c2)
        qc.x([x0, x2])

        # (x1 ∨ x2 ∨ ¬x3) is false iff x1=x2=0, x3=1
        qc.x([x1, x2])
        qc.mcx([x1, x2, x3], c3)
        qc.x([x1, x2])

        # (x0 ∨ x1 ∨ ¬x2) is false iff x0=x1=0, x2=1
        qc.x([x0, x1])
        qc.mcx([x0, x1, x2], c4)
        qc.x([x0, x1])

    compute_unsatisfied_clauses()

    # Convert unsatisfied-clause flags to satisfied-clause controls.
    qc.x([c0, c1, c2, c3, c4])
    qc.x(phase)
    qc.mcp(math.pi, [c0, c1, c2, c3, c4], phase)
    qc.x(phase)
    qc.x([c0, c1, c2, c3, c4])

    compute_unsatisfied_clauses()
```
