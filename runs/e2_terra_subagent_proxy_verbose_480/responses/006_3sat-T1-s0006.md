def build_oracle(qc, problem_qubits, ancilla_qubits):
    x0, x1, x2 = problem_qubits
    c0, c1, c2, c3, c4, flag = ancilla_qubits

    def mark_pattern(target, bits):
        controls = [x0, x1, x2]
        for qubit, bit in zip(controls, bits):
            if bit == 0:
                qc.x(qubit)
        qc.mcx(controls, target)
        for qubit, bit in reversed(list(zip(controls, bits))):
            if bit == 0:
                qc.x(qubit)

    # Compute whether each clause is unsatisfied.
    mark_pattern(c0, (1, 0, 0))
    mark_pattern(c1, (0, 0, 1))
    mark_pattern(c2, (0, 1, 0))
    mark_pattern(c3, (1, 0, 1))
    mark_pattern(c4, (1, 1, 0))

    # All clauses are satisfied exactly when every unsatisfied-clause flag is 0.
    qc.mcx([c0, c1, c2, c3, c4], flag, ctrl_state=0)
    qc.z(flag)
    qc.mcx([c0, c1, c2, c3, c4], flag, ctrl_state=0)

    # Uncompute clause flags.
    mark_pattern(c4, (1, 1, 0))
    mark_pattern(c3, (1, 0, 1))
    mark_pattern(c2, (0, 1, 0))
    mark_pattern(c1, (0, 0, 1))
    mark_pattern(c0, (1, 0, 0))
