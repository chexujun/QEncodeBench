def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = problem_qubits
    clause = ancilla_qubits[:6]
    flag = ancilla_qubits[6]

    def compute_fail(controls, zero_controls, target):
        for q in zero_controls:
            qc.x(q)
        qc.mcx(controls, target)
        for q in reversed(zero_controls):
            qc.x(q)

    # Compute whether each clause is false.
    compute_fail([x[0], x[1], x[3]], [x[3]], clause[0])
    compute_fail([x[1], x[4], x[5]], [x[1], x[4]], clause[1])
    compute_fail([x[1], x[2], x[3]], [x[1], x[2], x[3]], clause[2])
    compute_fail([x[0], x[1], x[4]], [], clause[3])
    compute_fail([x[0], x[1], x[2]], [], clause[4])
    compute_fail([x[1], x[3], x[4]], [x[3]], clause[5])

    # flag = 1 exactly when no clause is false.
    for q in clause:
        qc.x(q)
    qc.mcx(clause, flag)
    for q in reversed(clause):
        qc.x(q)

    qc.z(flag)

    # Uncompute the satisfying-assignment flag.
    for q in clause:
        qc.x(q)
    qc.mcx(clause, flag)
    for q in reversed(clause):
        qc.x(q)

    # Uncompute clause-failure flags.
    compute_fail([x[1], x[3], x[4]], [x[3]], clause[5])
    compute_fail([x[0], x[1], x[2]], [], clause[4])
    compute_fail([x[0], x[1], x[4]], [], clause[3])
    compute_fail([x[1], x[2], x[3]], [x[1], x[2], x[3]], clause[2])
    compute_fail([x[1], x[4], x[5]], [x[1], x[4]], clause[1])
    compute_fail([x[0], x[1], x[3]], [x[3]], clause[0])
