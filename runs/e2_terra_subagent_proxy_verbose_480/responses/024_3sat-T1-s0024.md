from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x0, x1, x2 = problem_qubits
    c0, c1, c2, c3, phase = ancilla_qubits

    def compute_clause(target, positive_literals):
        for qubit in positive_literals:
            qc.x(qubit)
        qc.mcx([x0, x1, x2], target)
        for qubit in reversed(positive_literals):
            qc.x(qubit)
        qc.x(target)

    # Compute the four clauses.
    compute_clause(c0, [])          # ~x0 OR ~x1 OR ~x2
    compute_clause(c1, [x0])        #  x0 OR ~x1 OR ~x2
    compute_clause(c2, [x0, x2])    #  x0 OR ~x1 OR  x2
    compute_clause(c3, [x0, x1])    #  x0 OR  x1 OR ~x2

    # Apply -1 iff every clause is satisfied.
    qc.h(phase)
    qc.mcx([c0, c1, c2, c3], phase)
    qc.h(phase)

    # Uncompute clauses in reverse order.
    qc.x(c3)
    qc.x(x1)
    qc.x(x0)
    qc.mcx([x0, x1, x2], c3)
    qc.x(x0)
    qc.x(x1)

    qc.x(c2)
    qc.x(x2)
    qc.x(x0)
    qc.mcx([x0, x1, x2], c2)
    qc.x(x0)
    qc.x(x2)

    qc.x(c1)
    qc.x(x0)
    qc.mcx([x0, x1, x2], c1)
    qc.x(x0)

    qc.x(c0)
    qc.mcx([x0, x1, x2], c0)
