from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    clauses = ancilla_qubits[:11]
    result = ancilla_qubits[11]

    # Each tuple is (variable index, is_positive_literal).
    clause_literals = [
        [(2, False), (5, True),  (6, False)],
        [(0, False), (1, False), (6, True)],
        [(0, True),  (2, True),  (4, True)],
        [(0, True),  (3, True),  (6, True)],
        [(2, True),  (4, True),  (5, False)],
        [(3, True),  (5, False), (6, True)],
        [(2, False), (5, False), (6, True)],
        [(4, True),  (5, True),  (6, True)],
        [(2, True),  (3, False), (5, True)],
        [(0, True),  (4, False), (5, True)],
        [(0, False), (1, False), (2, False)],
    ]

    def compute_clause(target, literals):
        # Start at 1; toggle to 0 precisely when all literals are false.
        qc.x(target)
        positive_controls = [
            problem_qubits[var] for var, positive in literals if positive
        ]
        for qubit in positive_controls:
            qc.x(qubit)
        qc.ccx(
            problem_qubits[literals[0][0]],
            problem_qubits[literals[1][0]],
            target,
        )
        # The third literal is incorporated by replacing the two-control
        # implementation below with a three-controlled X.
        qc.ccx(
            problem_qubits[literals[0][0]],
            problem_qubits[literals[1][0]],
            target,
        )
        for qubit in positive_controls:
            qc.x(qubit)

    # Compute every clause flag: 1 iff that clause is satisfied.
    for flag, literals in zip(clauses, clause_literals):
        qc.x(flag)
        positives = [problem_qubits[v] for v, pos in literals if pos]
        for qubit in positives:
            qc.x(qubit)
        qc.mcx([problem_qubits[v] for v, _ in literals], flag)
        for qubit in reversed(positives):
            qc.x(qubit)

    qc.mcx(clauses, result)
    qc.z(result)
    qc.mcx(clauses, result)

    # Uncompute clause flags.
    for flag, literals in reversed(list(zip(clauses, clause_literals))):
        positives = [problem_qubits[v] for v, pos in literals if pos]
        for qubit in positives:
            qc.x(qubit)
        qc.mcx([problem_qubits[v] for v, _ in literals], flag)
        for qubit in reversed(positives):
            qc.x(qubit)
        qc.x(flag)
