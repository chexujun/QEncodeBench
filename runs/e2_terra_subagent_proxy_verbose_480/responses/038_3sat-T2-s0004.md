from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2, x3, x4 = problem_qubits
    c0, c1, c2, c3, result = ancilla_qubits

    qc.mcx([x0, x1, x2], c0)

    qc.x(x1)
    qc.mcx([x0, x1, x2], c1)
    qc.x(x1)

    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x4], c2)
    qc.x(x1)
    qc.x(x0)

    qc.x(x2)
    qc.mcx([x0, x2, x3], c3)
    qc.x(x2)

    qc.x(c0)
    qc.x(c1)
    qc.x(c2)
    qc.x(c3)
    qc.mcx([c0, c1, c2, c3], result)
    qc.z(result)
    qc.mcx([c0, c1, c2, c3], result)
    qc.x(c3)
    qc.x(c2)
    qc.x(c1)
    qc.x(c0)

    qc.x(x2)
    qc.mcx([x0, x2, x3], c3)
    qc.x(x2)

    qc.x(x0)
    qc.x(x1)
    qc.mcx([x0, x1, x4], c2)
    qc.x(x1)
    qc.x(x0)

    qc.x(x1)
    qc.mcx([x0, x1, x2], c1)
    qc.x(x1)

    qc.mcx([x0, x1, x2], c0)
