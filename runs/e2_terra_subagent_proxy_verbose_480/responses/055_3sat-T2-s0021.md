from qiskit import QuantumCircuit

def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x = problem_qubits
    clause = ancilla_qubits[:6]
    result = ancilla_qubits[6]

    # Compute each clause's UNSATISFIED flag.
    qc.x(x[1])
    qc.mcx([x[0], x[1], x[2]], clause[0])
    qc.x(x[1])

    qc.x(x[2])
    qc.x(x[4])
    qc.mcx([x[1], x[2], x[4]], clause[1])
    qc.x(x[4])
    qc.x(x[2])

    qc.x(x[0])
    qc.x(x[3])
    qc.mcx([x[0], x[1], x[3]], clause[2])
    qc.x(x[3])
    qc.x(x[0])

    qc.x(x[3])
    qc.mcx([x[1], x[3], x[4]], clause[3])
    qc.x(x[3])

    qc.x(x[1])
    qc.mcx([x[1], x[2], x[3]], clause[4])
    qc.x(x[1])

    qc.x(x[3])
    qc.x(x[4])
    qc.mcx([x[1], x[3], x[4]], clause[5])
    qc.x(x[4])
    qc.x(x[3])

    # Mark precisely when every unsatisfied flag is zero.
    for q in clause:
        qc.x(q)
    qc.mcx(clause, result)
    qc.z(result)
    qc.mcx(clause, result)
    for q in reversed(clause):
        qc.x(q)

    # Uncompute clause flags.
    qc.x(x[3])
    qc.x(x[4])
    qc.mcx([x[1], x[3], x[4]], clause[5])
    qc.x(x[4])
    qc.x(x[3])

    qc.x(x[1])
    qc.mcx([x[1], x[2], x[3]], clause[4])
    qc.x(x[1])

    qc.x(x[3])
    qc.mcx([x[1], x[3], x[4]], clause[3])
    qc.x(x[3])

    qc.x(x[0])
    qc.x(x[3])
    qc.mcx([x[0], x[1], x[3]], clause[2])
    qc.x(x[3])
    qc.x(x[0])

    qc.x(x[2])
    qc.x(x[4])
    qc.mcx([x[1], x[2], x[4]], clause[1])
    qc.x(x[4])
    qc.x(x[2])

    qc.x(x[1])
    qc.mcx([x[0], x[1], x[2]], clause[0])
    qc.x(x[1])
