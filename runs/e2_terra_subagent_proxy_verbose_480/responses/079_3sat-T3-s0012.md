from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x = problem_qubits
    clause = ancilla_qubits[:6]
    phase = ancilla_qubits[6]
    work = ancilla_qubits[7:11]

    # An equivalent simplified CNF is:
    # (x3 OR x5)
    # AND (x1 OR NOT x3)
    # AND (x1 OR NOT x4)
    # AND (x3 OR NOT x2)
    # AND (NOT x0 OR NOT x5)
    # AND (x0 OR x2 OR NOT x1).
    #
    # Each tuple lists the variables that must be 0 in the unique
    # falsifying pattern, followed by those that must be 1.
    patterns = [
        ([3, 5], []),
        ([1], [3]),
        ([1], [4]),
        ([3], [2]),
        ([], [0, 5]),
        ([0, 2], [1]),
    ]

    def toggle_on_pattern(target: int, zeros: list[int],
                          ones: list[int]) -> None:
        for i in zeros:
            qc.x(x[i])
        controls = [x[i] for i in zeros] + [x[i] for i in ones]
        if len(controls) == 2:
            qc.ccx(controls[0], controls[1], target)
        else:
            qc.mcx(controls, target)
        for i in reversed(zeros):
            qc.x(x[i])

    # Compute the six clause-satisfaction bits.
    for target, (zeros, ones) in zip(clause, patterns):
        qc.x(target)
        toggle_on_pattern(target, zeros, ones)

    # Apply the phase exactly when all simplified clauses are satisfied.
    qc.mcx(clause, phase, ancilla_qubits=work, mode="v-chain")
    qc.z(phase)
    qc.mcx(clause, phase, ancilla_qubits=work, mode="v-chain")

    # Uncompute all clause bits.
    for target, (zeros, ones) in reversed(list(zip(clause, patterns))):
        toggle_on_pattern(target, zeros, ones)
        qc.x(target)
