```python
from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    x0, x1, x2, x3 = problem_qubits
    c0, c1, c2, c3, phase = ancilla_qubits

    # Compute whether each clause is violated.
    qc.mcx([x0, x1, x3], c0)

    qc.x(x1)
    qc.mcx([x1, x2, x3], c1)
    qc.x(x1)

    qc.x(x0)
    qc.x(x3)
    qc.mcx([x0, x1, x3], c2)
    qc.x(x3)
    qc.x(x0)

    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x2, x3], c3)
    qc.x(x2)
    qc.x(x0)

    # Mark when no clause is violated.
    qc.x(c0)
    qc.x(c1)
    qc.x(c2)
    qc.x(c3)
    qc.x(phase)
    qc.h(phase)
    qc.mcx([c0, c1, c2, c3], phase)
    qc.h(phase)
    qc.x(phase)
    qc.x(c3)
    qc.x(c2)
    qc.x(c1)
    qc.x(c0)

    # Uncompute clause-violation ancillas.
    qc.x(x0)
    qc.x(x2)
    qc.mcx([x0, x2, x3], c3)
    qc.x(x2)
    qc.x(x0)

    qc.x(x0)
    qc.x(x3)
    qc.mcx([x0, x1, x3], c2)
    qc.x(x3)
    qc.x(x0)

    qc.x(x1)
    qc.mcx([x1, x2, x3], c1)
    qc.x(x1)

    qc.mcx([x0, x1, x3], c0)
```
