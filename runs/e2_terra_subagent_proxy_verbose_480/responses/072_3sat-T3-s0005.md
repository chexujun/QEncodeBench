from math import pi
from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    clauses = [
        ((0, True),  (2, True),  (4, False)),
        ((0, False), (3, True),  (5, False)),
        ((2, True),  (3, False), (5, False)),
        ((2, False), (3, True),  (6, False)),
        ((2, True),  (4, True),  (5, False)),
        ((0, True),  (3, False), (7, False)),
        ((1, True),  (2, False), (4, True)),
        ((2, False), (3, False), (5, True)),
        ((1, False), (2, True),  (4, False)),
        ((0, False), (1, True),  (2, False)),
    ]
    clause_ancillas = ancilla_qubits[:10]

    def compute_clause(clause, target):
        # Start at 1 and toggle to 0 exactly when all three literals are false.
        qc.x(target)
        positive_vars = [i for i, is_positive in clause if is_positive]
        for i in positive_vars:
            qc.x(problem_qubits[i])
        qc.mcx([problem_qubits[i] for i, _ in clause], target)
        for i in reversed(positive_vars):
            qc.x(problem_qubits[i])

    def uncompute_clause(clause, target):
        positive_vars = [i for i, is_positive in clause if is_positive]
        for i in positive_vars:
            qc.x(problem_qubits[i])
        qc.mcx([problem_qubits[i] for i, _ in clause], target)
        for i in reversed(positive_vars):
            qc.x(problem_qubits[i])
        qc.x(target)

    for clause, target in zip(clauses, clause_ancillas):
        compute_clause(clause, target)

    qc.mcp(pi, clause_ancillas[:-1], clause_ancillas[-1])

    for clause, target in reversed(list(zip(clauses, clause_ancillas))):
        uncompute_clause(clause, target)
