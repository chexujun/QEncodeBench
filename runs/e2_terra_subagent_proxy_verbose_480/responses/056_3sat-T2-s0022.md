def build_oracle(qc, problem_qubits, ancilla_qubits):
    x = problem_qubits
    clause_flags = ancilla_qubits[:5]
    phase_flag = ancilla_qubits[5]

    # Each tuple specifies the variables that must be 0 and 1,
    # respectively, for its clause to be false.
    violations = [
        ([3, 5], [2]),  # (!x2 OR x3 OR x5)
        ([2, 4, 5], []),  # (x2 OR x4 OR x5)
        ([3], [1, 5]),  # (!x1 OR x3 OR !x5)
        ([5], [1, 4]),  # (!x1 OR !x4 OR x5)
        ([0], [2, 3]),  # (x0 OR !x2 OR !x3)
    ]

    def compute_violation(zero_vars, one_vars, target):
        for i in zero_vars:
            qc.x(x[i])
        qc.mcx([x[i] for i in zero_vars + one_vars], target)
        for i in reversed(zero_vars):
            qc.x(x[i])

    # Compute one flag per violated clause.
    for (zero_vars, one_vars), flag in zip(violations, clause_flags):
        compute_violation(zero_vars, one_vars, flag)

    # Mark exactly when no clause is violated.
    for flag in clause_flags:
        qc.x(flag)
    qc.mcx(clause_flags, phase_flag)
    qc.z(phase_flag)
    qc.mcx(clause_flags, phase_flag)
    for flag in reversed(clause_flags):
        qc.x(flag)

    # Uncompute the clause-violation flags.
    for (zero_vars, one_vars), flag in reversed(list(zip(violations, clause_flags))):
        compute_violation(zero_vars, one_vars, flag)
