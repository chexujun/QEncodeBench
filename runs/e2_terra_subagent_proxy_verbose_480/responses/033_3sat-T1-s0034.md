from qiskit import QuantumCircuit

def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    x0, x1, x2 = problem_qubits
    c0, c1, c2, c3, flag = ancilla_qubits

    def mark_unsatisfied(target, false_values):
        for qubit, value in zip((x0, x1, x2), false_values):
            if value == 0:
                qc.x(qubit)
        qc.mcx([x0, x1, x2], target)
        for qubit, value in reversed(list(zip((x0, x1, x2), false_values))):
            if value == 0:
                qc.x(qubit)

    # Compute whether each clause is unsatisfied.
    mark_unsatisfied(c0, (1, 0, 1))
    mark_unsatisfied(c1, (1, 0, 0))
    mark_unsatisfied(c2, (0, 0, 1))
    mark_unsatisfied(c3, (1, 1, 0))

    # All clauses are satisfied precisely when every unsatisfied flag is 0.
    qc.mcx([c0, c1, c2, c3], flag, ctrl_state="0000")
    qc.z(flag)
    qc.mcx([c0, c1, c2, c3], flag, ctrl_state="0000")

    # Uncompute clause flags.
    mark_unsatisfied(c3, (1, 1, 0))
    mark_unsatisfied(c2, (0, 0, 1))
    mark_unsatisfied(c1, (1, 0, 0))
    mark_unsatisfied(c0, (1, 0, 1))
