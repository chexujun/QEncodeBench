from qiskit import QuantumCircuit


def build_oracle(qc: QuantumCircuit, problem_qubits: list[int],
                 ancilla_qubits: list[int]) -> None:
    clauses = [
        ((0, False), (2, False), (4, False)),
        ((4, True),  (6, True),  (7, True)),
        ((0, True),  (2, True),  (6, True)),
        ((0, False), (6, False), (7, False)),
        ((1, False), (5, True),  (7, True)),
        ((0, False), (4, True),  (7, True)),
        ((1, False), (3, False), (7, False)),
        ((1, True),  (3, True),  (4, True)),
        ((2, False), (6, True),  (7, False)),
        ((2, False), (3, False), (4, False)),
    ]

    clause_qubits = ancilla_qubits[:10]
    phase_qubit = ancilla_qubits[10]

    def toggle_clause(clause, target):
        # Toggle target exactly when every literal in the clause is false.
        zero_controls = [
            problem_qubits[index]
            for index, is_positive in clause
            if is_positive
        ]
        controls = [problem_qubits[index] for index, _ in clause]

        for qubit in zero_controls:
            qc.x(qubit)
        qc.mcx(controls, target)
        for qubit in reversed(zero_controls):
            qc.x(qubit)

    # Compute each clause's satisfaction bit.
    for clause, target in zip(clauses, clause_qubits):
        toggle_clause(clause, target)
        qc.x(target)

    # Compute the conjunction, apply its phase, and uncompute it.
    qc.mcx(
        clause_qubits,
        phase_qubit,
        ancilla_qubits=problem_qubits,
        mode="v-chain-dirty",
    )
    qc.z(phase_qubit)
    qc.mcx(
        clause_qubits,
        phase_qubit,
        ancilla_qubits=problem_qubits,
        mode="v-chain-dirty",
    )

    # Uncompute all clause bits.
    for clause, target in reversed(list(zip(clauses, clause_qubits))):
        qc.x(target)
        toggle_clause(clause, target)
