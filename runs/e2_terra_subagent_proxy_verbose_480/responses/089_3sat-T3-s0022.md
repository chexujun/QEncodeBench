import math
from qiskit import QuantumCircuit


def build_oracle(
    qc: QuantumCircuit,
    problem_qubits: list[int],
    ancilla_qubits: list[int],
) -> None:
    # Each clause ancilla stores whether that clause is violated.
    # A positive literal x_i is false when x_i = 0, so invert it temporarily.
    clauses = [
        ([0, 5], [1]),       # x0 OR ~x1 OR x5
        ([4], [2, 3]),       # ~x2 OR ~x3 OR x4
        ([4, 5], [1]),       # ~x1 OR x4 OR x5
        ([2, 4, 5], []),     # x2 OR x4 OR x5
        ([5], [0, 3]),       # ~x0 OR ~x3 OR x5
        ([2], [0, 6]),       # ~x0 OR x2 OR ~x6
        ([5], [0, 2]),       # ~x0 OR ~x2 OR x5
        ([1, 3], [4]),       # x1 OR x3 OR ~x4
        ([3, 4, 5], []),     # x3 OR x4 OR x5
        ([2, 4], [6]),       # x2 OR x4 OR ~x6
        ([2, 6], [3]),       # x2 OR ~x3 OR x6
    ]

    def compute_clause_violation(positive, negative, target):
        for i in positive:
            qc.x(problem_qubits[i])
        qc.mcx(
            [problem_qubits[i] for i in positive + negative],
            target,
        )
        for i in reversed(positive):
            qc.x(problem_qubits[i])

    clause_flags = ancilla_qubits[:11]

    for (positive, negative), flag in zip(clauses, clause_flags):
        compute_clause_violation(positive, negative, flag)

    # All clauses are satisfied exactly when every violation flag is zero.
    for flag in clause_flags:
        qc.x(flag)
    qc.mcp(math.pi, clause_flags[:-1], clause_flags[-1])
    for flag in reversed(clause_flags):
        qc.x(flag)

    for (positive, negative), flag in reversed(list(zip(clauses, clause_flags))):
        compute_clause_violation(positive, negative, flag)
